```python

# 依赖: matplotlib, numpy, torch, torchvision
# 安装: pip install matplotlib numpy torch torchvision
import torch
from torch.utils.data import DataLoader
from torchvision import datasets, transforms
import matplotlib.pyplot as plt
import numpy as np

class MNISTDataLoader:
    """MNIST数据集加载器"""

    def __init__(self, batch_size=64, download=True):
        """
        初始化数据加载器

        Args:
            batch_size: 批次大小
            download: 是否下载数据集
        """
        self.batch_size = batch_size

        # 定义数据预处理
        # transforms.Compose将多个变换组合在一起
        self.transform = transforms.Compose([
            # 将PIL Image或numpy.ndarray转换为tensor
            # 并将像素值从[0, 255]缩放到[0, 1]
            transforms.ToTensor(),
            # 标准化：mean和std是根据MNIST训练集计算得出的
            # 这有助于模型更快收敛
            transforms.Normalize((0.1307,), (0.3081,))
        ])

        # 加载训练集
        self.train_dataset = datasets.MNIST(
            root='./data',           # 数据存储路径
            train=True,              # 加载训练集
            download=download,       # 如果不存在则下载
            transform=self.transform # 应用预处理
        )

        # 加载测试集
        self.test_dataset = datasets.MNIST(
            root='./data',
            train=False,             # 加载测试集
            download=download,
            transform=self.transform
        )

        # 创建数据加载器
        self.train_loader = DataLoader(
            self.train_dataset,
            batch_size=batch_size,
            shuffle=True             # 训练时打乱数据
        )

        self.test_loader = DataLoader(
            self.test_dataset,
            batch_size=batch_size,
            shuffle=False            # 测试时不打乱
        )

    def get_batch(self):
        """获取一个批次的数据用于可视化"""
        images, labels = next(iter(self.train_loader))
        return images, labels

    def visualize_batch(self, images, labels, n_rows=4, n_cols=8):
        """
        可视化一个批次的图像

        Args:
            images: 图像tensor [batch_size, 1, 28, 28]
            labels: 标签tensor [batch_size]
            n_rows: 行数
            n_cols: 列数
        """
        fig, axes = plt.subplots(n_rows, n_cols, figsize=(12, 6))
        fig.suptitle('MNIST数据集样本', fontsize=16)

        for i, ax in enumerate(axes.flat):
            if i < len(images):
                # 将tensor转换为numpy数组并去除channel维度
                img = images[i].numpy().squeeze()
                ax.imshow(img, cmap='gray')
                ax.set_title(f'标签: {labels[i].item()}')
                ax.axis('off')
            else:
                ax.axis('off')

        plt.tight_layout()
        plt.savefig('mnist_samples.png', dpi=150, bbox_inches='tight')
        plt.show()

    def get_data_info(self):
        """获取数据集信息"""
        return {
            '训练集大小': len(self.train_dataset),
            '测试集大小': len(self.test_dataset),
            '批次大小': self.batch_size,
            '训练批次数': len(self.train_loader),
            '测试批次数': len(self.test_loader),
            '图像形状': self.train_dataset[0][0].shape,
            '标签范围': '0-9'
        }


# 使用示例
if __name__ == "__main__":

        print(f"{key}: {value}")
```

## 模型构建 - LeNet架构

### 1. LeNet架构说明

LeNet是最早成功的CNN架构之一，由Yann LeCun在1998年提出。它的结构简单清晰，非常适合学习CNN的基本原理。

**LeNet-5架构**：
1. **输入层**: 28×28×1 (灰度图像)
2. **C1卷积层**: 6个5×5卷积核，步长1，padding=0 → 输出24×24×6
3. **S2池化层**: 2×2平均池化，步长2 → 输出12×12×6
4. **C3卷积层**: 16个5×5卷积核 → 输出8×8×16
5. **S4池化层**: 2×2平均池化，步长2 → 输出4×4×16
6. **C5卷积层**: 120个5×5卷积核 → 输出1×1×120
7. **F6全连接层**: 84个神经元
8. **输出层**: 10个神经元（对应10个数字类别）

### 2. 模型代码实现

```python

# 依赖: torch
# 安装: pip install torch
import torch
import torch.nn as nn
import torch.nn.functional as F


class LeNet(nn.Module):
    """LeNet-5卷积神经网络"""

    def __init__(self, num_classes=10):
        """
        初始化LeNet模型

        Args:
            num_classes: 分类类别数
        """
        super(LeNet, self).__init__()

        # 第一个卷积块：卷积层 + 激活 + 池化
        # 输入: 28×28×1
        # Conv2d参数说明：
        #   - in_channels: 输入通道数（灰度图为1）
        #   - out_channels: 输出通道数（卷积核数量）
        #   - kernel_size: 卷积核大小
        #   - stride: 步长
        #   - padding: 填充
        self.conv1 = nn.Conv2d(in_channels=1, out_channels=6,
                               kernel_size=5, stride=1, padding=0)
        # 输出: 24×24×6

        # MaxPool2d: 最大池化层
        # 参数：kernel_size=2, stride=2
        self.pool1 = nn.MaxPool2d(kernel_size=2, stride=2)
        # 输出: 12×12×6

        # 第二个卷积块
        self.conv2 = nn.Conv2d(in_channels=6, out_channels=16,
                               kernel_size=5, stride=1, padding=0)
        # 输出: 8×8×16

        self.pool2 = nn.MaxPool2d(kernel_size=2, stride=2)
        # 输出: 4×4×16

        # 第三个卷积层（这里用全连接层实现，因为特征图已经是4×4）
        self.fc1 = nn.Linear(in_features=16*4*4, out_features=120)
        self.fc2 = nn.Linear(in_features=120, out_features=84)
        self.fc3 = nn.Linear(in_features=84, out_features=num_classes)

    def forward(self, x):
        """
        前向传播

        Args:
            x: 输入tensor [batch_size, 1, 28, 28]

        Returns:
            output: 输出tensor [batch_size, 10]
        """
        # 第一层：卷积 + 激活 + 池化
        # F.relu: ReLU激活函数
        x = self.pool1(F.relu(self.conv1(x)))
        # shape: [batch_size, 6, 12, 12]

        # 第二层：卷积 + 激活 + 池化
        x = self.pool2(F.relu(self.conv2(x)))
        # shape: [batch_size, 16, 4, 4]

        # 展平特征图，为全连接层做准备
        # view()相当于reshape
        x = x.view(-1, 16*4*4)
        # shape: [batch_size, 256]

        # 全连接层 + 激活
        x = F.relu(self.fc1(x))
        # shape: [batch_size, 120]

        x = F.relu(self.fc2(x))
        # shape: [batch_size, 84]

        # 输出层（不需要激活函数，因为会在损失函数中处理）
        x = self.fc3(x)
        # shape: [batch_size, 10]

        return x

    def get_num_params(self):
        """获取模型参数总数"""
        return sum(p.numel() for p in self.parameters() if p.requires_grad)


# 现代化的LeNet变体（使用更多现代技术）
class ModernLeNet(nn.Module):
    """现代化的LeNet，使用BatchNorm和Dropout"""

    def __init__(self, num_classes=10, dropout_rate=0.5):
        super(ModernLeNet, self).__init__()

        # 卷积块：Conv -> BatchNorm -> ReLU -> MaxPool
        self.conv_block1 = nn.Sequential(
            nn.Conv2d(1, 32, kernel_size=5, stride=1, padding=2),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)
        )
        # 输出: 14×14×32

        self.conv_block2 = nn.Sequential(
            nn.Conv2d(32, 64, kernel_size=5, stride=1, padding=2),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)
        )
        # 输出: 7×7×64

        # 全连接层
        self.fc_block = nn.Sequential(
            nn.Linear(64 * 7 * 7, 256),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(256, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_block1(x)
        x = self.conv_block2(x)
        x = x.view(x.size(0), -1)  # 展平
        x = self.fc_block(x)
        return x


# 测试模型

    print(f"\n模型参数总数: {model.get_num_params():,}")
```

## 训练循环

### 1. 训练和评估函数

```python

# 依赖: matplotlib, time, torch, tqdm
# 安装: pip install matplotlib time torch tqdm
import torch
import torch.nn as nn
import torch.optim as optim
from tqdm import tqdm
import time
import matplotlib.pyplot as plt


class Trainer:
    """模型训练器"""

    def __init__(self, model, train_loader, test_loader, device='cuda'):
        """
        初始化训练器

        Args:
            model: 要训练的模型
            train_loader: 训练数据加载器
            test_loader: 测试数据加载器
            device: 训练设备 ('cuda' 或 'cpu')
        """
        self.model = model.to(device)
        self.train_loader = train_loader
        self.test_loader = test_loader
        self.device = device

        # 定义损失函数
        # CrossEntropyLoss = LogSoftmax + NLLLoss
        # 适用于多分类问题
        self.criterion = nn.CrossEntropyLoss()

        # 定义优化器
        # Adam是自适应学习率优化器，通常比SGD收敛更快
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)

        # 学习率调度器
        # 当验证损失不再下降时，降低学习率
        self.scheduler = optim.lr_scheduler.ReduceLROnPlateau(
            self.optimizer, mode='min', factor=0.5, patience=3, verbose=True
        )

        # 训练历史
        self.history = {
            'train_loss': [],
            'train_acc': [],
            'test_loss': [],
            'test_acc': [],
            'lr': []
        }

    def train_epoch(self, epoch):
        """
        训练一个epoch

        Args:
            epoch: 当前epoch数

        Returns:
            avg_loss: 平均损失
            avg_acc: 平均准确率
        """
        self.model.train()  # 设置为训练模式

        running_loss = 0.0
        correct = 0
        total = 0

        # 使用tqdm显示进度条
        pbar = tqdm(self.train_loader, desc=f'Epoch {epoch}')

        for batch_idx, (images, labels) in enumerate(pbar):
            # 将数据移动到GPU
            images, labels = images.to(self.device), labels.to(self.device)

            # 清零梯度
            # PyTorch默认会累积梯度，所以每个batch需要清零
            self.optimizer.zero_grad()

            # 前向传播
            outputs = self.model(images)

            # 计算损失
            loss = self.criterion(outputs, labels)

            # 反向传播
            loss.backward()

            # 更新参数
            self.optimizer.step()

            # 统计
            running_loss += loss.item()
            _, predicted = outputs.max(1)  # 获取预测类别
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()

            # 更新进度条
            pbar.set_postfix({
                'Loss': f'{loss.item():.4f}',
                'Acc': f'{100.*correct/total:.2f}%'
            })

        # 计算平均指标
        avg_loss = running_loss / len(self.train_loader)
        avg_acc = 100. * correct / total

        return avg_loss, avg_acc

    def test_epoch(self):
        """
        在测试集上评估模型

        Returns:
            avg_loss: 平均损失
            avg_acc: 平均准确率
        """
        self.model.eval()  # 设置为评估模式
        test_loss = 0.0
        correct = 0
        total = 0

        # 评估时不需要计算梯度
        with torch.no_grad():
            for images, labels in tqdm(self.test_loader, desc='Testing'):
                images, labels = images.to(self.device), labels.to(self.device)

                # 前向传播
                outputs = self.model(images)

                # 计算损失
                loss = self.criterion(outputs, labels)
                test_loss += loss.item()

                # 统计准确率
                _, predicted = outputs.max(1)
                total += labels.size(0)
                correct += predicted.eq(labels).sum().item()

        avg_loss = test_loss / len(self.test_loader)
        avg_acc = 100. * correct / total

        return avg_loss, avg_acc

    def train(self, num_epochs, save_best=True):
        """
        完整训练流程

        Args:
            num_epochs: 训练轮数
            save_best: 是否保存最佳模型
        """
        best_acc = 0.0

        print(f"\n开始训练，共 {num_epochs} 个epoch")
        print(f"设备: {self.device}")
        print(f"模型参数: {self.model.get_num_params():,}\n")

        start_time = time.time()

        for epoch in range(1, num_epochs + 1):
            # 训练一个epoch
            train_loss, train_acc = self.train_epoch(epoch)

            # 在测试集上评估
            test_loss, test_acc = self.test_epoch()

            # 获取当前学习率
            current_lr = self.optimizer.param_groups[0]['lr']

            # 更新学习率
            self.scheduler.step(test_loss)

            # 记录历史
            self.history['train_loss'].append(train_loss)
            self.history['train_acc'].append(train_acc)
            self.history['test_loss'].append(test_loss)
            self.history['test_acc'].append(test_acc)
            self.history['lr'].append(current_lr)

            # 打印结果
            print(f'\nEpoch {epoch}/{num_epochs}:')
            print(f'Train Loss: {train_loss:.4f}, Train Acc: {train_acc:.2f}%')
            print(f'Test Loss: {test_loss:.4f}, Test Acc: {test_acc:.2f}%')
            print(f'Learning Rate: {current_lr:.6f}')

            # 保存最佳模型
            if save_best and test_acc > best_acc:
                best_acc = test_acc
                self.save_model('best_model.pth')
                print(f'✓ 保存最佳模型 (准确率: {best_acc:.2f}%)')

        # 训练完成
        total_time = time.time() - start_time
        print(f'\n训练完成! 总用时: {total_time/60:.2f}分钟')
        print(f'最佳测试准确率: {best_acc:.2f}%')

    def save_model(self, path):
        """保存模型"""
        torch.save({
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'history': self.history
        }, path)

    def load_model(self, path):
        """加载模型"""
        checkpoint = torch.load(path)
        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
        self.history = checkpoint['history']
```

```python
def plot_training_history(history, save_path='training_history.png'):
    """
    绘制训练历史

    Args:
        history: 训练历史字典
        save_path: 保存路径
    """
    fig, axes = plt.subplots(1, 3, figsize=(18, 5))
```

```python
# 依赖: data_loader, model, torch, train
# 安装: pip install data_loader model torch train
import torch
from data_loader import MNISTDataLoader

def main():
    # 设置随机种子以保证可重复性
    torch.manual_seed(42)
    if torch.cuda.is_available():
        torch.cuda.manual_seed(42)

    # 设置设备
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    # 1. 加载数据

    trainer = Trainer(
```

```python

# 依赖: matplotlib, numpy, seaborn, torch
# 安装: pip install matplotlib numpy seaborn torch
import torch
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import confusion_matrix, classification_report
import seaborn as sns


class ModelEvaluator:
    """模型评估器"""

    def __init__(self, model, test_loader, device='cuda'):
        self.model = model.to(device)
        self.model.eval()
        self.test_loader = test_loader
        self.device = device

    def predict(self, images):
        """预测图像类别"""
        with torch.no_grad():
            images = images.to(self.device)
            outputs = self.model(images)
            _, predicted = outputs.max(1)
            return predicted.cpu().numpy()

    def evaluate_all(self):
        """在所有测试数据上评估"""
        all_predictions = []
        all_labels = []

        for images, labels in self.test_loader:
            predictions = self.predict(images)
            all_predictions.extend(predictions)
            all_labels.extend(labels.numpy())

        all_predictions = np.array(all_predictions)
        all_labels = np.array(all_labels)

        # 计算准确率
        accuracy = (all_predictions == all_labels).mean() * 100

        return all_predictions, all_labels, accuracy

    def plot_confusion_matrix(self, predictions, labels, save_path='confusion_matrix.png'):
        """绘制混淆矩阵"""
        cm = confusion_matrix(labels, predictions)

        plt.figure(figsize=(10, 8))
        sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                    xticklabels=range(10), yticklabels=range(10))
        plt.xlabel('预测标签', fontsize=12)
        plt.ylabel('真实标签', fontsize=12)
        plt.title('混淆矩阵', fontsize=14)
        plt.tight_layout()
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        plt.show()

        print(f'混淆矩阵已保存至: {save_path}')

    def print_classification_report(self, predictions, labels):
        """打印分类报告"""
        print("\n=== 分类报告 ===")
        print(classification_report(labels, predictions,
                                    target_names=[f'数字{i}' for i in range(10)],
                                    digits=4))

    def visualize_predictions(self, n_rows=5, n_cols=5,
                             save_path='predictions.png'):
        """可视化预测结果"""
        fig, axes = plt.subplots(n_rows, n_cols, figsize=(12, 12))
        fig.suptitle('模型预测结果', fontsize=16)

        # 获取一个批次的数据
        images, labels = next(iter(self.test_loader))
        predictions = self.predict(images)

        for i, ax in enumerate(axes.flat):
            if i < len(images):
                img = images[i].numpy().squeeze()
                true_label = labels[i].item()
                pred_label = predictions[i]

                # 设置标题颜色
                color = 'green' if true_label == pred_label else 'red'

                ax.imshow(img, cmap='gray')
                ax.set_title(f'真实: {true_label}\n预测: {pred_label}',
                            color=color, fontsize=10)
                ax.axis('off')
            else:
                ax.axis('off')

        plt.tight_layout()
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        plt.show()

        print(f'预测结果已保存至: {save_path}')

    def show_misclassified(self, num_samples=20, save_path='misclassified.png'):
        """展示错误分类的样本"""
        misclassified_images = []
        misclassified_true = []
        misclassified_pred = []

        # 收集错误分类的样本
        for images, labels in self.test_loader:
            predictions = self.predict(images)

            for i in range(len(images)):
                if predictions[i] != labels[i].item():
                    misclassified_images.append(images[i])
                    misclassified_true.append(labels[i].item())
                    misclassified_pred.append(predictions[i])

                    if len(misclassified_images) >= num_samples:
                        break

            if len(misclassified_images) >= num_samples:
                break

        # 可视化
        n_cols = 5
        n_rows = (len(misclassified_images) + n_cols - 1) // n_cols

        fig, axes = plt.subplots(n_rows, n_cols, figsize=(15, 3*n_rows))
        fig.suptitle('错误分类样本分析', fontsize=16)

        for i, ax in enumerate(axes.flat):
            if i < len(misclassified_images):
                img = misclassified_images[i].numpy().squeeze()
                ax.imshow(img, cmap='gray')
                ax.set_title(f'真实: {misclassified_true[i]}\n预测: {misclassified_pred[i]}',
                            color='red', fontsize=10)
                ax.axis('off')
            else:
                ax.axis('off')

        plt.tight_layout()
        plt.savefig(save_path, dpi=150, bbox_inches='tight')
        plt.show()

        print(f'错误分类样本已保存至: {save_path}')
```

```python

# 依赖: PIL, matplotlib, numpy, torch
# 安装: pip install PIL matplotlib numpy torch
import torch
import numpy as np
from PIL import Image
import matplotlib.pyplot as plt


class DigitPredictor:
    """手写数字预测器"""

    def __init__(self, model_path, device='cuda'):
        """
        初始化预测器

        Args:
            model_path: 模型文件路径
            device: 推理设备
        """
        self.device = torch.device(device if torch.cuda.is_available() else 'cpu')

        # 加载模型
        from model import ModernLeNet
        self.model = ModernLeNet(num_classes=10).to(self.device)
        checkpoint = torch.load(model_path, map_location=self.device)
        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.model.eval()

        # 图像预处理
        from torchvision import transforms
        self.transform = transforms.Compose([
            transforms.Grayscale(),  # 转为灰度图
            transforms.Resize(28),   # 调整大小
            transforms.ToTensor(),
            transforms.Normalize((0.1307,), (0.3081,))
        ])

    def predict_image(self, image_path):
        """
        预测单张图像

        Args:
            image_path: 图像路径

        Returns:
            prediction: 预测的数字
            confidence: 置信度
        """
        # 加载图像
        image = Image.open(image_path)

        # 预处理
        image_tensor = self.transform(image).unsqueeze(0).to(self.device)

        # 预测
        with torch.no_grad():
            output = self.model(image_tensor)
            probabilities = torch.softmax(output, dim=1)
            confidence, prediction = probabilities.max(1)

        return prediction.item(), confidence.item()

    def predict_batch(self, image_paths):
        """批量预测"""
        results = []
        for path in image_paths:
            pred, conf = self.predict_image(path)
            results.append({
                'path': path,
                'prediction': pred,
                'confidence': conf
            })
        return results


# 使用示例
if __name__ == "__main__":
    predictor = DigitPredictor('best_model.pth')

    # 预测单张图像
    image_path = 'test_digit.png'  # 替换为你的图像路径
    prediction, confidence = predictor.predict_image(image_path)

    print(f"预测结果: {prediction}")
    print(f"置信度: {confidence*100:.2f}%")
```
