## 图像噪声与滤波基础

### 1.1 什么是图像噪声？

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.ndimage import gaussian_filter, median_filter, convolve

print("=== 图像噪声基础 ===")

# 创建干净的测试图像
def create_test_pattern():
    """创建测试图案"""
    img = np.zeros((100, 100), dtype=np.uint8)
    # 绘制方块
    img[20:40, 20:40] = 100
    img[60:80, 20:40] = 150
    img[20:40, 60:80] = 200
    img[60:80, 60:80] = 255
    # 绘制线条
    img[45:55, :] = 128
    img[:, 45:55] = 128
    return img

# 1. 高斯噪声（Gaussian Noise）
def add_gaussian_noise(img, mean=0, sigma=25):
    """添加高斯噪声"""
    noise = np.random.normal(mean, sigma, img.shape)
    noisy = img.astype(np.float32) + noise
    return np.clip(noisy, 0, 255).astype(np.uint8)

# 2. 椒盐噪声（Salt & Pepper Noise）
def add_salt_pepper_noise(img, salt_prob=0.02, pepper_prob=0.02):
    """添加椒盐噪声"""
    noisy = img.copy()
    # 盐噪声（白点）
    salt_mask = np.random.random(img.shape) < salt_prob
    noisy[salt_mask] = 255
    # 椒噪声（黑点）
    pepper_mask = np.random.random(img.shape) < pepper_prob
    noisy[pepper_mask] = 0
    return noisy

# 3. 泊松噪声（Poisson Noise）
def add_poisson_noise(img):
    """添加泊松噪声（模拟光子噪声）"""
    # 归一化到0-1
    img_normalized = img.astype(np.float32) / 255.0
    # 应用泊松分布
    noisy = np.random.poisson(img_normalized * 50.0) / 50.0
    return np.clip(noisy * 255, 0, 255).astype(np.uint8)

# 4. 斑点噪声（Speckle Noise）
def add_speckle_noise(img, std=0.2):
    """添加斑点噪声"""
    noise = np.random.normal(0, std, img.shape)
    noisy = img.astype(np.float32) * (1 + noise)
    return np.clip(noisy, 0, 255).astype(np.uint8)

# 生成噪声图像
clean = create_test_pattern()
gaussian_noisy = add_gaussian_noise(clean, sigma=30)
sp_noisy = add_salt_pepper_noise(clean, salt_prob=0.03, pepper_prob=0.03)
poisson_noisy = add_poisson_noise(clean)
speckle_noisy = add_speckle_noise(clean)

# 可视化
fig, axes = plt.subplots(2, 3, figsize=(15, 10))
axes = axes.flatten()

images = [clean, gaussian_noisy, sp_noisy, poisson_noisy, speckle_noisy]
titles = ['原始', '高斯噪声\n(σ=30)', '椒盐噪声\n(3%)', '泊松噪声', '斑点噪声']

for i, (img, title) in enumerate(zip(images, titles)):
    axes[i].imshow(img, cmap='gray', vmin=0, vmax=255)
    axes[i].set_title(title)
    axes[i].axis('off')

# 显示直方图
axes[5].hist(gaussian_noisy.ravel(), bins=50, range=(0, 256), alpha=0.7)
axes[5].set_title('高斯噪声直方图')
axes[5].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

print("\n噪声类型总结：")
print("高斯噪声：随机分布，常见于传感器")
print("椒盐噪声：黑白点，常见于传输错误")
print("泊松噪声：光子计数噪声，低光照常见")
print("斑点噪声：乘性噪声，常见于雷达/超声")
```

```python
        [0, 0, 0, 0, 0],
```

```python
print("\n=== 高斯滤波器 ===")

def gaussian_filter_custom(img, kernel_size=5, sigma=1.0):
    """自定义高斯滤波器"""
    # 生成高斯核
    ax = np.linspace(-(kernel_size-1)/2, (kernel_size-1)/2, kernel_size)
```

```python
print("\n=== 中值滤波器 ===")

def median_filter_custom(img, kernel_size=3):
    """中值滤波器"""
    h, w = img.shape
    pad = kernel_size // 2
    padded = np.pad(img, pad, mode='reflect')
```

### 2.4 双边滤波器

```python
print("\n=== 双边滤波器 ===")

def bilateral_filter_custom(img, d=9, sigma_color=50, sigma_space=15):
    """简化版双边滤波器"""
    h, w = img.shape
    result = np.zeros_like(img, dtype=np.float32)

    # 空间高斯核
    pad = d // 2
    ax = np.linspace(-(pad), pad, d)
    xx, yy = np.meshgrid(ax, ax)
    spatial_kernel = np.exp(-0.5 * (xx**2 + yy**2) / sigma_space**2)

    padded = np.pad(img, pad, mode='reflect')

    for i in range(h):
        for j in range(w):
            # 提取邻域
            region = padded[i:i+d, j:j+d]

            # 颜色相似度
            center_val = img[i, j]
            color_sim = np.exp(-0.5 * ((region - center_val) ** 2) / sigma_color**2)

            # 组合权重
            weights = spatial_kernel * color_sim
            weights_sum = weights.sum()

            # 加权平均
            if weights_sum > 0:
                result[i, j] = np.sum(weights * region) / weights_sum
            else:
                result[i, j] = img[i, j]

    return result.astype(np.uint8)

# 创建边缘丰富的测试图像
```

```python
def fourier_transform_demo():
    """傅里叶变换演示"""

    # 创建不同频率的图像
    h, w = 100, 100
    x = np.arange(w)
    y = np.arange(h)
    xx, yy = np.meshgrid(x, y)

    # 低频图像（缓慢变化）
    low_freq = np.sin(2 * np.pi * xx / 50) * 128 + 128

    # 高频图像（快速变化）
    high_freq = np.sin(2 * np.pi * xx / 5) * 128 + 128

    # 混合图像
    mixed = low_freq * 0.5 + high_freq * 0.5

    # 计算傅里叶变换
    def compute_fft(img):
        f = np.fft.fft2(img)
        fshift = np.fft.fftshift(f)
        magnitude = np.log(np.abs(fshift) + 1)
        return magnitude

    fig, axes = plt.subplots(2, 3, figsize=(15, 10))

    images = [low_freq, high_freq, mixed]
    titles = ['低频图像\n(缓慢变化)', '高频图像\n(快速变化)', '混合图像']

        axes[0, i].set_title(title)
        axes[0, i].axis('off')
```

### 3.2 理想低通/高通滤波器

```python
print("\n=== 频域滤波器 ===")

def frequency_filter(img, cutoff, filter_type='lowpass'):
    """频域滤波器"""
    # FFT
    f = np.fft.fft2(img)
    fshift = np.fft.fftshift(f)

    h, w = img.shape
    crow, ccol = h // 2, w // 2

    # 创建掩码
    y, x = np.ogrid[:h, :w]
    distance = np.sqrt((x - ccol)**2 + (y - crow)**2)

    if filter_type == 'lowpass':
        mask = distance <= cutoff
    else:  # highpass
        mask = distance > cutoff

    # 应用掩码
    fshift_filtered = fshift * mask

    # 逆FFT
    f_filtered = np.fft.ifftshift(fshift_filtered)
    img_filtered = np.fft.ifft2(f_filtered)
    img_filtered = np.abs(img_filtered)

    return img_filtered

# 测试频域滤波
test_img = create_test_pattern()
noisy_test = add_gaussian_noise(test_img, sigma=20)

# 应用频域滤波

axes[0, 0].set_title('含噪图像')

axes[0, 1].axis('off')
```

### 4.1 对比度调整

```python
print("\n=== 对比度调整 ===")

def contrast_stretch(img, min_out=0, max_out=255):
    """线性对比度拉伸"""
    min_in = img.min()
    max_in = img.max()

    if max_in == min_in:
        return np.zeros_like(img)

    stretched = (img - min_in) * (max_out - min_out) / (max_in - min_in) + min_out
    return stretched.astype(np.uint8)

def gamma_correction(img, gamma=1.0):
    """伽马校正"""
    normalized = img / 255.0
    corrected = np.power(normalized, gamma)
    return (corrected * 255).astype(np.uint8)

# 创建低对比度图像
low_contrast = np.random.randint(80, 180, (100, 100), dtype=np.uint8)

# 应用增强
stretched = contrast_stretch(low_contrast)
gamma_low = gamma_correction(low_contrast, 0.5)  # 增亮暗部
gamma_high = gamma_correction(low_contrast, 2.0)  # 压暗亮部

fig, axes = plt.subplots(2, 4, figsize=(16, 8))

# 显示图像
images = [low_contrast, stretched, gamma_low, gamma_high]
titles = ['原始\n(低对比度)', '线性拉伸', '伽马=0.5\n(增亮)', '伽马=2.0\n(压暗)']

for i, (img, title) in enumerate(zip(images, titles)):
    axes[0, i].imshow(img, cmap='gray', vmin=0, vmax=255)
    axes[0, i].set_title(title)
    axes[0, i].axis('off')
```

```python
def histogram_equalization(img):
    """直方图均衡化"""
    # 计算直方图
    hist, bins = np.histogram(img.ravel(), 256, [0, 256])

    # 计算累积分布函数（CDF）
    cdf = hist.cumsum()
    cdf_normalized = cdf * hist.max() / cdf.max()

    # 归一化到0-255
    cdf_m = np.ma.masked_equal(cdf, 0)
    cdf_m = (cdf_m - cdf_m.min()) * 255 / (cdf_m.max() - cdf_m.min())
    cdf = np.ma.filled(cdf_m, 0).astype(np.uint8)

    # 应用映射
    img_eq = cdf[img]

    return img_eq, cdf
```

```python
print("\n=== 自适应直方图均衡化 ===")

def clahe_simple(img, clip_limit=2.0, tile_size=8):
    """简化版CLAHE"""
    h, w = img.shape
    result = np.zeros_like(img, dtype=np.float32)

    # 分块处理
    for i in range(0, h, tile_size):
        for j in range(0, w, tile_size):
            # 提取块
            i_end = min(i + tile_size, h)
            j_end = min(j + tile_size, w)
            tile = img[i:i_end, j:j_end]

            # 块内均衡化
            tile_eq, _ = histogram_equalization(tile)

            # 限制对比度
            hist, _ = np.histogram(tile_eq.ravel(), 256, [0, 256])
            excess = hist.sum() * (clip_limit - 1) / clip_limit
            hist = np.minimum(hist, clip_limit * 256 / hist.sum())

            # 重新分配
            redistributed = excess / 256
            hist += redistributed

            # 应用新直方图（简化）
            result[i:i_end, j:j_end] = tile_eq

    return result.astype(np.uint8)
```

```python
print("\n=== Canny边缘检测 ===")

def canny_edge_detector(img, low_threshold=30, high_threshold=70):
    """简化版Canny边缘检测"""

    # 1. 高斯平滑
    img_smooth = gaussian_filter(img, sigma=1.4)

    # 2. 计算梯度（Sobel）
    grad_x = convolve(img_smooth.astype(np.float32), sobel_x)
    grad_y = convolve(img_smooth.astype(np.float32), sobel_y)

    magnitude = np.sqrt(grad_x**2 + grad_y**2)
    direction = np.arctan2(grad_y, grad_x) * 180 / np.pi

    # 3. 非极大值抑制
    suppressed = np.zeros_like(magnitude)
    for i in range(1, img.shape[0]-1):
        for j in range(1, img.shape[1]-1):
            angle = direction[i, j]
            mag = magnitude[i, j]

            # 确定方向
            if (angle < -157.5 or angle >= 157.5) or (angle >= -22.5 and angle < 22.5):
                neighbors = [magnitude[i, j-1], magnitude[i, j+1]]
            elif (angle >= -67.5 and angle < -22.5) or (angle >= 112.5 and angle < 157.5):
                neighbors = [magnitude[i-1, j+1], magnitude[i+1, j-1]]
            elif (angle >= -112.5 and angle < -67.5) or (angle >= 67.5 and angle < 112.5):
                neighbors = [magnitude[i-1, j], magnitude[i+1, j]]
            else:
                neighbors = [magnitude[i-1, j-1], magnitude[i+1, j+1]]

            if mag >= max(neighbors):
                suppressed[i, j] = mag

    # 4. 双阈值滞后
    strong = suppressed >= high_threshold
    weak = (suppressed >= low_threshold) & (suppressed < high_threshold)

    result = strong.copy()

    # 连接弱边缘
    from scipy.ndimage import label
    labeled, num_features = label(weak)

    for i in range(1, num_features + 1):
        if np.any(strong[labeled == i]):
            result[labeled == i] = True

    return result.astype(np.uint8) * 255
```

### 核心技术回顾

#### 1. **噪声去除**
- ✅ **均值滤波**：简单快速，模糊边缘
- ✅ **高斯滤波**：自然平滑，保留细节
- ✅ **中值滤波**：去椒盐噪声，保留边缘
- ✅ **双边滤波**：去噪+保边，计算复杂

#### 2. **频域处理**
- ✅ **傅里叶变换**：空间域→频域
- ✅ **低通滤波**：去噪、模糊
- ✅ **高通滤波**：边缘提取

#### 3. **图像增强**
- ✅ **对比度拉伸**：线性扩展
- ✅ **伽马校正**：非线性调整
- ✅ **直方图均衡**：全局增强
- ✅ **CLAHE**：局部自适应

#### 4. **边缘检测**
- ✅ **一阶算子**：Sobel、Prewitt、Roberts
- ✅ **Canny**：多步骤，高质量
