```python
print("\n步骤1：边缘检测")

# 1. 高斯模糊去噪

blurred = gaussian_filter(document, sigma=1.5)

# 2. Canny边缘检测

def canny_edge(img, low=50, high=150):
    """简化Canny"""
    # Sobel梯度
    gx = convolve(img.astype(np.float32), np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]))
    gy = convolve(img.astype(np.float32), np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]]))
    mag = np.sqrt(gx**2 + gy**2)

    # 阈值
    edges = np.zeros_like(mag)
    edges[(mag >= low) & (mag <= high)] = 128
    edges[mag > high] = 255

    return edges

edges = canny_edge(blurred, low=50, high=100)

# 3. 轮廓检测（简化版）

def find_largest_contour(edges):
    """找到最大的轮廓"""
    from scipy.ndimage import label, binary_dilation

    # 二值化
    binary = edges > 0

    # 连通域
    labeled, num = label(binary)

    # 找最大连通域
    max_area = 0
    max_label = 0
    for i in range(1, num+1):
        area = np.sum(labeled == i)
        if area > max_area:
            max_area = area
            max_label = i

    # 提取边界
    mask = labeled == max_label
    dilated = binary_dilation(mask)
    boundary = dilated & ~mask

    y, x = np.where(boundary)
    return list(zip(x, y))

contour_points = find_largest_contour(edges)
```

```python
print("\n步骤2：透视校正")

def perspective_transform(img, src_points):
    """透视变换校正"""
    if len(src_points) < 4:
        return img

    # 简化：找到四个角点
    # 1. 按x排序找左右
    sorted_x = sorted(src_points, key=lambda p: p[0])
    left = sorted_x[:len(sorted_x)//2]
    right = sorted_x[len(sorted_x)//2:]

    # 2. 每侧按y排序找上下
    left_sorted = sorted(left, key=lambda p: p[1])
    right_sorted = sorted(right, key=lambda p: p[1])

    # 3. 取近似角点
    src_corners = np.array([
        left_sorted[0],   # 左上
        right_sorted[0],  # 右上
        right_sorted[-1], # 右下
        left_sorted[-1]   # 左下
    ], dtype=np.float32)

    # 目标矩形
    h, w = img.shape
    dst_corners = np.array([
        [50, 50],
        [w-50, 50],
        [w-50, h-50],
        [50, h-50]
    ], dtype=np.float32)

    # 计算变换矩阵
    M = cv2.getPerspectiveTransform(src_corners, dst_corners)

    # 应用变换
    corrected = cv2.warpPerspective(img, M, (w, h))

    return corrected
```

```python
def adaptive_threshold(img, block_size=15, C=5):
    """自适应阈值"""
    h, w = img.shape
    binary = np.zeros_like(img)

    pad = block_size // 2
    padded = np.pad(img, pad, mode='reflect')

    for i in range(h):
        for j in range(w):
            block = padded[i:i+block_size, j:j+block_size]
            threshold = np.mean(block) - C
            binary[i, j] = 255 if img[i, j] > threshold else 0

    return binary
```

### 2.2 Haar-like特征检测（简化版）

```python
print("\n人脸特征检测")

def haar_like_features(img):
    """简化版Haar-like特征检测"""
    h, w = img.shape

    # 积分图
    integral = np.cumsum(np.cumsum(img, axis=0), axis=1)

    def get_sum(x1, y1, x2, y2):
        if x1 < 0 or y1 < 0 or x2 >= w or y2 >= h:
            return 0
        a = integral[y2, x2]
        b = integral[y1-1, x2] if y1 > 0 else 0
        c = integral[y2, x1-1] if x1 > 0 else 0
        d = integral[y1-1, x1-1] if (x1 > 0 and y1 > 0) else 0
        return a - b - c + d

    # 检测眼睛区域（上亮下暗）
    eye_scores = []
    for y in range(20, h-40):
        for x in range(20, w-40):
            # 2x1矩形特征
            top = get_sum(x, y, x+20, y+10)
            bottom = get_sum(x, y+10, x+20, y+20)
            score = top - bottom
            if score > 500:  # 阈值
                eye_scores.append((x, y, score))

    # 非极大值抑制
    eye_scores = sorted(eye_scores, key=lambda p: p[2], reverse=True)
    eyes = []
    for score in eye_scores:
        x, y, s = score
        if not any(abs(ex-x) < 20 and abs(ey-y) < 20 for ex, ey, _ in eyes):
            eyes.append(score)

    return eyes[:2]  # 最多两个眼睛

eyes = haar_like_features(face_img)

# 可视化

    plt.gca().add_patch(rect)
    plt.text(x, y-5, f'{score:.0f}', color='red', fontsize=8)
```

### 2.3 美化处理

```python
print("\n美化处理")

def beautify_face(img, eyes):
    """人脸美化"""
    result = img.copy().astype(np.float32)

    # 1. 皮肤平滑（双边滤波简化版）
    def bilateral_smooth(img, d=7, sigma_color=30, sigma_space=10):
        # 简化：高斯+边缘保持
        smooth = gaussian_filter(img, sigma=1.5)

        # 边缘权重
        grad = np.gradient(img)
        edge_weight = 1 / (1 + grad[0]**2 + grad[1]**2 + 1e-6)

        # 混合
        result = img * edge_weight + smooth * (1 - edge_weight)
        return result

    result = bilateral_smooth(result)

    # 2. 亮度调整（眼睛区域）
    if len(eyes) >= 2:
        for x, y, _ in eyes:
            mask = np.zeros_like(result)
            for i in range(max(0, y-5), min(img.shape[0], y+25)):
                for j in range(max(0, x-5), min(img.shape[1], x+25)):
                    dist = np.sqrt((i-y-10)**2 + (j-x-10)**2)
                    if dist < 15:
                        mask[i, j] = np.exp(-dist**2 / 50)

            # 增亮
            result += mask * 20

    # 3. 对比度增强
    result = (result - result.min()) / (result.max() - result.min()) * 255

    return result.astype(np.uint8)

if len(eyes) > 0:
```

```python
# 创建模拟车牌图像

def create_license_plate():
    """创建模拟车牌"""
    img = np.ones((80, 200), dtype=np.uint8) * 120

    # 车牌区域
    img[20:60, 40:160] = 200

    # 边框
    img[20:25, 40:160] = 50
    img[55:60, 40:160] = 50
    img[20:60, 40:45] = 50
    img[20:60, 155:160] = 50

    # 模拟文字
    for i in range(3):
        img[30:50, 55+i*30:75+i*30] = 50

    # 添加噪声和模糊
    img = gaussian_filter(img, sigma=1)
    noise = np.random.normal(0, 8, img.shape)
    img = img + noise.astype(np.uint8)
    img = np.clip(img, 0, 255)

    return img

plate_img = create_license_plate()

# 车牌定位流程

def locate_plate(img):
    """定位车牌"""
    # 1. 灰度化
    gray = img

    # 2. Sobel边缘检测（水平边缘）
    sobel_x = convolve(gray.astype(np.float32), np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]))
    edges = np.abs(sobel_x)

    # 3. 二值化
    binary = edges > 50

    # 4. 形态学操作（连接水平区域）
    kernel = np.ones((1, 15), dtype=bool)
    from scipy.ndimage import binary_closing
    closed = binary_closing(binary, structure=kernel)

    # 5. 连通域分析
    from scipy.ndimage import label
    labeled, num = label(closed)

    # 6. 筛选候选区域
    candidates = []
    for i in range(1, num+1):
        mask = labeled == i
        y, x = np.where(mask)
        if len(x) == 0:
            continue

        width = x.max() - x.min()
        height = y.max() - y.min()
        area = len(x)

        # 车牌比例特征
        aspect = width / height if height > 0 else 0
        if 2 < aspect < 6 and width > 30 and height > 10:
            candidates.append({
                'x': x.min(),
                'y': y.min(),
                'w': width,
                'h': height,
                'area': area
            })

    return candidates, binary, labeled
```

```python
def match_features(feat1, feat2, threshold=50):
    """匹配特征"""
    matches = []

    for i, f1 in enumerate(feat1):
        best_dist = float('inf')
        best_idx = -1

        for j, f2 in enumerate(feat2):
            # 计算描述子距离
            dist = np.linalg.norm(f1['desc'] - f2['desc'])

            if dist < best_dist:
                best_dist = dist
                best_idx = j

        if best_dist < threshold:
            matches.append((i, best_idx, best_dist))

    return matches
```

```python
def simple_stitch(img1, img2, offset_x=50):
    """简单图像拼接"""
    h1, w1 = img1.shape
    h2, w2 = img2.shape

    # 计算拼接后尺寸
    new_w = w1 + offset_x + w2
    new_h = max(h1, h2)

    # 创建拼接图像
    stitched = np.zeros((new_h, new_w), dtype=np.uint8)

    # 放入左图
    stitched[:h1, :w1] = img1

    # 放入右图
    stitched[:h2, w1+offset_x:w1+offset_x+w2] = img2

    # 简单融合（重叠区域平均）
    overlap_start = w1 - 20
    overlap_end = w1 + offset_x + 20

    if overlap_end > w1 and overlap_start < w1+offset_x:
        # 重叠区域
        for y in range(min(h1, h2)):
            for x in range(max(overlap_start, 0), min(overlap_end, new_w)):
                if x < w1 and x >= 0:
                    val1 = img1[y, x] if y < h1 else 0
                else:
                    val1 = 0

                if x >= w1+offset_x and x-w1-offset_x < w2:
                    val2 = img2[y, x-w1-offset_x] if y < h2 else 0
                else:
                    val2 = 0

                if val1 > 0 and val2 > 0:
                    stitched[y, x] = (val1 + val2) // 2
                elif val1 > 0:
                    stitched[y, x] = val1
                elif val2 > 0:
                    stitched[y, x] = val2

    return stitched
```

```python
class SmartMonitor:
    """智能监控系统"""

    def __init__(self):
        self.background = None
        self.history = []

    def set_background(self, frame):
        """设置背景"""
        self.background = gaussian_filter(frame, sigma=2)

    def detect_motion(self, frame, threshold=30):
        """运动检测"""
        if self.background is None:
            return False, None

        # 计算差异
        diff = np.abs(frame.astype(np.float32) - self.background.astype(np.float32))

        # 阈值化
        motion_mask = diff > threshold

        # 形态学清理
        from scipy.ndimage import binary_opening, binary_closing
        motion_mask = binary_opening(motion_mask, structure=np.ones((3, 3)))
        motion_mask = binary_closing(motion_mask, structure=np.ones((5, 5)))

        # 检查是否有运动
        has_motion = np.any(motion_mask)

        return has_motion, motion_mask

    def detect_anomaly(self, frame):
        """异常检测"""
        # 统计特征
        mean = np.mean(frame)
        std = np.std(frame)

        # 历史记录
        self.history.append({'mean': mean, 'std': std})

        # 保持最近100帧
        if len(self.history) > 100:
            self.history.pop(0)

        # 异常判断
        if len(self.history) > 10:
            hist_means = [h['mean'] for h in self.history]
            hist_stds = [h['std'] for h in self.history]

            # 偏离历史均值
            mean_dev = abs(mean - np.mean(hist_means))
            std_dev = abs(std - np.mean(hist_stds))

            # 阈值
            if mean_dev > 20 or std_dev > 10:
                return True, mean_dev, std_dev

        return False, 0, 0

    def process_frame(self, frame):
        """处理一帧"""
        results = {}

        # 1. 运动检测
        has_motion, motion_mask = self.detect_motion(frame)
        results['motion'] = has_motion

        # 2. 异常检测
        is_anomaly, mean_dev, std_dev = self.detect_anomaly(frame)
        results['anomaly'] = is_anomaly
        results['mean_dev'] = mean_dev
        results['std_dev'] = std_dev

        # 3. 特征提取
        features = {
            'mean': np.mean(frame),
            'std': np.std(frame),
            'min': np.min(frame),
            'max': np.max(frame)
        }
        results['features'] = features

        return results, motion_mask
```
