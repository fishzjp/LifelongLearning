## 图像的本质：数字矩阵

### 1.1 什么是数字图像？

想象你用手机拍了一张照片。这张照片在计算机眼中是什么？

**答案：一个巨大的数字表格（矩阵）！**

```python

# 依赖: matplotlib, numpy
# 安装: pip install matplotlib numpy
import numpy as np
import matplotlib.pyplot as plt

# 创建一个最简单的5x5灰度图像
# 0=黑色，255=白色，中间值=灰色

simple_image = np.array([
    [0,   0,   0,   0,   0],
    [0, 255, 255, 255,   0],
    [0, 255, 255, 255,   0],
    [0, 255, 255, 255,   0],
    [0,   0,   0,   0,   0]
], dtype=np.uint8)

print("图像的数值表示：")
print(simple_image)
print(f"\n图像形状：{simple_image.shape}")
print(f"数据类型：{simple_image.dtype}")
print(f"像素值范围：{simple_image.min()} - {simple_image.max()}")

# 可视化

plt.figure(figsize=(8, 4))

# 左边显示矩阵数值

plt.subplot(1, 2, 1)
plt.text(0.5, 0.5, str(simple_image),
         ha='center', va='center', fontfamily='monospace', fontsize=10,
         bbox=dict(boxstyle='round', facecolor='white', alpha=0.8))
plt.axis('off')
plt.title('矩阵数值')

# 右边显示实际图像

plt.subplot(1, 2, 2)
plt.imshow(simple_image, cmap='gray', vmin=0, vmax=255)
plt.title('实际图像')
plt.axis('off')

plt.tight_layout()
plt.show()
```

**关键概念：**
- **像素（Pixel）**：图像的基本单位，每个数字代表一个像素的亮度
- **矩阵形状**：(高度, 宽度) - 注意：先高度后宽度
- **数据类型**：通常用`uint8`（0-255的整数）

### 1.2 灰度图像 vs 彩色图像

#### 灰度图像

```bash
# 创建一个渐变的灰度图像

height, width = 100, 200
gray_gradient = np.zeros((height, width), dtype=np.uint8)
```


**重要概念：**
- **灰度图像**：形状为 `(H, W)`，单通道
- **彩色图像**：形状为 `(H, W, 3)`，三通道（RGB）
- **通道顺序**：OpenCV使用BGR，matplotlib使用RGB

def resize_nearest(img, new_h, new_w):
    """最近邻插值缩放"""
    h, w = img.shape[:2]
    y = np.linspace(0, h-1, new_h).astype(int)
    x = np.linspace(0, w-1, new_w).astype(int)
    return img[np.ix_(y, x)]

# 方法2：使用scipy（双线性插值）

from scipy.ndimage import zoom

### 4.2 灰度空间

```python
print("=== 灰度空间 ===")

# RGB转灰度的原理

def rgb_to_gray_weighted(rgb_img):
    """加权平均法：人眼对绿光最敏感"""
    return (0.299 * rgb_img[:, :, 0] +
            0.587 * rgb_img[:, :, 1] +
            0.114 * rgb_img[:, :, 2]).astype(np.uint8)

def rgb_to_gray_average(rgb_img):
    """简单平均法"""
    return rgb_img.mean(axis=2).astype(np.uint8)

def rgb_to_gray_luminosity(rgb_img):
    """亮度法"""
    return (0.2126 * rgb_img[:, :, 0] +
            0.7152 * rgb_img[:, :, 1] +
            0.0722 * rgb_img[:, :, 2]).astype(np.uint8)
```

### 4.3 HSV颜色空间

```python
print("=== HSV颜色空间 ===")
print("H: Hue（色相）- 颜色的类型")
print("S: Saturation（饱和度）- 颜色的纯度")
print("V: Value（明度）- 颜色的亮度")

def rgb_to_hsv(rgb_img):
    """RGB转HSV（简化版本）"""
    rgb_img = rgb_img.astype(np.float32) / 255.0
    r, g, b = rgb_img[:, :, 0], rgb_img[:, :, 1], rgb_img[:, :, 2]

    cmax = np.maximum.reduce([r, g, b])
    cmin = np.minimum.reduce([r, g, b])
    delta = cmax - cmin

    # 计算H
    h = np.zeros_like(r)
    mask = delta != 0
    # Red
    mask_r = mask & (cmax == r)
    h[mask_r] = 60 * (((g[mask_r] - b[mask_r]) / delta[mask_r]) % 6)
    # Green
    mask_g = mask & (cmax == g)
    h[mask_g] = 60 * ((b[mask_g] - r[mask_g]) / delta[mask_g] + 2)
    # Blue
    mask_b = mask & (cmax == b)
    h[mask_b] = 60 * ((r[mask_b] - g[mask_b]) / delta[mask_b] + 4)

    # 计算S
    s = np.where(cmax != 0, delta / cmax, 0)

    # 计算V
    v = cmax

    return np.stack([h, s, v], axis=2)
```

```python
print("=== 图像旋转 ===")

def rotate_image(img, angle, center=None):
    """图像旋转：angle为角度（顺时针为正）"""
    h, w = img.shape[:2]

    if center is None:
        center = (w // 2, h // 2)

    # 转换为弧度
    angle_rad = np.radians(angle)

    # 旋转矩阵
    cos_a = np.cos(angle_rad)
    sin_a = np.sin(angle_rad)

    # 计算旋转后的边界
    new_w = int(abs(h * sin_a) + abs(w * cos_a))
    new_h = int(abs(h * cos_a) + abs(w * sin_a))

    # 调整中心
    center_x = w // 2
    center_y = h // 2
    new_center_x = new_w // 2
    new_center_y = new_h // 2

    # 创建输出
    rotated = np.zeros((new_h, new_w, 3), dtype=np.uint8)

    # 旋转映射
    for i in range(new_h):
        for j in range(new_w):
            # 反向映射
            x = j - new_center_x
            y = i - new_center_y

            # 逆旋转
            src_x = int(x * cos_a + y * sin_a + center_x)
            src_y = int(-x * sin_a + y * cos_a + center_y)

            if 0 <= src_x < w and 0 <= src_y < h:
                rotated[i, j] = img[src_y, src_x]

    return rotated
```

```python
def scale_image(img, scale_factor, method='bilinear'):
    """图像缩放"""
    h, w = img.shape[:2]
    new_h, new_w = int(h * scale_factor), int(w * scale_factor)

    if method == 'nearest':
        # 最近邻
        y = np.round(np.linspace(0, h-1, new_h)).astype(int)
        x = np.round(np.linspace(0, w-1, new_w)).astype(int)
        return img[np.ix_(y, x)]

    elif method == 'bilinear':
        # 双线性插值
        from scipy.ndimage import zoom
        if len(img.shape) == 3:
            return zoom(img, (scale_factor, scale_factor, 1), order=1).astype(np.uint8)
        else:
            return zoom(img, scale_factor, order=1).astype(np.uint8)
```

def apply_affine(img, matrix):
    """应用仿射变换矩阵"""
    h, w = img.shape[:2]

    # 计算变换后的边界
    corners = np.array([[0, 0], [w-1, 0], [0, h-1], [w-1, h-1]], dtype=np.float32)
    transformed_corners = np.dot(matrix, np.vstack([corners.T, np.ones(4)])).T

    min_x = int(max(0, transformed_corners[:, 0].min()))
    max_x = int(min(w, transformed_corners[:, 0].max()))
    min_y = int(max(0, transformed_corners[:, 1].min()))
    max_y = int(min(h, transformed_corners[:, 1].max()))

    new_w = max_x - min_x
    new_h = max_y - min_y

    # 创建输出
    output = np.zeros((new_h, new_w, 3), dtype=np.uint8)

    # 计算逆矩阵
    try:
        inv_matrix = np.linalg.inv(np.vstack([matrix, [0, 0, 1]]))
    except:
        return output

    # 应用变换
    for i in range(new_h):
        for j in range(new_w):
            # 目标坐标
            dst_x = j + min_x
            dst_y = i + min_y

            # 逆变换到源坐标
            src = inv_matrix @ np.array([dst_x, dst_y, 1])
            src_x, src_y = int(src[0]), int(src[1])

            if 0 <= src_x < w and 0 <= src_y < h:
                output[i, j] = img[src_y, src_x]

    return output

### 练习4：图像质量分析

```python
"""
练习4：图像质量分析

目标：理解图像质量的量化指标

任务：
1. 创建不同质量的图像（高斯模糊、噪声）
2. 计算图像的清晰度指标
3. 分析直方图特性
4. 比较不同处理的效果
"""

def practice4_image_quality():
    print("="*60)
    print("练习4：图像质量分析")
    print("="*60)

    # 创建基础图像
    base = np.zeros((100, 100, 3), dtype=np.uint8)
    base[20:80, 20:80] = [255, 255, 255]
    base[40:60, 40:60] = [0, 0, 0]

    # 1. 高斯模糊（手动实现简化版）
    def gaussian_blur(img, kernel_size=5):
        """简化高斯模糊"""
        from scipy.ndimage import gaussian_filter
        return gaussian_filter(img, sigma=kernel_size/6).astype(np.uint8)

    # 2. 添加高斯噪声
    def add_gaussian_noise(img, mean=0, sigma=20):
        noise = np.random.normal(mean, sigma, img.shape)
        noisy = img.astype(np.float32) + noise
        return np.clip(noisy, 0, 255).astype(np.uint8)

    # 3. 添加椒盐噪声
    def add_salt_pepper_noise(img, salt_prob=0.02, pepper_prob=0.02):
        noisy = img.copy()
        # 椒噪声（黑点）
        mask = np.random.random(img.shape[:2]) < pepper_prob
        noisy[mask] = [0, 0, 0]
        # 盐噪声（白点）
        mask = np.random.random(img.shape[:2]) < salt_prob
        noisy[mask] = [255, 255, 255]
        return noisy

    # 生成不同质量图像
    blurred = gaussian_blur(base, kernel_size=7)
    noisy_gaussian = add_gaussian_noise(base, sigma=30)
    noisy_sp = add_salt_pepper_noise(base, salt_prob=0.05, pepper_prob=0.05)

    # 计算清晰度指标（梯度能量）
    def sharpness_score(img):
        """计算图像清晰度（基于梯度）"""
        gray = img.mean(axis=2) if len(img.shape) == 3 else img
        dy, dx = np.gradient(gray)
        return np.sqrt(dx**2 + dy**2).sum()

    # 计算信噪比（简化）
    def psnr_like(img1, img2):
        """类似PSNR的指标"""
        mse = np.mean((img1.astype(float) - img2.astype(float))**2)
        if mse == 0:
            return float('inf')
        return 20 * np.log10(255.0 / np.sqrt(mse))

    print("\n图像质量指标：")
    images = {
        '原始': base,
        '高斯模糊': blurred,
        '高斯噪声': noisy_gaussian,
        '椒盐噪声': noisy_sp
    }

    for name, img in images.items():
        sharp = sharpness_score(img)
        quality = psnr_like(base, img)
        print(f"{name:10s}: 清晰度={sharp:8.0f}, 质量指标={quality:6.2f}")

    # 可视化
    fig, axes = plt.subplots(2, 4, figsize=(14, 7))

    # 显示图像
    for idx, (name, img) in enumerate(images.items()):
        axes[0, idx].imshow(img)
        axes[0, idx].set_title(f'{name}')
        axes[0, idx].axis('off')

        # 显示直方图
        gray = img.mean(axis=2).astype(np.uint8).ravel()
        axes[1, idx].hist(gray, bins=50, range=(0, 256), color='gray', alpha=0.7)
        axes[1, idx].set_title(f'直方图')
        axes[1, idx].set_xlabel('亮度')
        axes[1, idx].grid(True, alpha=0.3)

    plt.tight_layout()
    plt.show()

practice4_image_quality()
```
