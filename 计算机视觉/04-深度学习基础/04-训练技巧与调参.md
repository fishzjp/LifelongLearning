```python
def lr_scheduler_demo():
    """学习率调度策略对比"""

    print("学习率调度策略")
    print("="*60)

    # 模拟训练过程
    epochs = 100
    initial_lr = 0.1

    # 1. 固定学习率
    lr_fixed = np.full(epochs, initial_lr)

    # 2. 指数衰减
    lr_exp = initial_lr * np.exp(-0.03 * np.arange(epochs))

    # 3. 阶梯衰减（每20轮减半）
    lr_step = initial_lr * np.array([1, 1, 0.5, 0.5, 0.25, 0.25, 0.125, 0.125, 0.0625, 0.0625])
    lr_step = np.repeat(lr_step, 10)

    # 4. 余弦退火
    lr_cosine = initial_lr * 0.5 * (1 + np.cos(np.pi * np.arange(epochs) / epochs))

    # 5. Warmup + 余弦退火
    warmup_epochs = 10
    lr_warmup = np.linspace(0.01, initial_lr, warmup_epochs)
    lr_cosine_warmup = np.concatenate([
        lr_warmup,
        initial_lr * 0.5 * (1 + np.cos(np.pi * (np.arange(epochs - warmup_epochs)) / (epochs - warmup_epochs)))
    ])

    # 可视化
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))
    axes = axes.flatten()

    strategies = [
        (lr_fixed, '固定学习率', '固定'),
        (lr_exp, '指数衰减', 'exp'),
        (lr_step, '阶梯衰减', 'step'),
        (lr_cosine, '余弦退火', 'cosine'),
        (lr_warmup, 'Warmup阶段', 'warmup'),
        (lr_cosine_warmup, 'Warmup+余弦', 'warmup+cosine')
    ]

    for i, (lr, title, label) in enumerate(strategies):
        axes[i].plot(range(1, len(lr)+1), lr, linewidth=2)
        axes[i].set_xlabel('Epoch')
        axes[i].set_ylabel('学习率')
        axes[i].set_title(title, fontweight='bold')
        axes[i].grid(True, alpha=0.3)

        # 标注最终值
        final_lr = lr[-1]
        axes[i].text(0.7, 0.9, f'最终: {final_lr:.4f}',
                    transform=axes[i].transAxes, fontsize=9,
                    bbox=dict(boxstyle='round', facecolor='lightblue', alpha=0.5))

    # 移除多余的子图
    for i in range(len(strategies), len(axes)):
        fig.delaxes(axes[i])

    plt.tight_layout()
    plt.show()

    print("\n调度策略对比：")
    print("-" * 60)
    print(f"{'策略':<20} {'初始':<8} {'最终':<8} {'特点'}")
    print("-" * 60)
    print(f"{'固定':<20} {initial_lr:<8.4f} {lr_fixed[-1]:<8.4f} {'简单，但可能不是最优'}")
    print(f"{'指数衰减':<20} {initial_lr:<8.4f} {lr_exp[-1]:<8.4f} {'平滑衰减'}")
    print(f"{'阶梯衰减':<20} {initial_lr:<8.4f} {lr_step[-1]:<8.4f} {'简单有效'}")
    print(f"{'余弦退火':<20} {initial_lr:<8.4f} {lr_cosine[-1]:<8.4f} {'平滑，理论基础强'}")
```

#### 1.2.2 自适应学习率算法

```python
def adaptive_lr_demo():
    """自适应学习率对比"""

    print("自适应学习率算法")
    print("="*60)

    # 定义优化问题：f(x) = x²
    def f(x):
        return x**2

    def df(x):
        return 2*x

    # 不同优化器
    def sgd(x, lr=0.1):
        return x - lr * df(x)

    def momentum(x, v, lr=0.1, beta=0.9):
        v = beta * v + (1 - beta) * df(x)
        x = x - lr * v
        return x, v

    def rmsprop(x, r, lr=0.1, beta=0.9, epsilon=1e-8):
        r = beta * r + (1 - beta) * df(x)**2
        x = x - lr * df(x) / (np.sqrt(r) + epsilon)
        return x, r

    def adam(x, m, r, t, lr=0.1, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = beta1 * m + (1 - beta1) * df(x)
        r = beta2 * r + (1 - beta2) * df(x)**2
        m_hat = m / (1 - beta1**t)
        r_hat = r / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(r_hat) + epsilon)
        return x, m, r

    # 模拟优化过程
    x_start = 5.0
    n_steps = 20

    # SGD
    x_sgd = x_start
    path_sgd = [x_sgd]
    for _ in range(n_steps):
        x_sgd = sgd(x_sgd)
        path_sgd.append(x_sgd)

    # Momentum
    x_mom, v = x_start, 0.0
    path_mom = [x_mom]
    for _ in range(n_steps):
        x_mom, v = momentum(x_mom, v)
        path_mom.append(x_mom)

    # RMSprop
    x_rms, r = x_start, 0.0
    path_rms = [x_rms]
    for _ in range(n_steps):
        x_rms, r = rmsprop(x_rms, r)
        path_rms.append(x_rms)

    # Adam
    x_adam, m, r, t = x_start, 0.0, 0.0, 1
    path_adam = [x_adam]
    for _ in range(n_steps):
        x_adam, m, r = adam(x_adam, m, r, t)
        path_adam.append(x_adam)
        t += 1

    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 6))

    # 路径
    x_range = np.linspace(-6, 6, 100)
    y_range = f(x_range)

    ax1.plot(x_range, y_range, 'b-', linewidth=2, alpha=0.5)
    ax1.plot(path_sgd, [f(x) for x in path_sgd], 'o-', label='SGD', linewidth=2)
    ax1.plot(path_mom, [f(x) for x in path_mom], 's-', label='Momentum', linewidth=2)
    ax1.plot(path_rms, [f(x) for x in path_rms], '^-', label='RMSprop', linewidth=2)
    ax1.plot(path_adam, [f(x) for x in path_adam], 'd-', label='Adam', linewidth=2)

    ax1.scatter(0, 0, s=100, c='green', marker='*', zorder=5)
    ax1.set_xlabel('x')
    ax1.set_ylabel('f(x)')
    ax1.set_title('优化路径对比', fontweight='bold')
    ax1.legend()
    ax1.grid(True, alpha=0.3)

    # 损失下降
    ax2.plot([f(x) for x in path_sgd], 'o-', label='SGD', linewidth=2)
    ax2.plot([f(x) for x in path_mom], 's-', label='Momentum', linewidth=2)
    ax2.plot([f(x) for x in path_rms], '^-', label='RMSprop', linewidth=2)
    ax2.plot([f(x) for x in path_adam], 'd-', label='Adam', linewidth=2)

    ax2.set_xlabel('迭代次数')
    ax2.set_ylabel('损失')
    ax2.set_title('收敛速度对比', fontweight='bold')
    ax2.legend()
    ax2.grid(True, alpha=0.3)
    ax2.set_yscale('log')

    plt.tight_layout()
    plt.show()

    print("\n算法特点：")
    print("  SGD: 简单，但需要手动调学习率")
    print("  Momentum: 加速收敛，减少震荡")
    print("  RMSprop: 自适应学习率")
    print("  Adam: Momentum + RMSprop，最常用")
```

```python
def learning_rate_finder():
    """学习率搜索方法"""

    print("学习率搜索：LR Range Test")
    print("="*60)

    # 模拟模型训练
    def simulate_training(lr, n_batches=100):
        """模拟训练过程，返回最终损失"""
        loss = 10.0  # 初始损失
        for _ in range(n_batches):
            # 损失随时间下降，但学习率太大可能导致损失上升
            if lr > 0.5:
                loss += np.random.randn() * 0.5  # 不稳定
            else:
                loss *= (1 - lr * 0.1)  # 正常下降
            loss = max(loss, 0.1)
        return loss

    # 测试不同学习率
    lrs = np.logspace(-4, 0, 50)  # 从1e-4到1
    losses = [simulate_training(lr) for lr in lrs]

    # 找到最佳学习率
    min_idx = np.argmin(losses)
    best_lr = lrs[min_idx]
    best_loss = losses[min_idx]

    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

    # 对数坐标图
    ax1.plot(lrs, losses, 'b-', linewidth=2)
    ax1.scatter(best_lr, best_loss, s=100, c='red', marker='*', zorder=5)
    ax1.axvline(x=best_lr, color='red', linestyle='--', alpha=0.5)
    ax1.axvline(x=best_lr/10, color='orange', linestyle=':', alpha=0.5, label='10倍小')
    ax1.axvline(x=best_lr*10, color='green', linestyle=':', alpha=0.5, label='10倍大')

    ax1.set_xlabel('学习率 (log scale)')
    ax1.set_ylabel('损失')
    ax1.set_title('LR Range Test', fontweight='bold')
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    ax1.set_xscale('log')
    ax1.set_yscale('log')

    # 线性坐标图（放大最佳区域）
    mask = (lrs >= best_lr/20) & (lrs <= best_lr*20)
    ax2.plot(lrs[mask], losses[mask], 'b-', linewidth=2)
    ax2.scatter(best_lr, best_loss, s=100, c='red', marker='*', zorder=5)
    ax2.axvline(x=best_lr, color='red', linestyle='--', alpha=0.5)
    ax2.set_xlabel('学习率')
    ax2.set_ylabel('损失')
    ax2.set_title('最佳区域放大', fontweight='bold')
    ax2.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.show()

    print(f"\n搜索结果：")
    print(f"  最佳学习率: {best_lr:.6f}")
    print(f"  对应损失: {best_loss:.4f}")
    print(f"\n推荐范围:")
    print(f"  保守: {best_lr/10:.6f} - {best_lr:.6f}")
    print(f"  标准: {best_lr:.6f}")
    print(f"  激进: {best_lr:.6f} - {best_lr*10:.6f}")
```

```python
def gradient_accumulation_demo():
    """梯度累积模拟"""

    print("梯度累积：模拟大批次训练")
    print("="*60)

    # 模拟场景：内存只能支持batch_size=4，但想用batch_size=16
    real_bs = 4
    target_bs = 16
    accumulation_steps = target_bs // real_bs

    print(f"目标批次: {target_bs}")
    print(f"实际批次: {real_bs}")
    print(f"累积步数: {accumulation_steps}")
    print()

    # 模拟训练数据
    np.random.seed(42)
    n_samples = 100
    X = np.random.randn(n_samples, 2)
    y = 2 * X[:, 0] + 3 * X[:, 1] + np.random.randn(n_samples) * 0.1

    # 简单线性模型
    w = np.array([0.0, 0.0])
    b = 0.0
    lr = 0.01

    print("梯度累积过程：")
    print("-" * 60)

    # 模拟一个epoch
    epoch_loss = 0
    grad_accum_w = np.zeros(2)
    grad_accum_b = 0.0
    step_count = 0

    for i in range(0, n_samples, real_bs):
        X_batch = X[i:i+real_bs]
        y_batch = y[i:i+real_bs]

        # 前向传播
        y_pred = X_batch @ w + b
        loss = np.mean((y_pred - y_batch)**2)
        epoch_loss += loss

        # 计算梯度
        error = y_pred - y_batch
        dw = X_batch.T @ error / real_bs
        db = np.sum(error) / real_bs

        # 累积梯度
        grad_accum_w += dw
        grad_accum_b += db
        step_count += 1

        print(f"步骤 {step_count}: 损失={loss:.4f}, 累积梯度={grad_accum_w}")

        # 达到累积步数，更新参数
        if step_count == accumulation_steps:
            w -= lr * grad_accum_w / accumulation_steps
            b -= lr * grad_accum_b / accumulation_steps

            print(f"  → 更新参数: w={w}, b={b:.4f}")
            print()

            # 重置累积
            grad_accum_w = np.zeros(2)
            grad_accum_b = 0.0
            step_count = 0

    print(f"\n一个epoch平均损失: {epoch_loss / (n_samples // real_bs):.4f}")
    print(f"最终参数: w={w}, b={b:.4f}")

    print("\n梯度累积的优势：")
    print("✓ 模拟大批次训练")
    print("✓ 节省内存")
    print("✓ 保持小批次的泛化优势")
    print("✓ 适合：显存不足但需要大批次效果")
```

```python
def normalization_demo():
    """数据标准化演示"""

    print("数据标准化：让训练更稳定")
    print("="*60)

    # 生成不同尺度的特征
    np.random.seed(42)
    n_samples = 1000

    # 特征1：小范围
    f1 = np.random.randn(n_samples) * 0.1 + 5

    # 特征2：大范围
    f2 = np.random.randn(n_samples) * 100 + 500

    # 特征3：偏态分布
    f3 = np.random.exponential(scale=2, size=n_samples)

    X = np.stack([f1, f2, f3], axis=1)

    print("原始数据统计：")
    print(f"  特征1: 均值={np.mean(f1):.2f}, 标准差={np.std(f1):.2f}")
    print(f"  特征2: 均值={np.mean(f2):.2f}, 标准差={np.std(f2):.2f}")
    print(f"  特征3: 均值={np.mean(f3):.2f}, 标准差={np.std(f3):.2f}")

    # 标准化方法
    def standardize(X):
        """Z-score标准化：(X - mean) / std"""
        return (X - np.mean(X, axis=0)) / np.std(X, axis=0)

    def normalize(X):
        """Min-Max归一化：(X - min) / (max - min)"""
        return (X - np.min(X, axis=0)) / (np.max(X, axis=0) - np.min(X, axis=0))

    X_std = standardize(X)
    X_norm = normalize(X)

    # 可视化
    fig, axes = plt.subplots(3, 3, figsize=(15, 12))

    methods = ['原始', '标准化', '归一化']
    data_list = [X, X_std, X_norm]

    for col, (name, data) in enumerate(zip(methods, data_list)):
        for row in range(3):
            axes[row, col].hist(data[:, row], bins=30, alpha=0.7, color='skyblue')
            axes[row, col].set_title(f'{name} - 特征{row+1}', fontweight='bold')
            axes[row, col].grid(True, alpha=0.3)
```

```python
def initialization_demo():
    """初始化对比演示"""

    print("权重初始化的重要性")
    print("="*60)

    # 模拟不同初始化的网络
    def simulate_network_forward(n_layers=10, n_neurons=100, init_type='random'):
        """模拟前向传播，观察激活值分布"""
        np.random.seed(42)

        x = np.random.randn(n_neurons)  # 输入

        activations = [x]
        for i in range(n_layers):
            if init_type == 'random':
                # 普通随机
                w = np.random.randn(n_neurons, n_neurons)
            elif init_type == 'large':
                # 大初始化
                w = np.random.randn(n_neurons, n_neurons) * 2
            elif init_type == 'xavier':
                # Xavier初始化
                w = np.random.randn(n_neurons, n_neurons) / np.sqrt(n_neurons)
            elif init_type == 'he':
                # He初始化（ReLU）
                w = np.random.randn(n_neurons, n_neurons) * np.sqrt(2 / n_neurons)

            x = np.maximum(0, np.dot(w, x))  # ReLU
            activations.append(x)

        return activations

    # 测试不同初始化
    inits = ['random', 'large', 'xavier', 'he']
    titles = ['普通随机', '大初始化', 'Xavier', 'He']

    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()

    print("不同初始化下，各层激活值的分布：")
    print("-" * 60)

    for idx, (init, title) in enumerate(zip(inits, titles)):
        acts = simulate_network_forward(init_type=init)

        # 计算每层激活值的均值和标准差
        means = [np.mean(a) for a in acts]
        stds = [np.std(a) for a in acts]
        layers = range(len(acts))

        # 绘制
        axes[idx].plot(layers, means, 'o-', label='均值', linewidth=2, markersize=6)
        axes[idx].plot(layers, stds, 's-', label='标准差', linewidth=2, markersize=6)

        axes[idx].set_xlabel('层数')
        axes[idx].set_ylabel('值')
        axes[idx].set_title(f'{title}初始化', fontweight='bold')
        axes[idx].legend()
        axes[idx].grid(True, alpha=0.3)
        axes[idx].axhline(y=0, color='black', linestyle='--', alpha=0.3)

        # 检查问题
        final_mean = means[-1]
        final_std = stds[-1]

        if final_mean < 0.1 or final_mean > 10:
            status = "⚠ 消失/爆炸"
        elif final_std < 0.1 or final_std > 10:
            status = "⚠ 分布异常"
        else:
            status = "✓ 正常"

        print(f"{title:<10}: 均值={final_mean:.4f}, 标准差={final_std:.4f} {status}")

    plt.tight_layout()
    plt.show()

    print("\n初始化推荐：")
    print("  Sigmoid/Tanh: Xavier/Glorot初始化")
```

```python
def batch_normalization_demo():
    """批归一化演示"""

    print("批归一化 (Batch Normalization)")
    print("="*60)

    # 模拟层输出
    np.random.seed(42)
    batch_size = 32
    n_features = 10

    # 创建不同分布的激活值（模拟深层网络）
    # 前几层正常，后几层逐渐偏移
    activations = []
    for i in range(5):
        mean = i * 0.5
        std = 1.0 + i * 0.2
        act = np.random.randn(batch_size, n_features) * std + mean
        activations.append(act)

    # 批归一化函数
    def batch_norm(x, epsilon=1e-5):
        mean = np.mean(x, axis=0)
        var = np.var(x, axis=0)
        x_norm = (x - mean) / np.sqrt(var + epsilon)
        return x_norm

    # 可视化
    fig, axes = plt.subplots(2, 5, figsize=(16, 8))

    print("各层激活值分布（BN前 vs BN后）：")
    print("-" * 60)

    for i, act in enumerate(activations):
        # BN前
        axes[0, i].hist(act.flatten(), bins=20, alpha=0.7, color='red')
        axes[0, i].set_title(f'层{i+1} (BN前)', fontweight='bold')
        axes[0, i].set_xlabel('激活值')
        axes[0, i].set_ylabel('频数')
        axes[0, i].grid(True, alpha=0.3)

        # BN后
        act_norm = batch_norm(act)
        axes[1, i].hist(act_norm.flatten(), bins=20, alpha=0.7, color='blue')
        axes[1, i].set_title(f'层{i+1} (BN后)', fontweight='bold')

              f"BN后均值={np.mean(act_norm):.4f}, 标准差={np.std(act_norm):.4f}")

    plt.tight_layout()
    plt.show()

    print("\n批归一化的作用：")
```

```python
class ProductionTrainer:
    """生产级训练器"""

    def __init__(self, model, optimizer, criterion, device='cpu'):
        self.model = model
        self.optimizer = optimizer
        self.criterion = criterion
        self.device = device
        self.history = {
            'train_loss': [],
            'val_loss': [],
            'train_acc': [],
            'val_acc': []
        }

    def train_epoch(self, train_loader):
        """一个训练epoch"""
        self.model.train()
        total_loss = 0
        total_correct = 0
        total_samples = 0

        for batch_idx, (data, target) in enumerate(train_loader):
            # 前向传播
            output = self.model(data)
            loss = self.criterion(output, target)

            # 反向传播
            self.optimizer.zero_grad()
            loss.backward()

            # 梯度裁剪
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), max_norm=1.0)

            # 更新参数
            self.optimizer.step()

            # 统计
            total_loss += loss.item()
            pred = output.argmax(dim=1)
            total_correct += pred.eq(target).sum().item()
            total_samples += target.size(0)

        return total_loss / len(train_loader), total_correct / total_samples

    def validate(self, val_loader):
        """验证"""
        self.model.eval()
        total_loss = 0
        total_correct = 0
        total_samples = 0

        with torch.no_grad():
            for data, target in val_loader:
                output = self.model(data)
                loss = self.criterion(output, target)

                total_loss += loss.item()
                pred = output.argmax(dim=1)
                total_correct += pred.eq(target).sum().item()
                total_samples += target.size(0)

        return total_loss / len(val_loader), total_correct / total_samples

    def fit(self, train_loader, val_loader, epochs, patience=10):
        """完整训练流程"""
        best_val_loss = float('inf')
        patience_counter = 0
        best_state = None

        print("="*70)
        print("开始训练")
        print("="*70)
        print(f"{'Epoch':<6} {'Train Loss':<12} {'Val Loss':<12} {'Train Acc':<12} {'Val Acc':<12} {'Patience'}")
        print("-"*70)

        for epoch in range(1, epochs + 1):
            train_loss, train_acc = self.train_epoch(train_loader)
            val_loss, val_acc = self.validate(val_loader)

            self.history['train_loss'].append(train_loss)
            self.history['val_loss'].append(val_loss)
            self.history['train_acc'].append(train_acc)
            self.history['val_acc'].append(val_acc)

            print(f"{epoch:<6} {train_loss:<12.4f} {val_loss:<12.4f} {train_acc:<12.2%} {val_acc:<12.2%} {patience_counter}")

            # 早停逻辑
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                patience_counter = 0
                # 保存最佳模型
                best_state = {k: v.cpu().clone() for k, v in self.model.state_dict().items()}
            else:
                patience_counter += 1

            if patience_counter >= patience:
                print(f"\n早停触发！最佳验证损失: {best_val_loss:.4f}")
                break

        # 恢复最佳模型
        if best_state:
            self.model.load_state_dict(best_state)

        return self.history
```

## 🎯 本章要点总结

### 核心技巧

1. **学习率**：
   - 最重要的超参数
   - 使用调度策略（余弦、阶梯）
   - Adam是好的默认选择

2. **批次大小**：
   - 32-256是常用范围
   - 梯度累积模拟大批次

3. **数据预处理**：
   - 标准化必不可少
   - 数据增强提高泛化

4. **初始化**：
   - He（ReLU）或Xavier（Sigmoid）
   - 避免极端值

5. **监控**：
   - 分析损失曲线
   - 早停防止过拟合
   - 梯度裁剪防爆炸

### 调参哲学

- **从小开始**：简单模型 + 小数据集
- **系统实验**：一次只改一个变量
- **记录一切**：实验配置 + 结果
- **理解问题**：诊断 > 盲目调参
