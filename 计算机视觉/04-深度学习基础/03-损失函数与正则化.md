## 🎯 1. 损失函数详解

### 1.1 回归问题的损失函数

#### 1.1.1 均方误差（MSE）- 数学推导

**定义：**

$$
\text{MSE} = \frac{1}{n} \sum_{i=1}^{n} (y_i - \hat{y}_i)^2
$$

**变量说明：**
- $y_i$：真实值
- $\hat{y}_i$：预测值
- $n$：样本数量

**为什么使用平方？**

1. **惩罚大误差**：误差越大，惩罚越重（二次增长）
2. **可导性**：处处可导，便于优化
3. **最大似然**：假设噪声服从高斯分布时，MSE等价于最大似然估计

**最大似然推导：**

假设 $y = f(x) + \epsilon$，其中 $\epsilon \sim \mathcal{N}(0, \sigma^2)$

$$
P(y|x) = \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left(-\frac{(y - \hat{y})^2}{2\sigma^2}\right)
$$

似然函数：

$$
L = \prod_{i=1}^{n} P(y_i|x_i) = \prod_{i=1}^{n} \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left(-\frac{(y_i - \hat{y}_i)^2}{2\sigma^2}\right)
$$

对数似然：

$$
\log L = \sum_{i=1}^{n} \left[\log\left(\frac{1}{\sqrt{2\pi\sigma^2}}\right) - \frac{(y_i - \hat{y}_i)^2}{2\sigma^2}\right]
$$

最大化似然等价于最小化负对数似然：

$$
-\log L = \sum_{i=1}^{n} \frac{(y_i - \hat{y}_i)^2}{2\sigma^2} + \text{constant}
$$

忽略常数项和系数，得到MSE：

$$
\text{MSE} = \sum_{i=1}^{n} (y_i - \hat{y}_i)^2
$$

**梯度推导：**

$$
\frac{\partial \text{MSE}}{\partial \hat{y}_i} = \frac{\partial}{\partial \hat{y}_i} (y_i - \hat{y}_i)^2 = -2(y_i - \hat{y}_i)
$$

```python
def mse_mathematical_derivation():
    """MSE数学推导详解"""

    import numpy as np
    import matplotlib.pyplot as plt

    print("均方误差（MSE）数学推导")
    print("="*70)

    # 生成数据
    np.random.seed(42)
    n_samples = 100
    X = np.linspace(0, 10, n_samples)
    true_slope = 2.5
    true_intercept = 1.0

    # 添加高斯噪声
    noise = np.random.randn(n_samples) * 2
    y = true_slope * X + true_intercept + noise

    print("\n【数据生成】")
    print(f"真实模型: y = {true_slope}x + {true_intercept}")
    print(f"噪声分布: N(0, 4)")
    print(f"样本数: {n_samples}")

    # 初始化参数
    slope = np.random.randn()
    intercept = np.random.randn()

    print(f"\n初始参数: slope={slope:.4f}, intercept={intercept:.4f}")

    # 梯度下降
    learning_rate = 0.001
    n_epochs = 100

    trajectory = []

    for epoch in range(n_epochs):
        # 前向传播
        y_pred = slope * X + intercept

        # 计算MSE
        mse = np.mean((y - y_pred)**2)
        trajectory.append((slope, intercept, mse))

        # 计算梯度
        grad_slope = -2 * np.mean((y - y_pred) * X)
        grad_intercept = -2 * np.mean(y - y_pred)

        # 更新参数
        slope -= learning_rate * grad_slope
        intercept -= learning_rate * grad_intercept

        if (epoch + 1) % 20 == 0:
            print(f"Epoch {epoch+1:3d}: MSE={mse:.4f}, "
                 f"slope={slope:.4f}, intercept={intercept:.4f}")

    print(f"\n最终参数: slope={slope:.4f}, intercept={intercept:.4f}")
    print(f"真实参数: slope={true_slope:.4f}, intercept={true_intercept:.4f}")

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))

    # 数据拟合
    ax = axes[0, 0]
    ax.scatter(X, y, alpha=0.5, label='数据')
    ax.plot(X, true_slope * X + true_intercept, 'g--', linewidth=2,
           label='真实模型')
    ax.plot(X, slope * X + intercept, 'r-', linewidth=2,
           label='拟合模型')
    ax.set_xlabel('x')
    ax.set_ylabel('y')
    ax.set_title('线性回归拟合', fontweight='bold')
    ax.legend()
    ax.grid(True, alpha=0.3)

    # 损失曲线
    ax = axes[0, 1]
    mses = [t[2] for t in trajectory]
    ax.plot(mses, linewidth=2)
    ax.set_xlabel('Epoch')
    ax.set_ylabel('MSE')
    ax.set_title('损失下降曲线', fontweight='bold')
    ax.grid(True, alpha=0.3)
    ax.set_yscale('log')

    # 参数轨迹
    ax = axes[1, 0]
    slopes = [t[0] for t in trajectory]
    intercepts = [t[1] for t in trajectory]
    ax.plot(slopes, intercepts, 'o-', linewidth=2, markersize=3,
           label='优化路径')
    ax.scatter(slopes[0], intercepts[0], s=100, c='red', marker='o',
              label='起点', zorder=5)
    ax.scatter(slopes[-1], intercepts[-1], s=100, c='green', marker='*',
              label='终点', zorder=5)
    ax.scatter(true_slope, true_intercept, s=100, c='blue', marker='x',
              label='最优点', zorder=5)
    ax.set_xlabel('Slope')
    ax.set_ylabel('Intercept')
    ax.set_title('参数优化轨迹', fontweight='bold')
    ax.legend()
    ax.grid(True, alpha=0.3)

    # 误差分布
    ax = axes[1, 1]
    final_pred = slope * X + intercept
    errors = y - final_pred
    ax.hist(errors, bins=20, alpha=0.7, color='blue', edgecolor='black')
    ax.axvline(x=0, color='red', linestyle='--', linewidth=2, label='零误差')
    ax.set_xlabel('误差')
    ax.set_ylabel('频数')
    ax.set_title('误差分布（应接近高斯分布）', fontweight='bold')
    ax.legend()
    ax.grid(True, alpha=0.3, axis='y')

    plt.tight_layout()
    plt.show()

    print("\n【数学意义】")
    print("1. MSE对应高斯噪声假设下的最大似然估计")
    print("2. 对大误差惩罚更重（平方项）")
    print("3. 梯度简单：∂MSE/∂ŷ = -2(y - ŷ)")
    print("4. 全局凸优化，有唯一最优解")

mse_mathematical_derivation()
```

```python
import numpy as np
import matplotlib.pyplot as plt

def mse_demo():
    """MSE详解"""

    print("均方误差 (Mean Squared Error)")
    print("="*60)

    # 真实值和预测值
    y_true = np.array([3.0, 5.0, 7.0, 9.0])
    y_pred = np.array([2.5, 5.5, 6.5, 9.5])

    errors = y_pred - y_true

    print(f"真实值: {y_true}")
    print(f"预测值: {y_pred}")
    print(f"误差: {errors}")

    # 计算MSE
    mse = np.mean(errors ** 2)
    mae = np.mean(np.abs(errors))

    print(f"\nMSE: {mse:.4f}")
    print(f"MAE: {mae:.4f}")

    # 为什么MSE对大误差更敏感？
    print("\n【MSE vs MAE】")
    print("假设有一个异常值：误差 = 10")
    print(f"  MSE惩罚: {10**2} = 100")
    print(f"  MAE惩罚: {abs(10)} = 10")
    print("  MSE是MAE的10倍！")

    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

    # 误差对比
    x = np.linspace(-5, 5, 100)
    mse_curve = x**2
    mae_curve = np.abs(x)
```

```python
def mae_demo():
    """MAE详解"""

    print("平均绝对误差 (Mean Absolute Error)")
    print("="*60)

    # 对比MSE和MAE对异常值的敏感度
    np.random.seed(42)
    y_true = np.random.randn(100) * 2 + 10  # 正常数据

    # 情况1：正常预测
    y_pred_normal = y_true + np.random.randn(100) * 0.5

    # 情况2：包含异常值（10%的数据误差很大）
    y_pred_outlier = y_pred_normal.copy()
    outlier_indices = np.random.choice(100, 10, replace=False)
    y_pred_outlier[outlier_indices] += 15  # 大误差

    # 计算损失
    mse_normal = np.mean((y_pred_normal - y_true)**2)
    mae_normal = np.mean(np.abs(y_pred_normal - y_true))

    mse_outlier = np.mean((y_pred_outlier - y_true)**2)
    mae_outlier = np.mean(np.abs(y_pred_outlier - y_true))

    print("正常情况：")
    print(f"  MSE: {mse_normal:.4f}")
    print(f"  MAE: {mae_normal:.4f}")

    print("\n包含异常值：")
    print(f"  MSE: {mse_outlier:.4f} (+{((mse_outlier/mse_normal)-1)*100:.1f}%)")
    print(f"  MAE: {mae_outlier:.4f} (+{((mae_outlier/mae_normal)-1)*100:.1f}%)")

    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))

    # 散点图
    axes[0, 0].scatter(y_true, y_pred_normal, alpha=0.6)
```

#### 1.1.3 Huber损失（平滑版MAE）

```python
def huber_demo():
    """Huber损失详解"""

    print("Huber损失：MSE和MAE的结合")
    print("="*60)

    def huber_loss(y_true, y_pred, delta=1.0):
        """Huber损失"""
        error = y_pred - y_true
        abs_error = np.abs(error)

        # 小误差用MSE，大误差用MAE
        is_small_error = abs_error <= delta
        squared_loss = 0.5 * error**2
        linear_loss = delta * (abs_error - 0.5 * delta)

        return np.where(is_small_error, squared_loss, linear_loss)

    # 可视化不同delta的影响
    x = np.linspace(-5, 5, 100)

    fig, axes = plt.subplots(1, 3, figsize=(15, 5))

    deltas = [0.5, 1.0, 2.0]

    for i, delta in enumerate(deltas):
        loss = huber_loss(0, x, delta)

        axes[i].plot(x, loss, linewidth=2, color='blue')
        axes[i].plot(x, x**2, '--', linewidth=1, color='red', alpha=0.5, label='MSE')
        axes[i].plot(x, np.abs(x), '--', linewidth=1, color='green', alpha=0.5, label='MAE')
        axes[i].axvline(x=delta, color='gray', linestyle=':', alpha=0.5)
        axes[i].axvline(x=-delta, color='gray', linestyle=':', alpha=0.5)
        axes[i].set_xlabel('误差')
        axes[i].set_ylabel('损失')
        axes[i].set_title(f'Huber (δ={delta})', fontweight='bold')
        axes[i].grid(True, alpha=0.3)
        axes[i].legend()

    plt.tight_layout()
    plt.show()

    print("\nHuber损失的特点：")
    print("✓ 小误差：MSE（可导，收敛快）")
    print("✓ 大误差：MAE（鲁棒）")
    print("✓ 需要调参：delta阈值")
    print("✓ 适合：回归任务 + 数据噪声大")
```

### 1.2 分类问题的损失函数

#### 1.2.1 交叉熵损失 - 数学推导

**信息论基础：**

交叉熵源于信息论，用于衡量两个概率分布的差异。

**熵（Entropy）：**

$$
H(P) = -\sum_{i} P(i) \log P(i)
$$

**交叉熵（Cross-Entropy）：**

$$
H(P, Q) = -\sum_{i} P(i) \log Q(i)
$$

其中 $P$ 是真实分布，$Q$ 是预测分布。

**KL散度（Kullback-Leibler Divergence）：**

$$
D_{KL}(P \| Q) = \sum_{i} P(i) \log \frac{P(i)}{Q(i)}
$$

关系：

$$
H(P, Q) = H(P) + D_{KL}(P \| Q)
$$

由于 $H(P)$ 是常数，最小化交叉熵等价于最小化KL散度。

**二元交叉熵（Binary Cross-Entropy）：**

$$
\text{BCE} = -\frac{1}{n} \sum_{i=1}^{n} [y_i \log(\hat{y}_i) + (1-y_i) \log(1-\hat{y}_i)]
$$

**变量说明：**
- $y_i \in \{0, 1\}$：真实标签
- $\hat{y}_i \in (0, 1)$：预测概率

**推导（从最大似然）：**

假设 $y \sim \text{Bernoulli}(p)$，则：

$$
P(y|\hat{y}) = \hat{y}^y (1-\hat{y})^{1-y}
$$

似然函数：

$$
L = \prod_{i=1}^{n} \hat{y}_i^{y_i} (1-\hat{y}_i)^{1-y_i}
$$

对数似然：

$$
\log L = \sum_{i=1}^{n} [y_i \log \hat{y}_i + (1-y_i) \log (1-\hat{y}_i)]
$$

最大化似然等价于最小化负对数似然：

$$
\text{BCE} = -\log L = -\sum_{i=1}^{n} [y_i \log \hat{y}_i + (1-y_i) \log (1-\hat{y}_i)]
$$

**梯度推导：**

$$
\frac{\partial \text{BCE}}{\partial \hat{y}_i} = -\frac{y_i}{\hat{y}_i} + \frac{1-y_i}{1-\hat{y}_i} = \frac{\hat{y}_i - y_i}{\hat{y}_i(1-\hat{y}_i)}
$$

**多分类交叉熵（Categorical Cross-Entropy）：**

$$
\text{CCE} = -\sum_{i=1}^{n} \sum_{c=1}^{C} y_{i,c} \log(\hat{y}_{i,c})
$$

**变量说明：**
- $C$：类别数
- $y_{i,c}$：第 $i$ 个样本属于类别 $c$ 的真实标签（one-hot）
- $\hat{y}_{i,c}$：第 $i$ 个样本属于类别 $c$ 的预测概率

```python
def cross_entropy_mathematical_derivation():
    """交叉熵损失数学推导详解"""

    import numpy as np
    import matplotlib.pyplot as plt

    print("交叉熵损失数学推导")
    print("="*70)

    # 示例：3分类问题
    print("\n【多分类示例】")
    classes = ['猫', '狗', '鸟']

    # 真实标签（one-hot）
    y_true = np.array([1, 0, 0])  # 猫

    # 不同预测概率
    predictions = [
        np.array([0.7, 0.2, 0.1]),  # 正确且自信
        np.array([0.4, 0.3, 0.3]),  # 正确但不自信
        np.array([0.3, 0.6, 0.1]),  # 错误（预测为狗）
        np.array([0.1, 0.1, 0.8])   # 错误（预测为鸟）
    ]

    print(f"真实类别: {classes[np.argmax(y_true)]}（猫）")
    print("\n不同预测的损失:")
    print("-" * 70)

    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    axes = axes.flatten()

    for i, y_pred in enumerate(predictions):
        # 计算交叉熵
        cce = -np.sum(y_true * np.log(y_pred + 1e-10))

        print(f"\n情况 {i+1}:")
        print(f"  预测概率: {y_pred}")
        print(f"  预测类别: {classes[np.argmax(y_pred)]}")
        print(f"  交叉熵损失: {cce:.4f}")
        print(f"  计算过程: -({y_true[0]}×log({y_pred[0]:.2f}) + "
              f"{y_true[1]}×log({y_pred[1]:.2f}) + {y_true[2]}×log({y_pred[2]:.2f}))")
        print(f"          = -{y_true[0]}×{np.log(y_pred[0]+1e-10):.4f} = {cce:.4f}")

        # 可视化
        colors = ['green' if j == np.argmax(y_true) else 'gray'
                 for j in range(len(classes))]
        axes[i].bar(classes, y_pred, color=colors, alpha=0.7, edgecolor='black')
        axes[i].set_ylim(0, 1)
        axes[i].set_ylabel('概率')
        axes[i].set_title(f'预测{i+1}: 损失={cce:.4f}', fontweight='bold')
        axes[i].grid(True, alpha=0.3, axis='y')

        # 添加数值标签
        for j, (cls, prob) in enumerate(zip(classes, y_pred)):
            axes[i].text(j, prob + 0.02, f'{prob:.2f}',
                       ha='center', fontweight='bold')

    plt.tight_layout()
    plt.show()

    print("\n" + "="*70)
    print("【关键观察】")
    print("1. 预测正确且自信 → 损失小（0.357）")
    print("2. 预测正确但不确定 → 损失中等（0.916）")
    print("3. 预测错误 → 损失大（1.204, 2.303）")
    print("\n交叉熵不仅关注是否预测正确，还关注置信度！")

    # Softmax + 交叉熵的联合梯度
    print("\n" + "="*70)
    print("【Softmax + 交叉熵的简化梯度】")
    print("-" * 70)

    def softmax(z):
        exp_z = np.exp(z - np.max(z))
        return exp_z / np.sum(exp_z)

    # 假设logits
    logits = np.array([2.0, 1.0, 0.1])
    y_true = np.array([1, 0, 0])

    # Softmax
    y_pred = softmax(logits)

    print(f"\nLogits: {logits}")
    print(f"Softmax后: {y_pred}")
    print(f"真实标签: {y_true}")

    # 交叉熵梯度
    grad = y_pred - y_true
```

#### 1.2.3 焦点损失（Focal Loss）- 数学推导

**问题背景：**

在目标检测等任务中，存在严重的类别不平衡：
- 正样本（前景）：很少
- 负样本（背景）：很多

使用标准交叉熵时，大量易分类的负样本主导了梯度计算。

**Focal Loss定义：**

$$
\text{FL}(p_t) = -\alpha_t (1 - p_t)^\gamma \log(p_t)
$$

**变量说明：**
- $p_t$：模型对正确类别的预测概率
- $\gamma \geq 0$：聚焦参数（通常 $\gamma=2$）
- $\alpha_t$：类别权重

对于二分类：

$$
p_t = \begin{cases}
p, & \text{if } y = 1 \\
1 - p, & \text{if } y = 0
\end{cases}
$$

**设计思想：**

1. **降低易分类样本的权重**：当 $p_t \to 1$ 时，$(1-p_t)^\gamma \to 0$
2. **保持难分类样本的权重**：当 $p_t \ll 1$ 时，$(1-p_t)^\gamma \approx 1$

**推导过程：**

标准二元交叉熵：

$$
\text{BCE}(p_t) = -\log(p_t)
$$

添加调制因子：

$$
\text{FL}(p_t) = -(1 - p_t)^\gamma \log(p_t)
$$

当 $\gamma = 0$ 时，退化为标准交叉熵。

当 $\gamma > 0$ 时：
- 易分类样本（$p_t$ 大）：$(1-p_t)^\gamma$ 小，损失被降低
- 难分类样本（$p_t$ 小）：$(1-p_t)^\gamma \approx 1$，损失基本不变

**梯度分析：**

$$
\frac{\partial \text{FL}}{\partial p} = -\frac{\partial}{\partial p} [(1-p)^\gamma \log p]
$$

$$
= -[-\gamma(1-p)^{\gamma-1} \log p + \frac{(1-p)^\gamma}{p}]
$$

$$
= (1-p)^{\gamma-1} [\gamma \log p - \frac{1-p}{p}]
$$

**数值示例：**

设 $\gamma = 2$：

| $p_t$ | BCE | Focal | 比例 |
|-------|-----|-------|------|
| 0.9 | 0.105 | 0.001 | 1:105 |
| 0.7 | 0.357 | 0.032 | 1:11 |
| 0.5 | 0.693 | 0.173 | 1:4 |
| 0.3 | 1.204 | 0.590 | 1:2 |
| 0.1 | 2.303 | 1.864 | 1:1.2 |

**结论：**
- 易分类样本（$p_t=0.9$）的损失降低了105倍
- 难分类样本（$p_t=0.1$）的损失基本不变

```python
def l1_regularization_demo():
    """L1正则化演示"""

    print("L1正则化 (Lasso)")
    print("="*60)

    # 生成数据：只有部分特征有用
    np.random.seed(42)
    n_samples, n_features = 100, 5
    X = np.random.randn(n_samples, n_features)

    # 真实权重：只有前2个非零
    true_w = np.array([3.0, 2.0, 0.0, 0.0, 0.0])
    y = X @ true_w + np.random.randn(n_samples) * 0.1

    # 比较L1和L2
    def lasso_regression(X, y, lam, n_iter=1000, lr=0.01):
        """坐标下降实现Lasso"""
        n_samples, n_features = X.shape
        w = np.zeros(n_features)

        for _ in range(n_iter):
            for j in range(n_features):
                # 计算残差
                y_pred = X @ w
                r = y - y_pred + X[:, j] * w[j]

                # 坐标下降更新
                rho = X[:, j] @ r
                if rho < -lam / 2:
                    w[j] = (rho + lam / 2) / (X[:, j] @ X[:, j])
                elif rho > lam / 2:
                    w[j] = (rho - lam / 2) / (X[:, j] @ X[:, j])
                else:
                    w[j] = 0

        return w

    def ridge_regression(X, y, lam):
        """岭回归"""
        I = np.eye(X.shape[1])
        return np.linalg.inv(X.T @ X + lam * I) @ X.T @ y

    # 测试不同λ
    lambdas = [0.01, 0.1, 1.0, 10.0]

    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    axes = axes.flatten()

    print("L1 vs L2 正则化对比：")
    print("-" * 70)
    print(f"{'λ':<6} | {'L1 w':<35} | {'L2 w':<35} | {'L1稀疏性'}")
    print("-" * 70)

    for i, lam in enumerate(lambdas):
        w_l1 = lasso_regression(X, y, lam)
        w_l2 = ridge_regression(X, y, lam)

        # 稀疏性（零权重的数量）
        sparsity = np.sum(np.abs(w_l1) < 1e-6)
```

```python
def dropout_demo():

    print("Dropout - 随机失活")
    print("="*60)

    # 模拟神经网络层
    np.random.seed(42)
    n_neurons = 10
    activations = np.random.randn(n_neurons)

    print(f"原始激活值: {activations}")

    # 不同Dropout率
    dropout_rates = [0, 0.2, 0.5, 0.8]

    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()

    for i, p in enumerate(dropout_rates):
        # 训练时的Dropout
        keep_prob = 1 - p
        mask = np.random.binomial(1, keep_prob, size=n_neurons)

        # 应用Dropout
        dropped = activations * mask

        # 测试时的缩放（Inverted Dropout）
        scaled = dropped / keep_prob if keep_prob > 0 else dropped

        print(f"\nDropout率={p}:")
        print(f"  Mask: {mask}")
        print(f"  Dropped: {dropped}")
        print(f"  Scaled: {scaled}")

        axes[i].bar(x, activations, alpha=0.3, color='gray', label='原始')
        axes[i].bar(x, dropped, alpha=0.8, color='blue', label='Dropout后')
        axes[i].scatter(x, scaled, color='red', s=50, marker='x', label='缩放后')
```

```python
class RegularizedNetwork:
    """带正则化的神经网络"""

    def __init__(self, layer_sizes, l2_lambda=0.01, dropout_rate=0.2):
        self.layer_sizes = layer_sizes
        self.l2_lambda = l2_lambda
        self.dropout_rate = dropout_rate

        # 初始化参数
        self.parameters = {}
        for i in range(len(layer_sizes) - 1):
            self.parameters[f'W{i+1}'] = np.random.randn(layer_sizes[i], layer_sizes[i+1]) / np.sqrt(layer_sizes[i])
            self.parameters[f'b{i+1}'] = np.zeros((1, layer_sizes[i+1]))

        print(f"创建网络: {' → '.join(map(str, layer_sizes))}")
        print(f"L2 λ: {l2_lambda}, Dropout: {dropout_rate}")

    def forward(self, X, training=False):
        """前向传播"""
        self.cache = {'A0': X}
        A = X

        for i in range(len(self.layer_sizes) - 1):
            W = self.parameters[f'W{i+1}']
            b = self.parameters[f'b{i+1}']

            Z = np.dot(A, W) + b
            self.cache[f'Z{i+1}'] = Z

            # 激活
            if i == len(self.layer_sizes) - 2:
                # 输出层
                exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True))
                A = exp_Z / np.sum(exp_Z, axis=1, keepdims=True)
            else:
                # 隐藏层
                A = np.maximum(0, Z)

                # Dropout
                if training and self.dropout_rate > 0:
                    mask = np.random.binomial(1, 1 - self.dropout_rate, size=A.shape)
                    A = A * mask / (1 - self.dropout_rate)

            self.cache[f'A{i+1}'] = A

        return A

    def compute_loss(self, Y_pred, Y_true, include_regularization=True):
        """计算损失"""
        m = Y_true.shape[0]
        epsilon = 1e-10
        Y_pred = np.clip(Y_pred, epsilon, 1 - epsilon)

        # 交叉熵
        ce_loss = -np.sum(Y_true * np.log(Y_pred)) / m

        if not include_regularization:
            return ce_loss

        # L2正则化
        l2_loss = 0
        for i in range(len(self.layer_sizes) - 1):
            l2_loss += np.sum(self.parameters[f'W{i+1}'] ** 2)

        total_loss = ce_loss + self.l2_lambda * l2_loss / (2 * m)

        return total_loss

    def backward(self, X, Y_true):
        """反向传播"""
        m = X.shape[0]
        self.gradients = {}

        # 输出层梯度
        A_final = self.cache[f'A{len(self.layer_sizes)-1}']
        dZ = A_final - Y_true

        for i in reversed(range(1, len(self.layer_sizes))):
            A_prev = self.cache[f'A{i-1}']
            W = self.parameters[f'W{i}']

            # 计算梯度
            self.gradients[f'dW{i}'] = np.dot(A_prev.T, dZ) / m
            self.gradients[f'db{i}'] = np.sum(dZ, axis=0, keepdims=True) / m

            # L2正则化梯度
            self.gradients[f'dW{i}'] += self.l2_lambda * W / m

            # 传播到前一层
            if i > 1:
                dA_prev = np.dot(dZ, W.T)
                Z_prev = self.cache[f'Z{i-1}']
                dZ = dA_prev * (Z_prev > 0)  # ReLU导数

    def update_parameters(self, learning_rate):
        """更新参数"""
        for i in range(1, len(self.layer_sizes)):
            self.parameters[f'W{i}'] -= learning_rate * self.gradients[f'dW{i}']
            self.parameters[f'b{i}'] -= learning_rate * self.gradients[f'db{i}']

    def train(self, X_train, Y_train, X_val, Y_val, epochs=200, learning_rate=0.1, patience=20):
        """训练循环"""
        train_losses = []
        val_losses = []
        best_val_loss = float('inf')
        best_params = None
        patience_counter = 0

        print(f"\n开始训练: {epochs} epochs, lr={learning_rate}")
        print("="*70)

        for epoch in range(epochs):
            # 训练
            Y_pred_train = self.forward(X_train, training=True)
            train_loss = self.compute_loss(Y_pred_train, Y_train)
            train_losses.append(train_loss)

            self.backward(X_train, Y_train)
            self.update_parameters(learning_rate)

            # 验证
            Y_pred_val = self.forward(X_val, training=False)
            val_loss = self.compute_loss(Y_pred_val, Y_val, include_regularization=False)
            val_losses.append(val_loss)

            # 早停逻辑
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                best_params = {k: v.copy() for k, v in self.parameters.items()}
                patience_counter = 0
            else:
                patience_counter += 1

            if (epoch + 1) % 20 == 0:
                print(f"Epoch {epoch+1:3d} | Train: {train_loss:.4f} | Val: {val_loss:.4f} | Patience: {patience_counter}")

            if patience_counter >= patience:
                print(f"\n早停触发！在epoch {epoch+1}")
                break

        # 恢复最佳参数
        if best_params:
            self.parameters = best_params

        print(f"\n训练完成！最佳验证损失: {best_val_loss:.4f}")

        return train_losses, val_losses
```

```python
def exercise1():
    print("\n练习1：L2正则化线性回归")
    print("="*60)

    # 生成数据
    np.random.seed(42)
    X = np.random.randn(100, 3)
    true_w = np.array([2.0, -1.0, 0.5])
    y = X @ true_w + np.random.randn(100) * 0.5

    # TODO: 实现普通线性回归
    # w = (X^T X)^(-1) X^T y
    w_normal = np.linalg.inv(X.T @ X) @ X.T @ y

    # TODO: 实现L2正则化线性回归
    # w = (X^T X + λI)^(-1) X^T y
    lambdas = [0.01, 0.1, 1.0, 10.0]
    w_ridge = {}

    for lam in lambdas:
        I = np.eye(X.shape[1])
        w_ridge[lam] = np.linalg.inv(X.T @ X + lam * I) @ X.T @ y

    # 输出结果
    print(f"真实权重: {true_w}")
    print(f"普通回归: {w_normal}")
    print(f"权重范数: {np.linalg.norm(w_normal):.4f}")
    print()

    for lam in lambdas:
        w = w_ridge[lam]
        print(f"L2 (λ={lam}): {w}, 范数: {np.linalg.norm(w):.4f}")

    # 预测对比
    X_test = np.random.randn(20, 3)
    y_test = X_test @ true_w + np.random.randn(20) * 0.5

    print("\n预测误差 (MSE):")
    print(f"普通回归: {np.mean((X_test @ w_normal - y_test)**2):.4f}")

    for lam in lambdas:
        pred = X_test @ w_ridge[lam]
        mse = np.mean((pred - y_test)**2)
        print(f"L2 (λ={lam}): {mse:.4f}")
```

## 🎯 本章要点总结

### 必须掌握的概念

1. **损失函数**：
   - 回归：MSE、MAE、Huber
   - 分类：BCE、交叉熵、Focal Loss

2. **正则化技术**：
   - L2：权重衰减
   - L1：稀疏化
   - Dropout：随机失活
   - 数据增强：增加多样性
   - 早停：防止过拟合

### 选择建议

| 问题类型 | 推荐损失 | 推荐正则化 |
|---------|---------|-----------|
| 普通回归 | MSE | L2 |
| 有异常值 | MAE/Huber | L2 |
| 二分类 | BCE | Dropout + L2 |
| 多分类 | 交叉熵 | Dropout + L2 |
| 类别不平衡 | Focal Loss | 数据增强 |

### 常见陷阱

⚠️ **注意：**
- L1需要调整学习率（不可导）
- Dropout在测试时要关闭
- 数据增强只在训练时使用
- 早停需要验证集
