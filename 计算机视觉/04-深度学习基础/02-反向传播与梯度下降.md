## 🎯 1. 梯度下降 - 寻找最优参数

### 1.1 什么是梯度下降？

#### 直观理解：下山的故事

想象你蒙着眼睛在山上，想要下到山谷最低点：

```
当前位置 → 环顾四周，找到最陡的下坡方向 → 迈出一步 → 重复
```

**关键点：**
- **梯度**：最陡的上坡方向
- **负梯度**：最陡的下坡方向
- **学习率**：每一步的步长

```python
import numpy as np
import matplotlib.pyplot as plt

def gradient_descent_visualization():
    """梯度下降可视化"""

    # 定义一个简单的函数：f(x) = x² + 2x + 1
    def f(x):
        return x**2 + 2*x + 1

    def df(x):
        """导数"""
        return 2*x + 2

    # 梯度下降过程
    x = 5.0  # 起点
    learning_rate = 0.1
    n_steps = 10

    trajectory = [x]
    gradients = []

    print("梯度下降过程演示")
    print("="*60)
    print(f"函数: f(x) = x² + 2x + 1")
    print(f"导数: f'(x) = 2x + 2")
    print(f"起点: x = {x}")
    print(f"学习率: {learning_rate}")
    print("="*60)

    for i in range(n_steps):
        grad = df(x)  # 计算梯度
        x = x - learning_rate * grad  # 更新

        trajectory.append(x)
        gradients.append(grad)

        print(f"步骤 {i+1:2d}: x = {x:7.4f}, f(x) = {f(x):7.4f}, 梯度 = {grad:7.4f}")

    print("="*60)
    print(f"最终结果: x = {x:.4f}, f(x) = {f(x):.4f}")
    print(f"理论最小值: x = -1.0000, f(x) = 0.0000")

    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

    # 左图：函数曲线和下降路径
    x_range = np.linspace(-6, 6, 100)
    y_range = f(x_range)

    ax1.plot(x_range, y_range, label='f(x) = x² + 2x + 1', linewidth=2)

    trajectory_y = [f(x) for x in trajectory]
    ax1.plot(trajectory, trajectory_y, 'ro-', label='下降路径', linewidth=2)

    # 标注每一步
    for i, (x_val, y_val) in enumerate(zip(trajectory, trajectory_y)):
        ax1.annotate(f'{i+1}', (x_val, y_val), xytext=(5, 5),
                    textcoords='offset points', fontsize=9, fontweight='bold')

    ax1.set_xlabel('x', fontsize=12)
    ax1.set_ylabel('f(x)', fontsize=12)
    ax1.set_title('梯度下降：寻找最小值', fontsize=14, fontweight='bold')
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    ax1.axhline(y=0, color='black', linestyle='--', alpha=0.3)
    ax1.axvline(x=-1, color='green', linestyle='--', alpha=0.5, label='最优解')

    # 右图：损失下降曲线
    ax2.plot(range(1, len(trajectory_y)+1), trajectory_y, 'o-', linewidth=2, markersize=8)
    ax2.set_xlabel('迭代次数', fontsize=12)
    ax2.set_ylabel('函数值', fontsize=12)
    ax2.set_title('损失下降过程', fontsize=14, fontweight='bold')
    ax2.grid(True, alpha=0.3)

    # 添加收敛说明
    ax2.text(0.5, 0.95, '损失逐渐减小\n趋近于最优值',
             transform=ax2.transAxes, ha='center', va='top',
             bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.5))

    plt.tight_layout()
```

```python
def multidimensional_gradient_descent():
    """二维梯度下降"""

    # 定义二维函数：f(x, y) = x² + 2y²
    def f(x, y):
        return x**2 + 2*y**2

    def gradient(x, y):
        """梯度 = [∂f/∂x, ∂f/∂y]"""
        return np.array([2*x, 4*y])

    # 梯度下降
    point = np.array([5.0, 3.0])  # 起点
    learning_rate = 0.1
    n_steps = 15

    print(f"学习率: {learning_rate}")
    print("="*60)

    for i in range(n_steps):
        grad = gradient(point[0], point[1])
        point = point - learning_rate * grad
        trajectory.append(point.copy())
```

## 🔄 2. 反向传播 - 计算梯度的魔法

### 2.1 为什么需要反向传播？

**问题：** 在神经网络中，我们如何计算损失函数对每个权重的梯度？

```
输入 → 隐藏层 → 输出层 → 损失
  W1      W2        W3
  ↓       ↓         ↓
  需要知道 ∂L/∂W1, ∂L/∂W2, ∂L/∂W3
```

**解决方案：** 链式法则 + 从后向前计算

---

### 2.2 链式法则详解

#### 从简单到复杂

```python
def chain_rule_demo():
    """链式法则演示"""

    print("链式法则：从简单到复杂")
    print("="*60)

    # 例子1：简单复合函数
    print("\n【例子1：简单复合】")
    print("函数: y = f(g(x)) = (2x + 1)²")

    def g(x):
        return 2*x + 1

    def f(u):
        return u**2

    def composite(x):
        return f(g(x))

    # 链式法则计算导数
    def composite_derivative(x):
        # f'(u) = 2u, g'(x) = 2
        # dy/dx = f'(g(x)) * g'(x) = 2(2x+1) * 2 = 4(2x+1)
        return 4*(2*x + 1)

    x = 2
    print(f"在x={x}处:")
    print(f"  g(x) = {g(x)}")
    print(f"  f(g(x)) = {f(g(x))}")
    print(f"  导数 = {composite_derivative(x)}")

    # 验证
    h = 0.0001
    numerical = (composite(x+h) - composite(x)) / h
    print(f"  数值验证: {numerical:.6f}")
    print(f"  误差: {abs(composite_derivative(x) - numerical):.10f}")

    # 例子2：三层复合
    print("\n【例子2：三层复合】")
    print("函数: y = f(g(h(x))) = (3x + 2)² + 1")

    def h(x):
        return 3*x + 2

    def triple_composite(x):
        return f(h(x)) + 1

    def triple_derivative(x):
        # dy/dx = f'(h(x)) * h'(x) = 2(3x+2) * 3 = 6(3x+2)
        return 6*(3*x + 2)

    x = 1
    print(f"在x={x}处:")
    print(f"  h(x) = {h(x)}")
    print(f"  f(h(x)) = {f(h(x))}")
    print(f"  y = {triple_composite(x)}")
    print(f"  导数 = {triple_derivative(x)}")

    # 数值验证
    numerical = (triple_composite(x+h) - triple_composite(x)) / h
    print(f"  数值验证: {numerical:.6f}")

    print("\n" + "="*60)
    print("链式法则核心：")
    print("  dy/dx = dy/du × du/dx")
    print("  可以扩展到任意多层！")
    print("="*60)
```

### 2.3 神经网络中的反向传播

#### 完整推导过程

```python
def neural_network_backward_demo():
    """神经网络反向传播详解"""

    print("神经网络反向传播完整推导")
    print("="*70)

    # 定义一个简单的网络：输入(2) → 隐藏(3) → 输出(1)
    # 激活函数：隐藏层用ReLU，输出层用线性（回归问题）

    # 参数
    W1 = np.array([[0.5, -0.2, 0.3], [0.1, 0.4, -0.1]])
    b1 = np.array([0.1, 0.2, 0.0])
    W2 = np.array([[0.2], [0.3], [-0.1]])
    b2 = np.array([0.0])

    # 输入和真实值
    X = np.array([[0.5, 0.8]])
    y_true = np.array([[1.2]])

    print("\n【网络结构】")
    print(f"输入: {X.shape} → 隐藏: {W1.shape} → 输出: {W2.shape}")

    # ===== 前向传播 =====
    print("\n【前向传播】")

    # 隐藏层
    z1 = np.dot(X, W1) + b1
    a1 = np.maximum(0, z1)  # ReLU

    # 输出层
    z2 = np.dot(a1, W2) + b2
    a2 = z2  # 线性激活

    y_pred = a2[0, 0]
    y_true_val = y_true[0, 0]

    print(f"隐藏层输入 z1: {z1}")
    print(f"隐藏层输出 a1: {a1}")
    print(f"输出层输入 z2: {z2}")
    print(f"预测值: {y_pred:.4f}, 真实值: {y_true_val:.4f}")

    # 损失（均方误差）
    loss = 0.5 * (y_pred - y_true_val)**2
    print(f"损失 L: {loss:.4f}")

    # ===== 反向传播 =====
    print("\n【反向传播】")
    print("目标：计算 ∂L/∂W1, ∂L/∂b1, ∂L/∂W2, ∂L/∂b2")

    # 步骤1：计算输出层的梯度
    print("\n步骤1：输出层梯度")
    dL_dy = y_pred - y_true_val  # ∂L/∂y
    print(f"  ∂L/∂y = {dL_dy:.4f}")

    # 步骤2：计算W2和b2的梯度
    print("\n步骤2：W2和b2的梯度")
    dL_dz2 = dL_dy  # ∂L/∂z2 = ∂L/∂y (因为z2 = y)
    dL_dW2 = np.dot(a1.T, dL_dz2)  # ∂L/∂W2 = a1^T × ∂L/∂z2
    dL_db2 = dL_dz2  # ∂L/∂b2 = ∂L/∂z2

    print(f"  ∂L/∂z2 = {dL_dz2:.4f}")
    print(f"  ∂L/∂W2 = {dL_dW2.T}")
    print(f"  ∂L/∂b2 = {dL_db2:.4f}")

    # 步骤3：传播到隐藏层
    print("\n步骤3：传播到隐藏层")
    dL_da1 = np.dot(dL_dz2, W2.T)  # ∂L/∂a1 = ∂L/∂z2 × W2^T
    print(f"  ∂L/∂a1 = {dL_da1}")

    # 步骤4：计算隐藏层激活的梯度（ReLU的导数）
    print("\n步骤4：ReLU导数")
    relu_grad = (z1 > 0).astype(float)  # ReLU导数：x>0时为1，否则为0
    print(f"  ReLU'(z1) = {relu_grad}")

    dL_dz1 = dL_da1 * relu_grad  # ∂L/∂z1 = ∂L/∂a1 × ReLU'(z1)
    print(f"  ∂L/∂z1 = {dL_dz1}")

    # 步骤5：计算W1和b1的梯度
    print("\n步骤5：W1和b1的梯度")
    dL_dW1 = np.dot(X.T, dL_dz1)  # ∂L/∂W1 = X^T × ∂L/∂z1
    dL_db1 = dL_dz1  # ∂L/∂b1 = ∂L/∂z1

    print(f"  ∂L/∂W1 = {dL_dW1}")
    print(f"  ∂L/∂b1 = {dL_db1}")

    print("\n" + "="*70)
    print("反向传播总结：")
    print("  1. 从输出开始，计算损失对输出的梯度")
    print("  2. 逐层向前传播梯度")
    print("  3. 每层使用链式法则计算")
    print("  4. 最终得到所有参数的梯度")
    print("="*70)

    # 验证：数值梯度
    print("\n【数值验证】")
    epsilon = 1e-5

    def compute_loss(W1, b1, W2, b2):
        z1 = np.dot(X, W1) + b1
        a1 = np.maximum(0, z1)
        z2 = np.dot(a1, W2) + b2
        y_pred = z2[0, 0]
        return 0.5 * (y_pred - y_true_val)**2

    # 验证W2[0,0]的梯度
    W2_plus = W2.copy()
    W2_plus[0, 0] += epsilon
    loss_plus = compute_loss(W1, b1, W2_plus, b2)
    numerical_grad = (loss_plus - loss) / epsilon

    print(f"W2[0,0]的解析梯度: {dL_dW2[0,0]:.6f}")
    print(f"W2[0,0]的数值梯度: {numerical_grad:.6f}")
    print(f"误差: {abs(dL_dW2[0,0] - numerical_grad):.10f}")

neural_network_backward_demo()
```

### 2.4 完整的反向传播实现

```python
class BackpropNetwork:
    """支持反向传播的神经网络"""

    def __init__(self, layer_sizes):
        """
        layer_sizes: [输入, 隐藏1, 隐藏2, ..., 输出]
        """
        self.layer_sizes = layer_sizes
        self.num_layers = len(layer_sizes)
        self.parameters = {}
        self.cache = {}

        # 初始化参数
        for i in range(self.num_layers - 1):
            self.parameters[f'W{i+1}'] = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * 0.1
            self.parameters[f'b{i+1}'] = np.zeros((1, layer_sizes[i+1]))

        print(f"创建网络: {' → '.join(map(str, layer_sizes))}")
        print(f"总参数: {sum(p.size for p in self.parameters.values())}")

    def relu(self, Z):
        return np.maximum(0, Z)

    def relu_derivative(self, Z):
        return (Z > 0).astype(float)

    def softmax(self, Z):
        exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True))
        return exp_Z / np.sum(exp_Z, axis=1, keepdims=True)

    def forward(self, X):
        """前向传播"""
        A = X
        self.cache['A0'] = A

        for i in range(self.num_layers - 1):
            W = self.parameters[f'W{i+1}']
            b = self.parameters[f'b{i+1}']

            Z = np.dot(A, W) + b
            self.cache[f'Z{i+1}'] = Z

            if i == self.num_layers - 2:
                A = self.softmax(Z)
            else:
                A = self.relu(Z)

            self.cache[f'A{i+1}'] = A

        return A

    def compute_loss(self, Y_pred, Y_true):
        """计算交叉熵损失"""
        m = Y_true.shape[0]
        # 避免log(0)
        epsilon = 1e-10
        Y_pred = np.clip(Y_pred, epsilon, 1 - epsilon)

        # 交叉熵
        loss = -np.sum(Y_true * np.log(Y_pred)) / m
        return loss

    def backward(self, X, Y_true):
        """反向传播"""
        m = X.shape[0]
        self.gradients = {}

        # 输出层梯度
        A_final = self.cache[f'A{self.num_layers-1}']
        dZ = A_final - Y_true  # Softmax + 交叉熵的简化梯度

        for i in reversed(range(1, self.num_layers)):
            A_prev = self.cache[f'A{i-1}']
            W = self.parameters[f'W{i}']

            # 计算梯度
            self.gradients[f'dW{i}'] = np.dot(A_prev.T, dZ) / m
            self.gradients[f'db{i}'] = np.sum(dZ, axis=0, keepdims=True) / m

            # 传播到前一层
            if i > 1:
                dA_prev = np.dot(dZ, W.T)
                Z_prev = self.cache[f'Z{i-1}']
                dZ = dA_prev * self.relu_derivative(Z_prev)

    def update_parameters(self, learning_rate):
        """更新参数"""
        for i in range(1, self.num_layers):
            self.parameters[f'W{i}'] -= learning_rate * self.gradients[f'dW{i}']
            self.parameters[f'b{i}'] -= learning_rate * self.gradients[f'db{i}']

    def train(self, X, Y, epochs=100, learning_rate=0.1, verbose=True):
        """训练循环"""
        losses = []

        for epoch in range(epochs):
            # 前向传播
            Y_pred = self.forward(X)

            # 计算损失
            loss = self.compute_loss(Y_pred, Y)
            losses.append(loss)

            # 反向传播
            self.backward(X, Y)

            # 更新参数
            self.update_parameters(learning_rate)

            if verbose and (epoch + 1) % 10 == 0:
                print(f"Epoch {epoch+1:3d}/{epochs}, Loss: {loss:.6f}")

        return losses

# 测试网络
```

```python
def sgd_demo():
    """随机梯度下降演示"""

    print("随机梯度下降 (SGD)")
    print("="*60)

    # 模拟数据
    np.random.seed(42)
    X = np.random.randn(100, 2)
    y = 2 * X[:, 0] + 3 * X[:, 1] + np.random.randn(100) * 0.1

    # 线性回归：y = w1*x1 + w2*x2 + b
    w = np.array([0.0, 0.0])
    b = 0.0

    # SGD参数
    learning_rate = 0.01
    n_epochs = 50
    batch_size = 10

    losses = []

    print(f"数据量: {len(X)}")
    print(f"批次大小: {batch_size}")
    print(f"学习率: {learning_rate}")
    print("="*60)

    for epoch in range(n_epochs):
        # 打乱数据
        indices = np.random.permutation(len(X))
        X_shuffled = X[indices]
        y_shuffled = y[indices]

        epoch_loss = 0

        # 小批量梯度下降
        for i in range(0, len(X), batch_size):
            X_batch = X_shuffled[i:i+batch_size]
            y_batch = y_shuffled[i:i+batch_size]

            # 前向传播
            y_pred = np.dot(X_batch, w) + b

            # 计算梯度
            error = y_pred - y_batch
            dw = np.dot(X_batch.T, error) / len(X_batch)
            db = np.sum(error) / len(X_batch)

            # 更新
            w -= learning_rate * dw
            b -= learning_rate * db

            epoch_loss += np.mean(error**2)

        losses.append(epoch_loss / (len(X) // batch_size))

        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1:2d}: Loss = {losses[-1]:.4f}, w = {w}, b = {b:.4f}")

    print("\n最终结果:")
    print(f"  学到的参数: w = {w}, b = {b:.4f}")
    print(f"  真实参数: w = [2, 3], b ≈ 0")

    # 可视化
    plt.figure(figsize=(12, 5))

    plt.subplot(1, 2, 1)
    plt.plot(losses, linewidth=2)
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.title('SGD训练过程')
    plt.grid(True, alpha=0.3)

    plt.subplot(1, 2, 2)
    plt.scatter(X[:, 0], y, alpha=0.5, label='数据')
    x_range = np.array([X[:, 0].min(), X[:, 0].max()])
    y_pred_line = w[0] * x_range + b
    plt.plot(x_range, y_pred_line, 'r-', linewidth=2, label='拟合直线')
    plt.xlabel('x1')
    plt.ylabel('y')
    plt.title('拟合结果')
    plt.legend()
    plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.show()
```

```python
def momentum_demo():

    def loss(x, y):
        return 0.01 * x**2 + y**2

    def gradient(x, y):
        return np.array([0.02 * x, 2 * y])

    # 普通SGD
    x_sgd, y_sgd = 5.0, 5.0
    lr = 0.1
    trajectory_sgd = [(x_sgd, y_sgd)]

    for i in range(30):
        grad = gradient(x_sgd, y_sgd)
        x_sgd -= lr * grad[0]
        y_sgd -= lr * grad[1]
        trajectory_sgd.append((x_sgd, y_sgd))

    # 动量优化
    x_mom, y_mom = 5.0, 5.0
    velocity = np.array([0.0, 0.0])
    beta = 0.9  # 动量系数
    trajectory_mom = [(x_mom, y_mom)]

    for i in range(30):
        grad = gradient(x_mom, y_mom)
        velocity = beta * velocity + (1 - beta) * grad
        x_mom -= lr * velocity[0]
        y_mom -= lr * velocity[1]
        trajectory_mom.append((x_mom, y_mom))
```

```python
def adam_demo():
    """Adam优化器演示"""

    print("Adam优化器")
    print("="*60)

    # 简单的二次函数
    def f(x):
        return x**2

    def df(x):
        return 2*x

    # Adam参数
    def adam_update(x, m, v, t, beta1=0.9, beta2=0.999, epsilon=1e-8, lr=0.1):
        grad = df(x)

        # 更新一阶矩估计
        m = beta1 * m + (1 - beta1) * grad

        # 更新二阶矩估计
        v = beta2 * v + (1 - beta2) * grad**2

        # 偏差修正
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # 更新参数
        x = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        return x, m, v

    # 比较不同优化器
    x_start = 5.0

    # SGD
    x_sgd = x_start
    trajectory_sgd = [x_sgd]
    for i in range(30):
        x_sgd = x_sgd - 0.1 * df(x_sgd)
        trajectory_sgd.append(x_sgd)

    # Adam
    x_adam = x_start
    m, v = 0.0, 0.0
    trajectory_adam = [x_adam]
    for t in range(1, 31):
        x_adam, m, v = adam_update(x_adam, m, v, t)
        trajectory_adam.append(x_adam)

    # 可视化
    plt.figure(figsize=(12, 5))

    # 路径对比
    plt.subplot(1, 2, 1)
    x_range = np.linspace(-6, 6, 100)
```

```python
def complete_training_loop():
    """完整的训练循环示例"""

    # 创建数据集

    # 生成螺旋数据（非线性可分）
    def generate_spiral_data(n_samples=200):
        n = n_samples // 2
        data = []
        labels = []

        for class_id in range(2):
            r = np.linspace(0, 1, n)
            t = np.linspace(class_id * 4, (class_id + 1) * 4, n) + 0.2 * np.random.randn(n)

            x = r * np.sin(t)
            y = r * np.cos(t)

            data.append(np.stack([x, y], axis=1))
            labels.append(np.full(n, class_id))

        X = np.vstack(data)
        y = np.hstack(labels)

        # 打乱
        idx = np.random.permutation(len(X))
        return X[idx], y[idx]

    X, y = generate_spiral_data(200)
    Y = np.eye(2)[y]  # one-hot

    # 划分训练集和验证集
    split = int(0.8 * len(X))

    Y_train, Y_val = Y[:split], Y[split:]
```

```python
def exercise1():
    # 数据
    X = np.array([[0.5, 0.3]])
    y_true = np.array([[1.0]])

    # 参数
    W1 = np.array([[0.1, 0.2], [0.3, 0.4]])
    b1 = np.array([[0.1, 0.2]])
    W2 = np.array([[0.5], [0.6]])
    b2 = np.array([[0.3]])

    print("练习1：手动反向传播")
    print("="*60)

    # TODO: 实现前向传播
    # 隐藏层
    z1 = np.dot(X, W1) + b1
    a1 = np.maximum(0, z1)  # ReLU
    # 输出层
    z2 = np.dot(a1, W2) + b2
    a2 = z2  # 线性

    y_pred = a2[0, 0]
    loss = 0.5 * (y_pred - y_true[0, 0])**2

    print(f"预测值: {y_pred:.4f}")
    print(f"损失: {loss:.4f}")

    # TODO: 手动实现反向传播
    # 1. 输出层梯度
    dL_dy = y_pred - y_true[0, 0]
    dL_dz2 = dL_dy

    # 2. W2, b2梯度
    dL_dW2 = np.dot(a1.T, dL_dz2)
    dL_db2 = dL_dz2

    # 3. 传播到隐藏层
    dL_da1 = np.dot(dL_dz2, W2.T)

    # 4. ReLU导数
    relu_grad = (z1 > 0).astype(float)
    dL_dz1 = dL_da1 * relu_grad

    # 5. W1, b1梯度
    dL_dW1 = np.dot(X.T, dL_dz1)
    dL_db1 = dL_dz1

    print("\n解析梯度:")
    print(f"dL/dW2 = {dL_dW2.T}")
    print(f"dL/db2 = {dL_db2}")
    print(f"dL/dW1 = {dL_dW1}")
    print(f"dL/db1 = {dL_db1}")

    # 数值验证
    epsilon = 1e-5

    def compute_loss(W1, b1, W2, b2):
        z1 = np.dot(X, W1) + b1
        a1 = np.maximum(0, z1)
        z2 = np.dot(a1, W2) + b2
        return 0.5 * (z2[0,0] - y_true[0,0])**2

    # 验证W2[0,0]
    W2_plus = W2.copy()
    W2_plus[0,0] += epsilon
    numerical_grad = (compute_loss(W1, b1, W2_plus, b2) - loss) / epsilon

    print(f"\n数值验证 W2[0,0]:")
    print(f"  解析梯度: {dL_dW2[0,0]:.6f}")
    print(f"  数值梯度: {numerical_grad:.6f}")
    print(f"  误差: {abs(dL_dW2[0,0] - numerical_grad):.10f}")
```

```python
    adam_path = [(x_adam, y_adam)]

    for t in range(1, 31):
        g = grad(x_adam, y_adam)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        x_adam -= update[0]
        y_adam -= update[1]
        adam_path.append((x_adam, y_adam))

    # 可视化
    plt.figure(figsize=(12, 5))

    # 路径
    plt.subplot(1, 2, 1)
    x_range = np.linspace(-6, 6, 100)
    y_range = np.linspace(-6, 6, 100)
    X, Y = np.meshgrid(x_range, y_range)
    Z = f(X, Y)

    plt.contour(X, Y, Z, levels=20, cmap='viridis')

    sgd_x = [p[0] for p in sgd_path]
    sgd_y = [p[1] for p in sgd_path]
    plt.plot(sgd_x, sgd_y, 'o-', color='red', label='SGD', linewidth=2)

    mom_x = [p[0] for p in mom_path]
    mom_y = [p[1] for p in mom_path]
    plt.plot(mom_x, mom_y, 's-', color='blue', label='Momentum', linewidth=2)

    adam_x = [p[0] for p in adam_path]
    adam_y = [p[1] for p in adam_path]
    plt.plot(adam_x, adam_y, '^-', color='orange', label='Adam', linewidth=2)

    plt.scatter(0, 0, s=100, c='green', marker='*', zorder=5)
    plt.xlabel('x')
    plt.ylabel('y')
    plt.title('优化器路径对比', fontweight='bold')
    plt.legend()
    plt.grid(True, alpha=0.3)

    # 损失
    plt.subplot(1, 2, 2)
    sgd_loss = [f(p[0], p[1]) for p in sgd_path]
    mom_loss = [f(p[0], p[1]) for p in mom_path]
    adam_loss = [f(p[0], p[1]) for p in adam_path]

    plt.plot(sgd_loss, 'o-', color='red', label='SGD', linewidth=2)
    plt.plot(mom_loss, 's-', color='blue', label='Momentum', linewidth=2)
    plt.plot(adam_loss, '^-', color='orange', label='Adam', linewidth=2)

    plt.xlabel('迭代次数')
    plt.ylabel('损失')
    plt.title('收敛速度对比', fontweight='bold')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.yscale('log')

    plt.tight_layout()
    plt.show()

    print("\n结果分析:")
    print(f"  SGD最终损失: {sgd_loss[-1]:.6f}")
    print(f"  Momentum最终损失: {mom_loss[-1]:.6f}")
    print(f"  Adam最终损失: {adam_loss[-1]:.6f}")
    print("\n结论:")
    print("  - Adam通常收敛最快")
    print("  - Momentum减少震荡")
    print("  - SGD简单但可能慢")
```
