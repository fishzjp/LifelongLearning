## ❌ 训练问题

### 1. LeNet-5训练不收敛

**症状**: Loss不下降或剧烈震荡

**原因分析**:
- 学习率过大/过小
- 数据未标准化
- 权重初始化不当
- 激活函数选择错误

**解决方案**:
```python
# 1. 学习率调整
optimizer = torch.optim.SGD(model.parameters(), lr=0.01, momentum=0.9)
# MNIST推荐: 0.01 ~ 0.1

# 2. 数据标准化（关键！）
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))  # MNIST统计值
])

# 3. 权重初始化
def init_weights(m):
    if isinstance(m, nn.Conv2d) or isinstance(m, nn.Linear):
        nn.init.xavier_uniform_(m.weight)
        nn.init.constant_(m.bias, 0)
model.apply(init_weights)

# 4. 添加BatchNorm（改进LeNet）
class LeNet5_BN(nn.Module):
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(1, 6, 5, padding=2),
            nn.BatchNorm2d(6),  # 添加BN
            nn.ReLU(),
            nn.AvgPool2d(2, 2),
            nn.Conv2d(6, 16, 5),
            nn.BatchNorm2d(16),  # 添加BN
            nn.ReLU(),
            nn.AvgPool2d(2, 2),
        )
        # ...
```

**调试步骤**:
1. 打印输入数据的均值和方差
2. 检查权重初始化值
3. 使用小batch_size观察loss变化
4. 添加梯度裁剪

### 2. VGG训练过慢

**症状**: 每轮训练时间过长，显存不足

**原因**: 参数量大（1.38亿），计算量大（15.3G FLOPs）

**解决方案**:
```python
# 1. 使用预训练模型（强烈推荐）
model = torchvision.models.vgg16(pretrained=True)
# 只训练分类器，冻结卷积层

# 2. 冻结部分层
for param in model.features.parameters():
    param.requires_grad = False
# 只训练classifier
optimizer = torch.optim.Adam(model.classifier.parameters(), lr=0.001)

# 3. 使用混合精度（速度↑2-3倍）
from torch.cuda.amp import autocast, GradScaler
scaler = GradScaler()
for inputs, labels in train_loader:
    with autocast():
        outputs = model(inputs)
        loss = criterion(outputs, labels)
    scaler.scale(loss).backward()
    scaler.step(optimizer)
    scaler.update()

# 4. 减小batch_size
train_loader = DataLoader(dataset, batch_size=16, shuffle=True)

# 5. 使用更小的VGG
model = torchvision.models.vgg11(pretrained=True)  # 1.33亿参数
```

**性能对比**:
- VGG16 + 全训练: 10小时
- VGG16 + 预训练: 2小时
- VGG16 + 冻结+混合精度: 30分钟

### 3. ResNet梯度爆炸

**症状**: Loss变成NaN，训练崩溃

**原因**: 深度网络梯度累积，初始学习率过大

**解决方案**:
```python
# 1. 梯度裁剪（防止爆炸）
torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)

# 2. BatchNorm（必须有）
# 确保所有卷积层后都有BN

# 3. 学习率warmup（防止初期不稳定）
def warmup_lr(epoch):
    if epoch < 5:
        return 0.01 * (epoch + 1) / 5
    else:
        return 0.01

scheduler = torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=warmup_lr)

# 4. 权重初始化检查
for m in model.modules():
    if isinstance(m, nn.Conv2d):
        nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')

# 5. 检查数据
# 确保输入数据没有NaN/Inf
assert not torch.isnan(inputs).any()
assert not torch.isinf(inputs).any()
```

**预防措施**:
- 使用小学习率初始值 (0.01)
- 添加warmup阶段
- 监控梯度范数
- 定期保存checkpoint

### 4. DenseNet内存溢出

**症状**: CUDA out of memory

**原因**: 特征拼接导致内存消耗大

**解决方案**:
```python
# 1. 减小batch_size
train_loader = DataLoader(dataset, batch_size=8, shuffle=True)

# 2. 降低growth_rate
model = DenseNet(growth_rate=16, block_config=(6, 12, 24, 16))

# 3. 减少DenseBlock层数
model = DenseNet(growth_rate=32, block_config=(4, 8, 16, 8))

# 4. 使用gradient checkpointing
from torch.utils.checkpoint import checkpoint
def forward(self, x):
    features = [x]
    for layer in self.layers:
        # 检查点技术
        new_feature = checkpoint(layer, torch.cat(features, dim=1))
        features.append(new_feature)
    return torch.cat(features, dim=1)

# 5. 使用更小的压缩率
model = DenseNet(compression=0.3)  # 默认0.5
```

## ❌ 架构问题

### 1. 残差连接维度不匹配

**问题**: `x + F(x)` 时维度不同，无法相加

**解决方案**:
```python
class BasicBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3, stride, 1)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3, 1, 1)
        self.bn2 = nn.BatchNorm2d(out_channels)
        
        # 关键: 使用shortcut调整维度
        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, 1, stride),
                nn.BatchNorm2d(out_channels)
            )
    
    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        
        # 维度必须匹配！
        out += self.shortcut(x)
        return F.relu(out)

# 测试维度匹配
block = BasicBlock(64, 128, stride=2)
x = torch.randn(1, 64, 32, 32)
out = block(x)
print(f"输入: {x.shape}")  # [1, 64, 32, 32]
print(f"输出: {out.shape}")  # [1, 128, 16, 16]
```

**常见错误**:
- 忘记添加shortcut
- stride=1但通道数变化
- shortcut中没有BN

### 2. DenseNet通道爆炸

**问题**: 特征拼接导致通道数快速增长

**解决方案**:
```python
# 使用TransitionLayer压缩
class TransitionLayer(nn.Module):
    def __init__(self, in_channels, out_channels):
        super().__init__()
        self.conv = nn.Sequential(
            nn.BatchNorm2d(in_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(in_channels, out_channels, 1),  # 1×1卷积压缩
            nn.AvgPool2d(2, stride=2)
        )
    
    def forward(self, x):
        return self.conv(x)

# 在DenseNet中使用
# 每个DenseBlock后接TransitionLayer
# compression=0.5 将通道数减半

# 通道增长示例:
# DenseBlock1: 64 → 64 + 6×32 = 256
# Transition1: 256 → 128 (压缩×0.5)
# DenseBlock2: 128 → 128 + 12×32 = 512
# Transition2: 512 → 256 (压缩×0.5)
```

**监控通道数**:
```python
def count_channels(model):
    """监控DenseNet通道增长"""
    x = torch.randn(1, 3, 224, 224)
    x = model.conv1(x)
    x = model.bn1(x)
    x = model.relu(x)
    x = model.maxpool(x)
    
    for i, block in enumerate(model.features):
        before = x.shape[1]
        x = block(x)
        after = x.shape[1]
        print(f"Block {i}: {before} → {after}")
```

### 3. 梯度消失/爆炸

**症状**: 浅层梯度接近0或极大

**解决方案**:
```python
# 1. 检查是否有残差连接
# ResNet/DenseNet天然解决

# 2. 添加BatchNorm
nn.Conv2d(..., bias=False)
nn.BatchNorm2d(out_channels)

# 3. 使用合适的初始化
for m in model.modules():
    if isinstance(m, nn.Conv2d):
        nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
    elif isinstance(m, nn.BatchNorm2d):
        nn.init.constant_(m.weight, 1)
        nn.init.constant_(m.bias, 0)

# 4. 梯度裁剪
torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)

# 5. 监控梯度
def monitor_gradients(model):
    """检查各层梯度"""
    for name, param in model.named_parameters():
        if param.grad is not None:
            grad_norm = param.grad.norm().item()
            print(f"{name}: {grad_norm:.6f}")
            if grad_norm > 100:
                print(f"  ⚠️ 梯度过大!")
            if grad_norm < 1e-7:
                print(f"  ⚠️ 梯度消失!")
```

```python
class TrainingMonitor:
    def __init__(self):
        self.history = {'loss': [], 'acc': [], 'lr': []}
    
    def log(self, loss, acc, lr):
        self.history['loss'].append(loss)
        self.history['acc'].append(acc)
        self.history['lr'].append(lr)
    
    def plot(self):
        import matplotlib.pyplot as plt
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 4))
        
        # Loss曲线
        ax1.plot(self.history['loss'])
        ax1.set_title('Training Loss')
        ax1.set_xlabel('Epoch')
        ax1.set_ylabel('Loss')
        
        # Accuracy曲线
        ax2.plot(self.history['acc'])
        ax2.set_title('Training Accuracy')
        ax2.set_xlabel('Epoch')
        ax2.set_ylabel('Accuracy')
        
        plt.tight_layout()
        plt.show()
```

## 📊 常见错误速查

| 错误现象 | 可能原因 | 快速修复 |
|----------|----------|----------|
| Loss = NaN | 学习率过大 | lr *= 0.1 |
| Loss不下降 | 数据未标准化 | 添加Normalize |
| 准确率低 | 模型太浅 | 增加深度 |
| 内存溢出 | batch_size过大 | 减小batch_size |
| 训练极慢 | 未使用预训练 | 加载预训练权重 |
| 梯度爆炸 | 梯度裁剪 | clip_grad_norm_ |
| 过拟合 | 数据增强不足 | 增强数据 |
| 维度错误 | shortcut问题 | 检查维度匹配 |

### 步骤5: 监控训练
```python
# 训练循环监控
for epoch in range(epochs):
    model.train()
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        # 前向传播
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        
        # 检查异常
        if torch.isnan(loss) or torch.isinf(loss):
            print(f"NaN/Inf detected at epoch {epoch}, batch {batch_idx}")
            break
        
        # 反向传播
        optimizer.zero_grad()
        loss.backward()
        
        # 梯度检查
        grad_norm = torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        
        # 更新参数
        optimizer.step()
        
        # 打印进度
        if batch_idx % 100 == 0:
            print(f"Epoch {epoch}, Batch {batch_idx}, Loss: {loss.item():.4f}, Grad: {grad_norm:.4f}")
```

### 修复1: 数据标准化
```python
# 计算数据集统计值
def compute_stats(dataloader):
    mean = 0.0
    std = 0.0
    total_pixels = 0
    
    for inputs, _ in dataloader:
        # inputs: [B, C, H, W]
        mean += inputs.mean(dim=(0, 2, 3)) * inputs.size(0)
        std += inputs.std(dim=(0, 2, 3)) * inputs.size(0)
        total_pixels += inputs.size(0)
    
    mean /= total_pixels
    std /= total_pixels
    
    return mean, std

# 使用
mean, std = compute_stats(train_loader)
transform = transforms.Normalize(mean, std)
```

```python
def check_gradients(model):
    """检查梯度问题"""
    for name, param in model.named_parameters():
        if param.grad is None:
            print(f"{name}: No gradient")
        elif torch.isnan(param.grad).any():
            print(f"{name}: NaN gradient")
        elif torch.isinf(param.grad).any():
            print(f"{name}: Inf gradient")
        elif param.grad.abs().max() > 100:
            print(f"{name}: Large gradient ({param.grad.abs().max():.2f})")
        elif param.grad.abs().max() < 1e-7:
            print(f"{name}: Small gradient ({param.grad.abs().max():.2f})")
```
