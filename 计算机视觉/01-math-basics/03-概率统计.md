## 3.1 概率基础

### 什么是概率？

概率描述事件发生的可能性，范围是0到1。

```python

# 依赖: matplotlib, numpy
# 安装: pip install matplotlib numpy
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import norm

# 简单概率计算

def probability_demo():
    # 抛硬币
    coin = {'正面': 0.5, '反面': 0.5}

    # 掷骰子
    dice = {i: 1/6 for i in range(1, 7)}

    # 图像分类的概率解释
    # 假设模型预测一张图是猫的概率是0.8，是狗的概率是0.2
    image_prob = {'猫': 0.8, '狗': 0.2}

    print("概率示例：")
    print(f"抛硬币：{coin}")
    print(f"掷骰子：{dice}")
    print(f"图像分类：{image_prob}")

    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 4))

    # 硬币
    axes[0].bar(coin.keys(), coin.values(), color=['red', 'blue'])
    axes[0].set_title('抛硬币概率')
    axes[0].set_ylabel('概率')
    axes[0].set_ylim(0, 1)

    # 骰子
    axes[1].bar(dice.keys(), dice.values(), color='green')
    axes[1].set_title('掷骰子概率')
    axes[1].set_xlabel('点数')
    axes[1].set_ylabel('概率')

    # 图像分类
    axes[2].bar(image_prob.keys(), image_prob.values(), color=['orange', 'purple'])
    axes[2].set_title('图像分类概率')
    axes[2].set_ylabel('概率')
    axes[2].set_ylim(0, 1)

    plt.tight_layout()
    plt.show()

probability_demo()
```

## 3.2 贝叶斯定理：从已知推断未知

### 为什么要学贝叶斯？

贝叶斯定理是**机器学习分类器**的理论基础，特别是朴素贝叶斯分类器。在计算机视觉中，它帮助我们：
- **图像分类**：根据像素特征判断图片内容
- **目标检测**：从局部特征推断目标身份
- **噪声处理**：从含噪图像恢复原始信息

### 直观理解：贝叶斯思维的本质

想象你在玩一个猜谜游戏：
- **先验概率**：游戏开始前的猜测（基于常识）
- **新证据**：游戏过程中获得的线索
- **后验概率**：结合线索后的更新猜测

**核心思想：用新证据更新你的信念**

```
贝叶斯公式：
                似然 × 先验
    后验概率 = ——————————————
                   证据

P(A|B) = P(B|A) × P(A) / P(B)

其中：
- P(A)：先验概率 - 事件A发生的初始概率
- P(B|A)：似然 - 如果A发生，观察到B的可能性
- P(B)：证据 - 观察到B的总概率（归一化因子）
- P(A|B)：后验概率 - 观察到B后，事件A发生的概率
```

### 为什么贝叶斯如此强大？

**案例：罕见病检测的陷阱**

假设某种罕见病发病率仅1%，检测准确率99%：
- 你检测结果为阳性，真的生病了吗？

```python
def bayes_medical_deep_dive():
    """深入理解贝叶斯：罕见病检测"""

    # ===== 先验知识 =====
    P_disease = 0.01      # 1%的人患病（罕见！）
    P_healthy = 0.99      # 99%的人健康

    # ===== 检测性能 =====
    P_positive_given_disease = 0.99    # 真患病时阳性（99%敏感度）
    P_positive_given_healthy = 0.01    # 健康时阳性（1%误报率）

    print("="*50)
    print("贝叶斯分析：罕见病检测")
    print("="*50)

    print("\n【已知信息】")
    print(f"  人群患病率（先验）：{P_disease:.1%}")
    print(f"  检测敏感度：{P_positive_given_disease:.1%}")
    print(f"  误报率：{P_positive_given_healthy:.1%}")

    # ===== 计算过程 =====
    print("\n【计算过程】")

    # 步骤1：计算证据 P(阳性)
    # 在10000人中：
    # - 真病人：10000 × 1% = 100人，其中99人检测阳性
    # - 健康人：10000 × 99% = 9900人，其中99人检测阳性
    # 总阳性：99 + 99 = 198人

    P_positive = (P_positive_given_disease * P_disease +
                  P_positive_given_healthy * P_healthy)

    print(f"  P(阳性) = P(阳性|患病)×P(患病) + P(阳性|健康)×P(健康)")
    print(f"         = {P_positive_given_disease:.2f}×{P_disease:.2f} + {P_positive_given_healthy:.2f}×{P_healthy:.2f}")
    print(f"         = {P_positive:.4f}")

    # 步骤2：贝叶斯计算
    P_disease_given_positive = (P_positive_given_disease * P_disease) / P_positive

    print(f"\n  P(患病|阳性) = P(阳性|患病)×P(患病) / P(阳性)")
    print(f"              = {P_positive_given_disease:.2f}×{P_disease:.2f} / {P_positive:.4f}")
    print(f"              = {P_disease_given_positive:.1%}")

    # ===== 直观解释 =====
    print("\n【直观解释】")
    print(f"  在10000人中检测：")
    print(f"  - 真病人：100人，99人阳性 ✓")
    print(f"  - 健康人：9900人，99人阳性 ✗（误报）")
    print(f"  - 总阳性：198人，其中真病人仅99人")
    print(f"  → 阳性中真病人的比例：99/198 = {P_disease_given_positive:.1%}")

    print(f"\n【结论】")
    print(f"  即使检测准确率99%，由于疾病罕见，")
    print(f"  阳性结果中只有{P_disease_given_positive:.1%}是真病人！")
    print(f"  这就是『先验概率』的力量！")

bayes_medical_deep_dive()
```

### 计算机视觉中的贝叶斯：图像分类实战

```python
def bayes_image_classification_detailed():
    """贝叶斯在图像分类中的详细应用"""

    print("="*60)
    print("计算机视觉案例：猫狗分类器")
    print("="*60)

    # ===== 场景设定 =====
    print("\n【任务】")
    print("  根据『耳朵形状』特征，分类图片是猫还是狗")
    print("  特征：耳朵是否竖直（是/否）")

    # ===== 训练数据统计 =====
    print("\n【训练集统计】")
    P_cat = 0.4      # 40%是猫
    P_dog = 0.6      # 60%是狗
    print(f"  猫的比例：{P_cat:.0%}")
    print(f"  狗的比例：{P_dog:.0%}")

    # ===== 特征统计 =====
    P_upright_given_cat = 0.7    # 70%的猫耳朵竖直
    P_upright_given_dog = 0.3    # 30%的狗耳朵竖直
    print(f"\n【特征分布】")
    print(f"  P(竖直耳朵|猫) = {P_upright_given_cat:.0%}")
    print(f"  P(竖直耳朵|狗) = {P_upright_given_dog:.0%}")

    # ===== 推理过程 =====
    print("\n【推理：检测到竖直耳朵，是猫还是狗？】")

    # 计算证据
    P_upright = (P_upright_given_cat * P_cat +
                 P_upright_given_dog * P_dog)

    print(f"\n步骤1：计算P(竖直耳朵)")
    print(f"  P(竖直耳朵) = P(竖直|猫)×P(猫) + P(竖直|狗)×P(狗)")
    print(f"             = {P_upright_given_cat:.1f}×{P_cat:.1f} + {P_upright_given_dog:.1f}×{P_dog:.1f}")
    print(f"             = {P_upright:.2f}")

    # 贝叶斯计算
    P_cat_given_upright = (P_upright_given_cat * P_cat) / P_upright
    P_dog_given_upright = (P_upright_given_dog * P_dog) / P_upright

    print(f"\n步骤2：计算后验概率")
    print(f"  P(猫|竖直耳朵) = {P_upright_given_cat:.1f}×{P_cat:.1f}/{P_upright:.2f} = {P_cat_given_upright:.3f}")
    print(f"  P(狗|竖直耳朵) = {P_upright_given_dog:.1f}×{P_dog:.1f}/{P_upright:.2f} = {P_dog_given_upright:.3f}")

    # 决策
    print(f"\n【决策】")
    decision = "猫" if P_cat_given_upright > P_dog_given_upright else "狗"
    confidence = max(P_cat_given_upright, P_dog_given_upright)
    print(f"  分类结果：{decision}")
    print(f"  置信度：{confidence:.1%}")

    # ===== 多特征扩展 =====
    print("\n【扩展：加入多个特征】")
    print("  实际的分类器会使用多个特征：")
    print("  - 耳朵形状")
    print("  - 瞳孔形状")
    print("  - 胡须长度")
    print("  - 体型大小")
    print("  - 毛色模式")
    print("  → 朴素贝叶斯：假设特征独立，简化计算")

    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

    # 先验 vs 后验
    classes = ['猫', '狗']
    prior = [P_cat, P_dog]
    posterior = [P_cat_given_upright, P_dog_given_upright]

    x = np.arange(len(classes))
    width = 0.35

    ax1.bar(x - width/2, prior, width, label='先验概率', alpha=0.8, color='skyblue')
    ax1.bar(x + width/2, posterior, width, label='后验概率', alpha=0.8, color='lightcoral')

    ax1.set_xlabel('类别')
    ax1.set_ylabel('概率')
    ax1.set_title('先验 vs 后验概率')
    ax1.set_xticks(x)
    ax1.set_xticklabels(classes)
    ax1.legend()
    ax1.grid(True, alpha=0.3, axis='y')

    # 添加数值标注
    for i, (p, post) in enumerate(zip(prior, posterior)):
        ax1.text(i - width/2, p + 0.02, f'{p:.2f}', ha='center', va='bottom')
        ax1.text(i + width/2, post + 0.02, f'{post:.2f}', ha='center', va='bottom')

    # 似然比
    likelihood_ratio = P_upright_given_cat / P_upright_given_dog
    ax2.bar(['似然比'], [likelihood_ratio], color='purple', alpha=0.7)
    ax2.axhline(y=1, color='red', linestyle='--', label='基准线')
    ax2.set_ylabel('比值')
    ax2.set_title(f'似然比（猫/狗）= {likelihood_ratio:.1f}')
    ax2.legend()
    ax2.grid(True, alpha=0.3, axis='y')
    ax2.text(0, likelihood_ratio + 0.1, f'竖直耳朵使猫的\n可能性增加{likelihood_ratio:.1f}倍',
             ha='center', va='bottom')

    plt.tight_layout()
    plt.show()

bayes_image_classification_detailed()
```

## 3.3 常见概率分布

### 正态分布（高斯分布）

自然界中最常见的分布，钟形曲线。

```python
def normal_distribution_demo():
    """正态分布详解"""

    from scipy.stats import norm

    # 正态分布参数
    mu = 0      # 均值
    sigma = 1   # 标准差

    # 生成数据
    x = np.linspace(-4, 4, 100)
    y = norm.pdf(x, mu, sigma)

    plt.figure(figsize=(12, 5))

    # 左图：概率密度函数
    plt.subplot(1, 2, 1)
    plt.plot(x, y, linewidth=2, label=f'N({mu}, {sigma}²)')

    # 标注68-95-99.7规则
    x_1sig = x[(x >= -1) & (x <= 1)]
    x_2sig = x[(x >= -2) & (x <= 2)]
    x_3sig = x[(x >= -3) & (x <= 3)]

    plt.fill_between(x_1sig, 0, norm.pdf(x_1sig, mu, sigma),
                     alpha=0.3, color='green', label='68%')
    plt.fill_between(x_2sig, 0, norm.pdf(x_2sig, mu, sigma),
                     alpha=0.3, color='blue', label='95%')
    plt.fill_between(x_3sig, 0, norm.pdf(x_3sig, mu, sigma),
                     alpha=0.3, color='red', label='99.7%')

    plt.axvline(mu, color='black', linestyle='--', alpha=0.5)
    plt.xlabel('x')
    plt.ylabel('概率密度')
    plt.title('正态分布：68-95-99.7规则')
    plt.legend()
    plt.grid(True, alpha=0.3)

    # 右图：不同标准差的对比
    plt.subplot(1, 2, 2)
    for sigma in [0.5, 1, 2]:
        y = norm.pdf(x, 0, sigma)
        plt.plot(x, y, label=f'σ={sigma}')

    plt.xlabel('x')
    plt.ylabel('概率密度')
    plt.title('不同标准差的正态分布')
    plt.legend()
    plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.show()

    print("正态分布特点：")
    print("- 均值μ决定中心位置")
    print("- 标准差σ决定分布的宽度（σ越大越扁平）")
    print("- 68%的数据在[μ-σ, μ+σ]范围内")
    print("- 95%的数据在[μ-2σ, μ+2σ]范围内")

    # 计算机视觉中的应用
    print("\n在计算机视觉中的应用：")
    print("1. 神经网络权重初始化：用正态分布随机初始化权重")
    print("2. 图像噪声建模：高斯噪声服从正态分布")
    print("3. 数据增强：随机调整亮度、对比度")

normal_distribution_demo()
```

#### Softmax函数（多分类概率）

```python
def softmax_demo():
    """Softmax函数详解"""

    def softmax(logits):
        """将logits转换为概率分布"""
        # 减去最大值提高数值稳定性（防止溢出）
        exp_logits = np.exp(logits - np.max(logits))
        return exp_logits / np.sum(exp_logits)

    # 示例：3类分类问题
    logits = np.array([2.0, 1.0, 0.1])
    probabilities = softmax(logits)

    print("Softmax函数演示：")
    print(f"原始logits: {logits}")
    print(f"Softmax概率: {probabilities}")
    print(f"概率和: {np.sum(probabilities):.6f} (必须等于1)")

    # 可视化
    fig, axes = plt.subplots(1, 2, figsize=(12, 4))

    # Logits
    axes[0].bar(['类0', '类1', '类2'], logits, color='skyblue')
    axes[0].set_title('原始Logits')
    axes[0].set_ylabel('值')
    axes[0].grid(True, alpha=0.3, axis='y')

    # Softmax概率
    axes[1].bar(['类0', '类1', '类2'], probabilities, color='lightcoral')
    axes[1].set_title('Softmax概率')
    axes[1].set_ylabel('概率')
    axes[1].set_ylim(0, 1)
    axes[1].grid(True, alpha=0.3, axis='y')

    plt.tight_layout()
    plt.show()

    print("\nSoftmax的特点：")
    print("1. 将任意实数转换为0-1之间的概率")
    print("2. 所有概率之和等于1")
    print("3. 数值越大，对应的概率越大")
    print("4. 是神经网络分类器的最后一层")
```
