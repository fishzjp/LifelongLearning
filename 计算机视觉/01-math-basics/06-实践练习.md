## 练习2：梯度计算

```python
"""
练习2：梯度计算

目标：掌握计算函数梯度的方法

任务：
1. 定义函数 f(x, y) = x² + 2y² + xy
2. 计算该函数的解析梯度 ∇f = [∂f/∂x, ∂f/∂y]
3. 在点(1, 2)处计算梯度
4. 使用数值方法验证解析梯度的正确性

提示：
- 解析梯度：∂f/∂x = 2x + y, ∂f/∂y = 4y + x
- 数值梯度：(f(x+h, y) - f(x, y)) / h
"""

def gradient_practice():
    # TODO: 你的代码在这里

    # 1. 定义函数
    def f(x, y):
        return x**2 + 2*y**2 + x*y

    # 2. 解析梯度
    def gradient(x, y):
        df_dx = 2*x + y
        df_dy = 4*y + x
        return np.array([df_dx, df_dy])

    # 3. 在点(1, 2)计算
    point = (1, 2)
    grad = gradient(point[0], point[1])

    print(f"函数：f(x, y) = x² + 2y² + xy")
    print(f"在点{point}处的解析梯度：{grad}")

    # 4. 数值验证
    h = 1e-5
    f_point = f(point[0], point[1])

    # x方向数值梯度
    df_dx_numerical = (f(point[0] + h, point[1]) - f_point) / h

    # y方向数值梯度
    df_dy_numerical = (f(point[0], point[1] + h) - f_point) / h

    numerical_grad = np.array([df_dx_numerical, df_dy_numerical])

    print(f"数值梯度：{numerical_grad}")
    print(f"误差：{np.abs(grad - numerical_grad)}")
    print(f"验证通过：{np.allclose(grad, numerical_grad)}")

    # 可视化
    x = np.linspace(-2, 4, 20)
    y = np.linspace(-2, 4, 20)
    X, Y = np.meshgrid(x, y)
    Z = f(X, Y)

    plt.figure(figsize=(8, 6))
    plt.contour(X, Y, Z, levels=20, cmap='viridis')

    # 在点(1,2)处画梯度箭头
    grad_x, grad_y = grad
    plt.quiver(1, 2, grad_x, grad_y, color='red', scale=20, width=0.005)
    plt.plot(1, 2, 'ro', markersize=10, label='点(1,2)')

    plt.xlabel('x')
    plt.ylabel('y')
    plt.title('函数等高线图和梯度')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

# 运行练习

gradient_practice()
```

## 练习3：贝叶斯分类器（基础版）

```python
"""
练习3：贝叶斯分类器 - 基础版

目标：掌握贝叶斯定理的基本计算流程

任务：
假设我们有一个简单的图像分类问题：
- 两类：猫和狗
- 特征：图像是否包含"竖直耳朵"（是/否）

已知信息：
- 训练集中：猫占40%，狗占60%
- 猫中70%有竖直耳朵
- 狗中30%有竖直耳朵

问题：
1. 计算P(猫|竖直耳朵)和P(狗|竖直耳朵)
2. 如果检测到竖直耳朵，应该分类为猫还是狗？
3. 计算分类的置信度
4. 如果检测到『没有竖直耳朵』，分类结果是什么？
"""

def bayes_practice_basic():
    # TODO: 你的代码在这里

    # 先验概率
    P_cat = 0.4
    P_dog = 0.6

    # 似然
    P_upright_given_cat = 0.7
    P_upright_given_dog = 0.3

    # 问题1-3：检测到竖直耳朵的情况
    print("="*50)
    print("情况1：检测到竖直耳朵")
    print("="*50)

    # 计算证据
    P_upright = (P_upright_given_cat * P_cat +
                 P_upright_given_dog * P_dog)

    # 后验概率
    P_cat_given_upright = (P_upright_given_cat * P_cat) / P_upright
    P_dog_given_upright = (P_upright_given_dog * P_dog) / P_upright

    print(f"先验：P(猫)={P_cat:.2f}, P(狗)={P_dog:.2f}")
    print(f"似然：P(竖直|猫)={P_upright_given_cat:.2f}, P(竖直|狗)={P_upright_given_dog:.2f}")
    print(f"证据：P(竖直)={P_upright:.3f}")
    print(f"后验：P(猫|竖直)={P_cat_given_upright:.3f}, P(狗|竖直)={P_dog_given_upright:.3f}")

    decision1 = "猫" if P_cat_given_upright > P_dog_given_upright else "狗"
    confidence1 = max(P_cat_given_upright, P_dog_given_upright)
    print(f"决策：{decision1}，置信度：{confidence1:.1%}")

    # 问题4：检测到『没有竖直耳朵』的情况
    print("\n" + "="*50)
    print("情况2：检测到『没有竖直耳朵』")
    print("="*50)

    # 似然（没有竖直耳朵）
    P_not_upright_given_cat = 1 - P_upright_given_cat
    P_not_upright_given_dog = 1 - P_upright_given_dog

    # 证据（没有竖直耳朵）
    P_not_upright = (P_not_upright_given_cat * P_cat +
                     P_not_upright_given_dog * P_dog)

    # 后验概率
    P_cat_given_not_upright = (P_not_upright_given_cat * P_cat) / P_not_upright
    P_dog_given_not_upright = (P_not_upright_given_dog * P_dog) / P_not_upright

    print(f"似然：P(非竖直|猫)={P_not_upright_given_cat:.2f}, P(非竖直|狗)={P_not_upright_given_dog:.2f}")
    print(f"证据：P(非竖直)={P_not_upright:.3f}")
    print(f"后验：P(猫|非竖直)={P_cat_given_not_upright:.3f}, P(狗|非竖直)={P_dog_given_not_upright:.3f}")

    decision2 = "猫" if P_cat_given_not_upright > P_dog_given_not_upright else "狗"
    confidence2 = max(P_cat_given_not_upright, P_dog_given_not_upright)
    print(f"决策：{decision2}，置信度：{confidence2:.1%}")

    # 可视化对比
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

    # 情况1
    classes = ['猫', '狗']
    probs1 = [P_cat_given_upright, P_dog_given_upright]
    bars1 = ax1.bar(classes, probs1, color=['orange', 'blue'], alpha=0.7)
    ax1.set_title('检测到竖直耳朵')
    ax1.set_ylabel('概率')
    ax1.set_ylim(0, 1)
    ax1.grid(True, alpha=0.3, axis='y')

    for bar, prob in zip(bars1, probs1):
        height = bar.get_height()
        ax1.text(bar.get_x() + bar.get_width()/2., height + 0.02,
                f'{prob:.2f}', ha='center', va='bottom')

    # 情况2
    probs2 = [P_cat_given_not_upright, P_dog_given_not_upright]
    bars2 = ax2.bar(classes, probs2, color=['orange', 'blue'], alpha=0.7)
    ax2.set_title('检测到『没有竖直耳朵』')
    ax2.set_ylabel('概率')
    ax2.set_ylim(0, 1)
    ax2.grid(True, alpha=0.3, axis='y')

    for bar, prob in zip(bars2, probs2):
        height = bar.get_height()
        ax2.text(bar.get_x() + bar.get_width()/2., height + 0.02,

bayes_practice_basic()
```

## 练习3扩展：多特征贝叶斯分类器

```python
"""
练习3扩展：多特征贝叶斯分类器

目标：理解朴素贝叶斯的多特征扩展

任务：
现在我们有两个特征：
- 特征1：竖直耳朵（是/否）
- 特征2：瞳孔圆形（是/否）

已知信息：
- 先验：P(猫)=0.4, P(狗)=0.6
- 特征1的似然：P(竖直|猫)=0.7, P(竖直|狗)=0.3
- 特征2的似然：P(圆形|猫)=0.8, P(圆形|狗)=0.4

问题：
1. 假设两个特征独立，计算P(猫|竖直,圆形)和P(狗|竖直,圆形)
2. 比较单特征 vs 多特征的分类效果
3. 为什么需要"朴素"假设（特征独立）？
"""

def bayes_practice_multi_feature():
    # TODO: 你的代码在这里

    # 先验概率
    P_cat = 0.4
    P_dog = 0.6

    # 特征1：竖直耳朵
    P_upright_given_cat = 0.7
    P_upright_given_dog = 0.3

    # 特征2：瞳孔圆形
    P_round_given_cat = 0.8
    P_round_given_dog = 0.4

    print("="*60)
    print("多特征贝叶斯分类器")
    print("="*60)

    # 观测：两个特征都为真
    print("\n观测：竖直耳朵 + 瞳孔圆形")

    # 朴素贝叶斯：假设特征独立
    # P(猫|竖直,圆形) ∝ P(竖直|猫) × P(圆形|猫) × P(猫)

    # 计算未归一化的后验
    cat_score = P_upright_given_cat * P_round_given_cat * P_cat
    dog_score = P_upright_given_dog * P_round_given_dog * P_dog

    # 归一化
    total = cat_score + dog_score
    P_cat_given_features = cat_score / total
    P_dog_given_features = dog_score / total

    print(f"\n【朴素贝叶斯计算】")
    print(f"  P(猫|特征) ∝ {P_upright_given_cat:.2f} × {P_round_given_cat:.2f} × {P_cat:.2f} = {cat_score:.4f}")
    print(f"  P(狗|特征) ∝ {P_upright_given_dog:.2f} × {P_round_given_dog:.2f} × {P_dog:.2f} = {dog_score:.4f}")
    print(f"\n归一化后：")
    print(f"  P(猫|特征) = {P_cat_given_features:.3f}")
    print(f"  P(狗|特征) = {P_dog_given_features:.3f}")

    # 对比单特征
    print(f"\n【对比：单特征 vs 多特征】")

    # 单特征1：仅竖直耳朵
    P_upright = (P_upright_given_cat * P_cat + P_upright_given_dog * P_dog)
    P_cat_given_upright = (P_upright_given_cat * P_cat) / P_upright

    # 单特征2：仅瞳孔圆形
    P_round = (P_round_given_cat * P_cat + P_round_given_dog * P_dog)
    P_cat_given_round = (P_round_given_cat * P_cat) / P_round

    print(f"  仅竖直耳朵：P(猫) = {P_cat_given_upright:.3f}")
    print(f"  仅瞳孔圆形：P(猫) = {P_cat_given_round:.3f}")
    print(f"  两个特征：P(猫) = {P_cat_given_features:.3f}")

    print(f"\n【结论】")
    print(f"  多特征提供了更多信息，分类更自信！")
    print(f"  朴素假设简化计算，但现实中特征可能相关")

    # 可视化
    fig, ax = plt.subplots(figsize=(10, 6))

    methods = ['仅竖直耳朵', '仅瞳孔圆形', '两个特征']
    cat_probs = [P_cat_given_upright, P_cat_given_round, P_cat_given_features]
    dog_probs = [1-p for p in cat_probs]

    x = np.arange(len(methods))
    width = 0.35

    ax.bar(x - width/2, cat_probs, width, label='猫', color='orange', alpha=0.8)
    ax.bar(x + width/2, dog_probs, width, label='狗', color='blue', alpha=0.8)

    ax.set_xlabel('特征组合')
    ax.set_ylabel('分类概率')
    ax.set_title('不同特征组合的分类效果对比')
    ax.set_xticks(x)
    ax.set_xticklabels(methods)
    ax.legend()
    ax.grid(True, alpha=0.3, axis='y')
    ax.set_ylim(0, 1)

    # 标注数值
    for i, (cat_p, dog_p) in enumerate(zip(cat_probs, dog_probs)):
        ax.text(i - width/2, cat_p + 0.02, f'{cat_p:.2f}', ha='center', va='bottom')
        ax.text(i + width/2, dog_p + 0.02, f'{dog_p:.2f}', ha='center', va='bottom')

    plt.tight_layout()
    plt.show()

bayes_practice_multi_feature()
```

## 练习4：Softmax实现

```python
"""
练习4：Softmax实现

目标：手动实现Softmax函数

任务：
1. 实现Softmax函数
2. 测试不同输入，观察输出
3. 验证输出的概率和为1
4. 探索温度参数对分布的影响

提示：
softmax(x) = exp(x) / sum(exp(x))
温度参数：softmax(x/T) = exp(x/T) / sum(exp(x/T))
"""

def softmax_practice():
    # TODO: 你的代码在这里

    def softmax(x, temperature=1.0):
        """Softmax函数，支持温度参数"""
        x = x / temperature  # 温度调整
        exp_x = np.exp(x - np.max(x))  # 数值稳定性
        return exp_x / np.sum(exp_x)

    # 测试数据
    logits = np.array([2.0, 1.0, 0.1, 3.0])

    print("Softmax练习：")
    print(f"原始logits: {logits}")

    # 不同温度下的结果
    temperatures = [0.5, 1.0, 2.0, 5.0]

    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()

    for i, T in enumerate(temperatures):
        probs = softmax(logits, T)

        # 绘制
        axes[i].bar(range(len(logits)), probs, color='lightblue')
        axes[i].set_title(f'温度 T={T}')
        axes[i].set_xlabel('类别')
        axes[i].set_ylabel('概率')
        axes[i].set_ylim(0, 1)
        axes[i].set_xticks(range(len(logits)))

        # 标注数值
        for j, prob in enumerate(probs):
            axes[i].text(j, prob + 0.02, f'{prob:.3f}',
                        ha='center', va='bottom', fontsize=9)

        print(f"\n温度 {T}: {probs}")
        print(f"概率和: {np.sum(probs):.6f}")

    plt.tight_layout()
    plt.show()

    print("\n温度参数的作用：")
    print("- T < 1: 分布更集中（更自信）")
    print("- T = 1: 标准Softmax")
    print("- T > 1: 分布更均匀（更不确定）")

softmax_practice()
```

## 练习6：梯度下降可视化器

```python
"""
练习6：梯度下降可视化器

目标：直观理解梯度下降的收敛过程

任务：
1. 实现2D函数的梯度下降
2. 可视化下降路径
3. 尝试不同的学习率和初始点
4. 观察局部最优和鞍点

函数建议：
- f(x, y) = x² + y² （凸函数，全局最优）
- f(x, y) = x² + y² + 10sin(x) （有局部最优）
- f(x, y) = x² - y² （鞍点）
"""

def gradient_descent_visualizer():
    # TODO: 你的代码在这里
    
    import numpy as np
    import matplotlib.pyplot as plt
    
    # 定义函数
    def f(x, y):
        return x**2 + y**2  # 简单的凸函数
    
    def grad_f(x, y):
        return np.array([2*x, 2*y])
    
    # 梯度下降
    def gd(start, lr=0.1, iterations=20):
        point = np.array(start, dtype=float)
        trajectory = [point.copy()]
        
        for _ in range(iterations):
            grad = grad_f(point[0], point[1])
            point -= lr * grad
            trajectory.append(point.copy())
        
        return np.array(trajectory)
    
    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    
    # 不同初始点
    starts = [(5, 5), (-3, 4), (0.5, 0.5), (2, -2)]
    lrs = [0.1, 0.3, 0.5, 0.8]
    
    for idx, (start, lr) in enumerate(zip(starts, lrs)):
        ax = axes[idx//2, idx%2]
        
        # 生成等高线
        x = np.linspace(-6, 6, 100)
        y = np.linspace(-6, 6, 100)
        X, Y = np.meshgrid(x, y)
        Z = f(X, Y)
        
        ax.contour(X, Y, Z, levels=20, cmap='viridis')
        
        # 梯度下降路径
        trajectory = gd(start, lr, 15)
        ax.plot(trajectory[:, 0], trajectory[:, 1], 'ro-', linewidth=2, markersize=5)
        
        # 标记起点和终点
        ax.plot(start[0], start[1], 'go', markersize=10, label='起点')
        ax.plot(trajectory[-1, 0], trajectory[-1, 1], 'b*', markersize=15, label='终点')
        
        ax.set_title(f'起点{start}, 学习率{lr}')
        ax.legend()
        ax.grid(True, alpha=0.3)
        ax.set_aspect('equal')
    
    plt.tight_layout()
    plt.show()
    
    print("观察要点：")
    print("1. 学习率太小：收敛慢")
    print("2. 学习率太大：可能震荡或发散")
    print("3. 初始点影响收敛路径")
    print("4. 凸函数保证全局最优")
```

## 练习7：贝叶斯分类器完整实现

```python
"""
练习7：贝叶斯分类器完整实现

目标：实现一个完整的朴素贝叶斯图像分类器

任务：
1. 生成模拟的图像特征数据
2. 计算各类别的先验概率
3. 计算各特征的似然概率
4. 实现预测函数
5. 评估分类准确率

数据结构：
- 图像特征：[亮度, 饱和度, 边缘强度]
- 类别：猫、狗、鸟
"""

def bayes_classifier_full():
    # TODO: 你的代码在这里
    
    import numpy as np
    from scipy.stats import norm
    
    # 生成模拟数据
    np.random.seed(42)
    
    # 猫：高亮度，中等饱和度，弱边缘
    cat_features = np.random.multivariate_normal(
        mean=[200, 120, 50],
        cov=[[100, 0, 0], [0, 50, 0], [0, 0, 20]],
        size=100
    )
    
    # 狗：中等亮度，高饱和度，强边缘
    dog_features = np.random.multivariate_normal(
        mean=[150, 180, 120],
        cov=[[80, 0, 0], [0, 60, 0], [0, 0, 40]],
        size=100
    )
    
    # 鸟：高亮度，高饱和度，中等边缘
    bird_features = np.random.multivariate_normal(
        mean=[180, 160, 80],
        cov=[[90, 0, 0], [0, 70, 0], [0, 0, 30]],
        size=100
    )
    
    # 先验概率
    total = len(cat_features) + len(dog_features) + len(bird_features)
    P_cat = len(cat_features) / total
    P_dog = len(dog_features) / total
    P_bird = len(bird_features) / total
    
    # 计算各特征的分布参数（假设正态分布）
    def get_params(data):
        return {
            'mean': np.mean(data, axis=0),
            'std': np.std(data, axis=0)
        }
    
    cat_params = get_params(cat_features)
    dog_params = get_params(dog_features)
    bird_params = get_params(bird_features)
    
    # 预测函数
    def predict(features):
        """
        features: [亮度, 饱和度, 边缘强度]
        """
        # 计算似然
        def likelihood(data_params, x):
            # P(x|class) = 乘积(各特征的概率密度)
            prob = 1.0
            for i in range(3):
                prob *= norm.pdf(x[i], 
                               loc=data_params['mean'][i], 
                               scale=data_params['std'][i])
            return prob
        
        # 计算未归一化的后验
        cat_score = likelihood(cat_params, features) * P_cat
        dog_score = likelihood(dog_params, features) * P_dog
        bird_score = likelihood(bird_params, features) * P_bird
        
        # 归一化
        total_score = cat_score + dog_score + bird_score
        cat_prob = cat_score / total_score
        dog_prob = dog_score / total_score
        bird_prob = bird_score / total_score
        
        # 决策
        classes = ['猫', '狗', '鸟']
        probs = [cat_prob, dog_prob, bird_prob]
        pred_idx = np.argmax(probs)
        
        return classes[pred_idx], probs, classes[pred_idx]
    
    # 测试
    test_samples = [
        ([195, 115, 45], '猫'),
        ([145, 175, 115], '狗'),
        ([175, 155, 75], '鸟'),
    ]
    
    print("="*60)
    print("朴素贝叶斯分类器测试")
    print("="*60)
    
    correct = 0
    for features, true_label in test_samples:
        pred_label, probs, _ = predict(features)
        is_correct = (pred_label == true_label)
        correct += is_correct
        
        print(f"\n特征：{features}")
        print(f"真实类别：{true_label}")
        print(f"预测类别：{pred_label} {'✓' if is_correct else '✗'}")
        print(f"概率分布：")
        print(f"  猫：{probs[0]:.3f}")
        print(f"  狗：{probs[1]:.3f}")
        print(f"  鸟：{probs[2]:.3f}")
    
    print(f"\n准确率：{correct}/{len(test_samples)} ({correct/len(test_samples)*100:.1f}%)")
    
    # 可视化决策边界
    fig = plt.figure(figsize=(14, 5))
    
    # 选择两个特征进行可视化
    ax1 = fig.add_subplot(131)
    ax1.scatter(cat_features[:, 0], cat_features[:, 1], alpha=0.5, label='猫', c='orange')
    ax1.scatter(dog_features[:, 0], dog_features[:, 1], alpha=0.5, label='狗', c='blue')
    ax1.scatter(bird_features[:, 0], bird_features[:, 1], alpha=0.5, label='鸟', c='green')
    ax1.set_xlabel('亮度')
    ax1.set_ylabel('饱和度')
    ax1.legend()
    ax1.set_title('特征分布：亮度 vs 饱和度')
    ax1.grid(True, alpha=0.3)
    
    ax2 = fig.add_subplot(132)
    ax2.scatter(cat_features[:, 1], cat_features[:, 2], alpha=0.5, label='猫', c='orange')
    ax2.scatter(dog_features[:, 1], dog_features[:, 2], alpha=0.5, label='狗', c='blue')
    ax2.scatter(bird_features[:, 1], bird_features[:, 2], alpha=0.5, label='鸟', c='green')
    ax2.set_xlabel('饱和度')
    ax2.set_ylabel('边缘强度')
    ax2.legend()
    ax2.set_title('特征分布：饱和度 vs 边缘强度')
    ax2.grid(True, alpha=0.3)
    
    ax3 = fig.add_subplot(133)
    ax3.scatter(cat_features[:, 0], cat_features[:, 2], alpha=0.5, label='猫', c='orange')
    ax3.scatter(dog_features[:, 0], dog_features[:, 2], alpha=0.5, label='狗', c='blue')
```

## 练习8：数学概念综合应用

```python
"""
练习8：数学概念综合应用

目标：将所有数学概念结合，实现一个简单的图像处理流水线

任务：
1. 加载/创建图像
2. 使用矩阵运算进行变换
3. 计算梯度进行边缘检测
4. 使用概率进行噪声去除
5. 可视化每一步的结果

完整流程：
原始图像 → 几何变换 → 添加噪声 → 梯度检测 → 贝叶斯去噪 → 最终结果
"""

def math_pipeline_demo():
    # TODO: 你的代码在这里
    
    import numpy as np
    import matplotlib.pyplot as plt
    from scipy.ndimage import gaussian_filter
    
    # 1. 创建测试图像
    img = np.zeros((50, 50))
    img[15:35, 15:35] = 1  # 白色方块
    
    # 2. 几何变换（旋转）
    theta = np.pi / 6  # 30度
    center = 25
    
    # 旋转矩阵
    R = np.array([
        [np.cos(theta), -np.sin(theta)],
        [np.sin(theta), np.cos(theta)]
    ])
    
    # 创建旋转后的图像
    img_rot = np.zeros_like(img)
    for i in range(img.shape[0]):
        for j in range(img.shape[1]):
            # 相对中心的坐标
            x, y = j - center, i - center
            # 旋转
            x_new, y_new = R @ np.array([x, y])
            # 映射回原坐标
            j_new, i_new = int(x_new + center), int(y_new + center)
            if 0 <= i_new < img.shape[0] and 0 <= j_new < img.shape[1]:
                img_rot[i_new, j_new] = img[i, j]
    
    # 插值平滑
    img_rot = gaussian_filter(img_rot, sigma=1)
    
    # 3. 添加高斯噪声
    noise = np.random.normal(0, 0.2, img.shape)
    img_noisy = np.clip(img_rot + noise, 0, 1)
    
    # 4. 梯度计算（Sobel算子）
    sobel_x = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]])
    sobel_y = np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]])
    
    from scipy.signal import convolve2d
    grad_x = convolve2d(img_noisy, sobel_x, mode='same')
    grad_y = convolve2d(img_noisy, sobel_y, mode='same')
    gradient_magnitude = np.sqrt(grad_x**2 + grad_y**2)
    
    # 5. 贝叶斯去噪
    # 先验：自然图像平滑（使用高斯滤波）
    prior = gaussian_filter(img_noisy, sigma=1.5)
    
    # 贝叶斯融合（简单加权）
    # 后验 = (观测/噪声² + 先验/平滑²) / (1/噪声² + 1/平滑²)
    noise_var = 0.04  # 噪声方差
    smooth_var = 0.1  # 平滑先验方差
    
    weight_obs = 1 / noise_var
    weight_prior = 1 / smooth_var
    
    img_denoised = (img_noisy * weight_obs + prior * weight_prior) / (weight_obs + weight_prior)
    
    # 可视化
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))
    
    images = [
        img, img_rot, img_noisy, 
        gradient_magnitude, img_denoised, img
    ]
    titles = [
        '1. 原始图像', '2. 旋转+平滑', '3. 添加噪声',
        '4. 梯度检测', '5. 贝叶斯去噪', '6. 对比（原始）'
    ]
    
    for ax, img_i, title in zip(axes.flat, images, titles):
        if '梯度' in title:
            im = ax.imshow(img_i, cmap='hot')
        else:
            im = ax.imshow(img_i, cmap='gray', vmin=0, vmax=1)
        ax.set_title(title)
        ax.axis('off')
        plt.colorbar(im, ax=ax, fraction=0.046, pad=0.04)
    
    plt.tight_layout()
    plt.show()
    
    # 计算质量指标
    mse_noisy = np.mean((img_rot - img_noisy)**2)
    mse_denoised = np.mean((img_rot - img_denoised)**2)
    
    print("图像处理流水线完成！")
    print(f"噪声图像MSE：{mse_noisy:.4f}")
    print(f"去噪后MSE：{mse_denoised:.4f}")
    print(f"改善程度：{(mse_noisy - mse_denoised) / mse_noisy * 100:.1f}%")

# 运行练习

math_pipeline_demo()
```
