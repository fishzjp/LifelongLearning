## 4.1 图像滤波与卷积（线性代数应用）
### 什么是卷积？
卷积是计算机视觉中最核心的运算，用于图像滤波、特征提取。

```python

# 依赖: matplotlib, numpy
# 安装: pip install matplotlib numpy
import numpy as np
import matplotlib.pyplot as plt
from scipy.signal import convolve2d

def convolution_demo():
    """卷积运算可视化"""
    
    # 简单的5x5图像
    image = np.array([
        [10, 20, 30, 40, 50],
        [15, 25, 35, 45, 55],
        [20, 30, 40, 50, 60],
        [25, 35, 45, 55, 65],
        [30, 40, 50, 60, 70]
    ], dtype=np.float32)
    
    # 边缘检测卷积核
    kernel = np.array([
        [-1, 0, 1],
        [-2, 0, 2],
        [-1, 0, 1]
    ], dtype=np.float32)
    
    print("原始图像：")
    print(image.astype(int))
    print("\n卷积核（Sobel边缘检测）：")
    print(kernel)
    
    # 手动卷积计算（中心点）
    h, w = image.shape
    kh, kw = kernel.shape
    
    # 在图像中心位置计算卷积
    center_h, center_w = 2, 2
    result = 0
    
    print(f"\n在位置({center_h}, {center_w})的卷积计算：")
    for i in range(kh):
        for j in range(kw):
            img_val = image[center_h - 1 + i, center_w - 1 + j]
            kernel_val = kernel[i, j]
            product = img_val * kernel_val
            print(f"  image[{center_h-1+i},{center_w-1+j}] × kernel[{i},{j}] = {img_val} × {kernel_val} = {product}")
            result += product
    
    print(f"卷积结果：{result}")
    
    # 使用scipy进行完整卷积
    conv_result = convolve2d(image, kernel, mode='valid')
    
    print(f"\n完整卷积结果（valid模式）：")
    print(conv_result)
    
    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))
    
    axes[0].imshow(image, cmap='viridis')
    axes[0].set_title('原始图像')
    axes[0].axis('off')
    
    axes[1].imshow(kernel, cmap='coolwarm')
    axes[1].set_title('卷积核')
    axes[1].axis('off')
    
    axes[2].imshow(conv_result, cmap='viridis')
    axes[2].set_title('卷积结果（边缘检测）')
    axes[2].axis('off')
    
    plt.tight_layout()
    plt.show()

convolution_demo()
```

**实际应用：**
- **模糊**：均值滤波、高斯滤波
- **锐化**：拉普拉斯算子
- **边缘检测**：Sobel、Prewitt、Canny
- **特征提取**：各种CNN卷积核

## 4.2 图像变换（矩阵运算应用）
### 几何变换
```python
def image_transformations():
    """图像几何变换"""
    
    # 原始图像点
    point = np.array([1, 1, 1])  # 齐次坐标 [x, y, 1]
    
    print("原始点：", point[:2])
    
    # 1. 平移
    T_translate = np.array([
        [1, 0, 2],  # x方向平移2
        [0, 1, 1],  # y方向平移1
        [0, 0, 1]
    ])
    translated = T_translate @ point
    print(f"平移后：{translated[:2]}")
    
    # 2. 旋转
    theta = np.pi / 4  # 45度
    T_rotate = np.array([
        [np.cos(theta), -np.sin(theta), 0],
        [np.sin(theta), np.cos(theta), 0],
        [0, 0, 1]
    ])
    rotated = T_rotate @ point
    print(f"旋转45°后：{rotated[:2]}")
    
    # 3. 缩放
    T_scale = np.array([
        [2, 0, 0],  # x方向放大2倍
        [0, 0.5, 0],  # y方向缩小0.5倍
        [0, 0, 1]
    ])
    scaled = T_scale @ point
    print(f"缩放后：{scaled[:2]}")
    
    # 4. 组合变换
    T_combined = T_translate @ T_rotate @ T_scale
    combined = T_combined @ point
    print(f"组合变换后：{combined[:2]}")
    
    # 可视化
    fig, ax = plt.subplots(figsize=(8, 8))
    
    points = {
        '原始': point[:2],
        '平移': translated[:2],
        '旋转': rotated[:2],
        '缩放': scaled[:2],
        '组合': combined[:2]
    }
    
    colors = ['black', 'red', 'blue', 'green', 'purple']
    
    for (name, p), color in zip(points.items(), colors):
        ax.plot(p[0], p[1], 'o', markersize=10, label=name, color=color)
    
    ax.axhline(y=0, color='gray', linestyle='--', alpha=0.3)
    ax.axvline(x=0, color='gray', linestyle='--', alpha=0.3)
    ax.grid(True, alpha=0.3)
    ax.legend()
    ax.set_xlabel('x')
    ax.set_ylabel('y')
    ax.set_title('图像几何变换')
    ax.set_aspect('equal')
    
    plt.show()

image_transformations()
```

## 4.3 梯度下降优化（微积分应用）
### 训练神经网络
```python
def gradient_descent_cv():
    """梯度下降在CV中的应用：拟合图像亮度曲线"""
    
    # 模拟图像亮度数据
    np.random.seed(42)
    x = np.linspace(0, 10, 20)
    # 真实关系：y = 2x + 3 + 噪声
    y = 2 * x + 3 + np.random.normal(0, 1, 20)
    
    print("任务：用线性模型 y = wx + b 拟合图像亮度数据")
    print(f"数据点：{len(x)}个")
    
    # 损失函数：MSE
    def loss(w, b):
        y_pred = w * x + b
        return np.mean((y_pred - y) ** 2)
    
    # 梯度
    def gradient(w, b):
        y_pred = w * x + b
        dw = 2 * np.mean((y_pred - y) * x)
        db = 2 * np.mean(y_pred - y)
        return dw, db
    
    # 梯度下降
    w, b = 0.0, 0.0  # 初始参数
    lr = 0.01  # 学习率
    epochs = 100
    
    losses = []
    params_history = []
    
    print(f"\n初始参数：w={w:.2f}, b={b:.2f}, 损失={loss(w, b):.4f}")
    
    for epoch in range(epochs):
        dw, db = gradient(w, b)
        w -= lr * dw
        b -= lr * db
        
        if epoch % 20 == 0:
            losses.append(loss(w, b))
            params_history.append((w, b))
            print(f"Epoch {epoch:3d}: w={w:.3f}, b={b:.3f}, 损失={loss(w, b):.4f}")
    
    print(f"\n最终参数：w={w:.3f}, b={b:.3f}")
    print(f"真实值：w=2.0, b=3.0")
    
    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))
    
    # 左图：拟合结果
    ax1.scatter(x, y, alpha=0.6, label='原始数据')
    
    # 绘制不同阶段的拟合线
    colors = ['red', 'orange', 'green', 'purple']
    for i, (w_i, b_i) in enumerate(params_history):
        x_line = np.linspace(0, 10, 100)
        y_line = w_i * x_line + b_i
        ax1.plot(x_line, y_line, '--', color=colors[i], alpha=0.5, 
                label=f'迭代{i*20}次' if i>0 else '初始')
    
    # 最终拟合线
    x_line = np.linspace(0, 10, 100)
    y_line = w * x_line + b
    ax1.plot(x_line, y_line, 'b-', linewidth=2, label='最终拟合')
    
    ax1.set_xlabel('输入亮度')
    ax1.set_ylabel('输出亮度')
    ax1.set_title('梯度下降：图像亮度校准')
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    
    # 右图：损失下降
    all_losses = [loss(p[0], p[1]) for p in params_history]
    ax2.plot(range(0, epochs, 20), all_losses, 'o-')
    ax2.set_xlabel('迭代次数')
    ax2.set_ylabel('损失 (MSE)')
    ax2.set_title('损失函数下降曲线')
    ax2.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()

gradient_descent_cv()
```

## 4.4 贝叶斯图像分类（概率应用）
```python
def bayes_image_filter():
    """贝叶斯方法：图像去噪"""
    
    print("="*60)
    print("贝叶斯图像去噪：从含噪像素恢复原始值")
    print("="*60)
    
    # 模拟场景：已知自然图像像素值通常在128附近（先验）
    # 观测到含噪像素值（证据）
    
    # 先验：自然图像像素值服从正态分布 N(128, 30²)
    mu_prior = 128
    sigma_prior = 30
    
    # 似然：噪声模型 N(observed, 10²)
    sigma_noise = 10
    
    # 观测值
    observed_values = [150, 90, 200, 100]
    
    print("\n【先验知识】")
    print(f"  自然图像像素分布：N({mu_prior}, {sigma_prior}²)")
    print(f"  即：大部分像素在 [{mu_prior-2*sigma_prior:.0f}, {mu_prior+2*sigma_prior:.0f}] 范围内")
    
    print("\n【观测数据】")
    for i, obs in enumerate(observed_values):
        print(f"  像素{i+1}观测值：{obs}")
    
    # 贝叶斯更新
    print("\n【贝叶斯更新】")
    print("后验均值 = (观测值/σ² + 先验均值/σ_prior²) / (1/σ² + 1/σ_prior²)")
    
    results = []
    for obs in observed_values:
        # 贝叶斯估计（高斯-高斯共轭）
        posterior_sigma = 1 / np.sqrt(1/sigma_noise**2 + 1/sigma_prior**2)
        posterior_mu = (obs/sigma_noise**2 + mu_prior/sigma_prior**2) / (1/sigma_noise**2 + 1/sigma_prior**2)
        
        results.append((obs, posterior_mu, posterior_sigma))
        
        print(f"\n  观测值 {obs}:")
        print(f"    后验均值：{posterior_mu:.1f}")
        print(f"    后验标准差：{posterior_sigma:.1f}")
        print(f"    去噪后估计：{posterior_mu:.0f}")
    
    # 可视化
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    axes = axes.flatten()
    
    x = np.linspace(50, 250, 200)
    
    for idx, (obs, post_mu, post_sigma) in enumerate(results):
        ax = axes[idx]
        
        # 先验分布
        prior_pdf = (1/(sigma_prior*np.sqrt(2*np.pi))) * np.exp(-(x-mu_prior)**2/(2*sigma_prior**2))
        ax.plot(x, prior_pdf, 'b--', label=f'先验 N({mu_prior},{sigma_prior}²)', linewidth=2)
        
        # 似然（观测）
        likelihood_pdf = (1/(sigma_noise*np.sqrt(2*np.pi))) * np.exp(-(x-obs)**2/(2*sigma_noise**2))
        ax.plot(x, likelihood_pdf, 'r--', label=f'似然 N({obs},{sigma_noise}²)', linewidth=2)
        
        # 后验分布
        posterior_pdf = (1/(post_sigma*np.sqrt(2*np.pi))) * np.exp(-(x-post_mu)**2/(2*post_sigma**2))
        ax.plot(x, posterior_pdf, 'g-', label=f'后验 N({post_mu:.1f},{post_sigma**2:.1f})', linewidth=2)
        
        # 标记观测值和估计值
        ax.axvline(obs, color='red', linestyle=':', alpha=0.5, label=f'观测: {obs}')
        ax.axvline(post_mu, color='green', linestyle='-', alpha=0.7, label=f'估计: {post_mu:.0f}')
        
        ax.set_title(f'像素{idx+1}的贝叶斯更新')
        ax.set_xlabel('像素值')
        ax.set_ylabel('概率密度')
        ax.legend(fontsize=8)
        ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    print("\n【结论】")
    print("  贝叶斯方法平衡了观测值和先验知识")
    print("  极端值被拉向典型值，实现去噪效果")

bayes_image_filter()
```
