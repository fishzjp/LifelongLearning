## 1. 测试环境概述

### 1.1 什么是测试环境

测试环境（Test Environment）是为软件测试而配置的硬件、软件、网络和数据的集合，用于模拟生产环境，验证软件功能和性能。

```
测试环境组成：
├── 硬件环境
│   ├── 服务器配置
│   ├── 客户端设备
│   └── 网络设备
├── 软件环境
│   ├── 操作系统
│   ├── 数据库
│   ├── 中间件
│   └── 应用程序
├── 数据环境
│   ├── 测试数据
│   ├── 基础数据
│   └── 清理策略
└── 网络环境
    ├── 网络拓扑
    ├── 防火墙规则
    └── 代理配置
```

```bash
# Linux测试环境初始化脚本

apt-get install -y \
    curl wget git vim htop \
    net-tools telnet traceroute \
    python3 python3-pip \
    docker.io docker-compose

# 3. 配置用户权限
useradd -m -s /bin/bash tester
usermod -aG docker tester

# 4. 配置SSH
sed -i 's/#PasswordAuthentication yes/PasswordAuthentication yes/' /etc/ssh/sshd_config
systemctl restart sshd

# 5. 防火墙配置
ufw allow 22/tcp
ufw allow 80/tcp
ufw allow 443/tcp
ufw --force enable

# 6. 时区配置
timedatectl set-timezone Asia/Shanghai

# 7. 配置资源限制
cat >> /etc/security/limits.conf << EOF
* soft nofile 65536
* hard nofile 65536
* soft nproc 65536
* hard nproc 65536
EOF

echo "基础环境配置完成"
```

### 2.3 环境配置管理

```python
# config_manager.py - 环境配置管理
import os
import yaml
import json
from pathlib import Path

class EnvironmentConfig:
    """环境配置管理器"""

    def __init__(self, env_name):
        self.env_name = env_name
        self.config_dir = Path(__file__).parent / "configs"
        self.config_file = self.config_dir / f"{env_name}.yaml"
        self.load_config()

    def load_config(self):
        """加载配置"""
        if not self.config_file.exists():
            raise FileNotFoundError(f"配置文件不存在: {self.config_file}")

        with open(self.config_file, 'r', encoding='utf-8') as f:
            self.config = yaml.safe_load(f)

    def get_database_config(self):
        """获取数据库配置"""
        db = self.config.get('database', {})
        return {
            'host': db.get('host', 'localhost'),
            'port': db.get('port', 3306),
            'name': db.get('name', 'test_db'),
            'user': db.get('user', 'test_user'),
            'password': db.get('password', 'test_pass'),
            'pool_size': db.get('pool_size', 10)
        }

    def get_application_config(self):
        """获取应用配置"""
        app = self.config.get('application', {})
        return {
            'debug': app.get('debug', True),
            'host': app.get('host', '0.0.0.0'),
            'port': app.get('port', 5000),
            'log_level': app.get('log_level', 'DEBUG'),
            'enable_swagger': app.get('enable_swagger', True)
        }

    def get_service_config(self):
        """获取服务配置"""
        services = self.config.get('services', {})
        return {
            'redis': services.get('redis', {'host': 'localhost', 'port': 6379}),
            'rabbitmq': services.get('rabbitmq', {'host': 'localhost', 'port': 5672}),
            'storage': services.get('storage', {'type': 'local', 'path': '/tmp/storage'})
        }

    def get_test_data_config(self):
        """获取测试数据配置"""
        return self.config.get('test_data', {
            'seed_data': True,
            'cleanup_after_test': True,
            'backup_before_test': False
        })

    def to_env_file(self):
        """生成.env文件"""
        env_content = []

        # 数据库
        db = self.get_database_config()
        env_content.append(f"DB_HOST={db['host']}")
        env_content.append(f"DB_PORT={db['port']}")
        env_content.append(f"DB_NAME={db['name']}")
        env_content.append(f"DB_USER={db['user']}")
        env_content.append(f"DB_PASS={db['password']}")

        # 应用
        app = self.get_application_config()
        env_content.append(f"APP_DEBUG={app['debug']}")
        env_content.append(f"APP_PORT={app['port']}")
        env_content.append(f"LOG_LEVEL={app['log_level']}")

        # 服务
        services = self.get_service_config()
        env_content.append(f"REDIS_HOST={services['redis']['host']}")
        env_content.append(f"REDIS_PORT={services['redis']['port']}")

        env_file = self.config_dir / f".env.{self.env_name}"
        with open(env_file, 'w') as f:
            f.write('\n'.join(env_content))

        return env_file

# 配置文件示例 (configs/test.yaml)
"""
database:
  host: localhost
  port: 3306
  name: test_db
  user: test_user
  password: test_pass
  pool_size: 10

application:
  debug: true
  host: 0.0.0.0
  port: 5000
  log_level: DEBUG
  enable_swagger: true

services:
  redis:
    host: localhost
    port: 6379
  rabbitmq:
    host: localhost
    port: 5672
  storage:
    type: local
    path: /tmp/test_storage

test_data:
  seed_data: true
  cleanup_after_test: true
  backup_before_test: false

environment:
  name: test
  description: "测试环境"
  auto_cleanup: true
  max_lifetime: "24h"
"""

# 使用示例
if __name__ == '__main__':
    # 加载测试环境配置
    config = EnvironmentConfig('test')

    # 获取配置
    db_config = config.get_database_config()
    app_config = config.get_application_config()

    print("数据库配置:", db_config)
    print("应用配置:", app_config)

    # 生成.env文件
    env_file = config.to_env_file()
    print(f"环境文件已生成: {env_file}")
```

### 3.3 环境管理脚本

```bash
#!/bin/bash
# manage_test_env.sh - 测试环境管理脚本

ENV_NAME=${1:-test}
ACTION=${2:-up}

# 颜色定义
GREEN='\033[0;32m'
RED='\033[0;31m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

# 启动环境
start_env() {
    log_info "启动测试环境: $ENV_NAME"

    # 检查Docker是否运行
    if ! docker info > /dev/null 2>&1; then
        log_error "Docker未运行"
        exit 1
    fi

    # 选择配置文件
    COMPOSE_FILE="docker-compose.$ENV_NAME.yml"
    if [ ! -f "$COMPOSE_FILE" ]; then
        COMPOSE_FILE="docker-compose.yml"
    fi

    # 启动服务
    docker-compose -f "$COMPOSE_FILE" up -d

    # 等待健康检查
    log_info "等待服务健康检查..."
    sleep 10

    # 验证服务
    check_health

    log_info "环境启动完成"
    show_status
}

stop_env() {
```

```python
# data_cleanup.py
class DataCleanup:
    """测试数据清理"""

    def __init__(self, db_manager):
        self.db = db_manager

    def cleanup_after_test(self, test_name):
        """测试后清理"""
        print(f"清理测试 {test_name} 的数据")

        # 1. 删除测试产生的数据
        self.delete_test_data(test_name)

        # 2. 重置序列
        self.reset_sequences()

        # 3. 清理缓存
        self.clear_cache()

        # 4. 验证清理结果
        self.verify_cleanup()

    def delete_test_data(self, test_name):
        """删除测试数据"""
        # 根据测试名称删除
        tables = ['users', 'products', 'orders']

        for table in tables:
            self.db.execute(
                f"DELETE FROM {table} WHERE test_run_id = %s",
                (test_name,)
            )

        # 或者删除所有created_at在测试开始后的数据
        import datetime
        test_start = datetime.datetime.now() - datetime.timedelta(hours=1)

        for table in tables:
            self.db.execute(
                f"DELETE FROM {table} WHERE created_at > %s",
                (test_start,)
            )

    def reset_sequences(self):
        """重置序列"""
        sequences = [
            'users_id_seq',
            'products_id_seq',
            'orders_id_seq'
        ]

        for seq in sequences:
            self.db.execute(f"ALTER SEQUENCE {seq} RESTART WITH 1")

    def clear_cache(self):
        """清理缓存"""
        # Redis缓存
        try:
            import redis
            r = redis.Redis(host='localhost', port=6379, db=0)
            r.flushdb()  # 清空当前数据库
            print("Redis缓存已清理")
        except:
            print("Redis清理失败")

    def verify_cleanup(self):
        """验证清理结果"""
        tables = ['users', 'products', 'orders']

        for table in tables:
            result = self.db.execute(f"SELECT COUNT(*) FROM {table}")
            count = result.fetchone()[0]

            if count > 0:
                print(f"警告: {table} 仍有 {count} 条数据")
            else:
                print(f"✓ {table} 已清理")

    def cleanup_all(self):
        """清理所有测试数据"""
        print("清理所有测试数据")

        # 确认
        confirm = input("确认清理所有测试数据? (yes/no): ")
        if confirm.lower() != 'yes':
            print("取消清理")
            return

        # 清理所有表
        tables = ['users', 'products', 'orders', 'reviews', 'logs']
        for table in tables:
            self.db.execute(f"DELETE FROM {table}")
            print(f"已清空 {table}")

        # 重置所有序列
        self.reset_sequences()

        # 清理缓存
        self.clear_cache()

        print("所有测试数据已清理")

# 使用示例
```

```python
# monitor_env.py
import psutil
import requests
import time
from datetime import datetime

class EnvironmentMonitor:
    """环境监控"""

    def __init__(self):
        self.metrics = {}

    def check_system_resources(self):
        """检查系统资源"""
        return {
            'cpu_percent': psutil.cpu_percent(interval=1),
            'memory_percent': psutil.virtual_memory().percent,
            'disk_usage': psutil.disk_usage('/').percent,
            'network_io': psutil.net_io_counters()._asdict()
        }

    def check_service_health(self, services):
        """检查服务健康"""
        results = {}

        for name, url in services.items():
            try:
                start = time.time()
                response = requests.get(url, timeout=5)
                elapsed = time.time() - start

                results[name] = {
                    'status': 'healthy' if response.status_code == 200 else 'unhealthy',
                    'response_time': round(elapsed * 1000, 2),
                    'status_code': response.status_code
                }
            except Exception as e:
                results[name] = {
                    'status': 'error',
                    'error': str(e)
                }

        return results

    def check_database(self, db_config):
        """检查数据库"""
        import psycopg2

        try:
            conn = psycopg2.connect(**db_config)
            cursor = conn.cursor()

            # 检查连接
            cursor.execute("SELECT 1")

            # 检查表数量
            cursor.execute("SELECT COUNT(*) FROM information_schema.tables WHERE table_schema = 'public'")
            table_count = cursor.fetchone()[0]

            # 检查数据量
            cursor.execute("SELECT COUNT(*) FROM users")
            user_count = cursor.fetchone()[0]

            conn.close()

            return {
                'status': 'healthy',
                'table_count': table_count,
                'user_count': user_count
            }
        except Exception as e:
            return {
                'status': 'error',
                'error': str(e)
            }

    def check_disk_space(self, threshold=80):
        """检查磁盘空间"""
        disk = psutil.disk_usage('/')
        percent = disk.percent

        return {
            'used_percent': percent,
            'warning': percent > threshold,
            'free_gb': round(disk.free / (1024**3), 2)
        }

    def check_log_errors(self, log_file, error_keywords=['ERROR', 'FATAL', 'EXCEPTION']):
        """检查日志错误"""
        try:
            with open(log_file, 'r') as f:
                lines = f.readlines()[-100:]  # 最近100行

            errors = []
            for line in lines:
                if any(keyword in line for keyword in error_keywords):
                    errors.append(line.strip())

            return {
                'error_count': len(errors),
                'errors': errors[-5:]  # 最近5个错误
            }
        except Exception as e:
            return {'error': str(e)}

    def generate_report(self):
        """生成监控报告"""
        report = {
            'timestamp': datetime.now().isoformat(),
            'system': self.check_system_resources(),
            'services': self.check_service_health({
                'web': 'http://localhost:5000/health',
                'db': 'http://localhost:5432',
                'redis': 'http://localhost:6379'
            }),
            'disk': self.check_disk_space(),
            'alerts': []
        }

        # 生成告警
        if report['system']['cpu_percent'] > 80:
            report['alerts'].append('CPU使用率过高')

        if report['system']['memory_percent'] > 80:
            report['alerts'].append('内存使用率过高')

        if report['disk']['warning']:
            report['alerts'].append(f"磁盘空间不足: {report['disk']['used_percent']}%")

        for service, info in report['services'].items():
            if info['status'] != 'healthy':
                report['alerts'].append(f"{service} 服务异常")

        return report

    def run_continuous_monitoring(self, interval=60):
        """持续监控"""
        print(f"开始持续监控，间隔: {interval}秒")
        print("按 Ctrl+C 停止")

        try:
            while True:
                report = self.generate_report()

                # 打印报告
                print(f"\n{'='*60}")
                print(f"监控时间: {report['timestamp']}")
                print(f"CPU: {report['system']['cpu_percent']}%")
                print(f"内存: {report['system']['memory_percent']}%")
                print(f"磁盘: {report['disk']['used_percent']}%")

                if report['alerts']:
                    print(f"\n告警:")
                    for alert in report['alerts']:
                        print(f"  ⚠ {alert}")

                time.sleep(interval)

        except KeyboardInterrupt:
            print("\n监控已停止")

# 使用示例
if __name__ == '__main__':
    monitor = EnvironmentMonitor()

    # 单次检查
    report = monitor.generate_report()
    print("当前状态:", report)

    # 持续监控
    monitor.run_continuous_monitoring(interval=30)
```

```bash
# 清理旧日志
cleanup_logs() {

    find /opt/test_env/logs -name "*.log" -mtime +30 -delete
    echo "日志清理完成"
}

# 备份数据
backup_data() {
    BACKUP_DIR="/opt/test_env/backups/$(date +%Y%m%d)"
    mkdir -p $BACKUP_DIR

    echo "备份数据库..."
    docker-compose exec -T db pg_dump -U test testdb > $BACKUP_DIR/db.sql

    echo "备份配置..."
    cp docker-compose.yml $BACKUP_DIR/

    echo "备份完成: $BACKUP_DIR"
}

# 检查磁盘空间
check_disk() {
    USAGE=$(df / | tail -1 | awk '{print $5}' | sed 's/%//')

    if [ $USAGE -gt 80 ]; then
        echo "警告: 磁盘使用率 ${USAGE}%"

        # 自动清理
        echo "尝试清理..."
        docker system prune -f
        docker volume prune -f

        return 1
    fi

    echo "磁盘正常: ${USAGE}%"
    return 0
}

# 重启异常服务
restart_unhealthy() {
    echo "检查服务健康..."

    # 检查Web服务
    if ! curl -f http://localhost:5000/health > /dev/null 2>&1; then
        echo "Web服务异常，重启..."
        docker-compose restart app
    fi

    # 检查数据库
    if ! docker-compose exec -T db pg_isready > /dev/null 2>&1; then
        echo "数据库异常，重启..."
        docker-compose restart db
    fi

    echo "健康检查完成"
}

# 清理旧容器
cleanup_old_containers() {
    echo "清理停止的容器..."
    docker container prune -f

    echo "清理未使用的镜像..."
    docker image prune -f

    echo "清理构建缓存..."
    docker builder prune -f
}

# 主函数
main() {
    case "$1" in
        logs)
            cleanup_logs
            ;;
        backup)
            backup_data
            ;;
        disk)
            check_disk
            ;;
        restart)
            restart_unhealthy
            ;;
        cleanup)
            cleanup_old_containers
            ;;
        all)
            cleanup_logs
            check_disk
            restart_unhealthy
            cleanup_old_containers
            ;;
        *)
            echo "用法: $0 {logs|backup|disk|restart|cleanup|all}"
            exit 1
            ;;
    esac
}

main "$@"
```

```bash
# 启动环境
start_environment() {
    echo "启动测试环境..."

    docker-compose -f docker-compose.test.yml up -d

    echo "等待服务启动..."
    sleep 15

    # 健康检查
    if curl -f http://localhost:5000/health > /dev/null 2>&1; then
        echo -e "${GREEN}✓ 环境启动成功${NC}"

        echo "服务列表:"
        echo "  - Web应用: http://localhost:8080"
        echo "  - API服务: http://localhost:5000"
        echo "  - Swagger: http://localhost:5000/swagger"
        echo "  - Grafana: http://localhost:3000 (admin/admin)"
        echo "  - Kibana: http://localhost:5601"
        echo "  - RabbitMQ: http://localhost:15672 (test/test)"
        echo ""
        echo "运行测试: docker-compose -f docker-compose.test.yml run test-runner"
    else
        echo -e "${RED}✗ 环境启动失败${NC}"
        docker-compose -f docker-compose.test.yml logs
        exit 1
    fi
}

# 运行测试
run_tests() {
    echo "运行测试..."

    docker-compose -f docker-compose.test.yml run test-runner

    echo -e "${GREEN}✓ 测试完成${NC}"
    echo "报告位置: ./reports/"
}

# 清理环境
cleanup_environment() {
    echo "清理环境..."

    docker-compose -f docker-compose.test.yml down -v --rmi all

    echo -e "${GREEN}✓ 环境已清理${NC}"
}

# 显示帮助
show_help() {
    echo "用法: $0 [选项]"
    echo ""
    echo "选项:"
    echo "  deploy    - 部署测试环境"
    echo "  start     - 启动环境"
    echo "  stop      - 停止环境"
    echo "  restart   - 重启环境"
    echo "  test      - 运行测试"
    echo "  logs      - 查看日志"
    echo "  clean     - 清理环境"
    echo "  status    - 查看状态"
    echo ""
    echo "示例:"
    echo "  $0 deploy    # 首次部署"
    echo "  $0 test      # 运行测试"
    echo "  $0 clean     # 清理环境"
}

# 主逻辑
case "$1" in
    deploy)
        check_dependencies
        prepare_directories
        generate_configs
        generate_app_code
        generate_monitoring
        start_environment
        ;;
    start)
        start_environment
        ;;
    stop)
        docker-compose -f docker-compose.test.yml down
        ;;
    restart)
        docker-compose -f docker-compose.test.yml restart
        ;;
    test)
        run_tests
        ;;
    logs)
        docker-compose -f docker-compose.test.yml logs -f
        ;;
    clean)
        cleanup_environment
        ;;
    status)
        docker-compose -f docker-compose.test.yml ps
        ;;
    help|--help|-h)
        show_help
        ;;
    *)
        if [ -z "$1" ]; then
            show_help
        else
            echo "未知选项: $1"
            show_help
            exit 1
        fi
        ;;
esac
```
