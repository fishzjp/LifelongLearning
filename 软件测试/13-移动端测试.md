## 1. 移动端测试概述

### 1.1 移动端测试的特点

```
移动端测试的挑战：
├── 设备碎片化
│   ├── 不同品牌（Apple、Samsung、Huawei等）
│   ├── 不同屏幕尺寸（4.7寸到7寸+）
│   ├── 不同分辨率（720p到4K）
│   └── 不同硬件配置（CPU、内存、存储）
├── 操作系统碎片化
│   ├── iOS（多个版本并存）
│   ├── Android（多个版本+厂商定制）
│   └── 其他（HarmonyOS等）
├── 网络环境多样
│   ├── WiFi、4G、5G
│   ├── 信号强弱变化
│   └── 网络切换（WiFi↔4G）
├── 用户交互方式
│   ├── 触摸（单点、多点、手势）
│   ├── 传感器（陀螺仪、加速度计）
│   ├── 摄像头、GPS
│   └── 语音输入
└── 安装和更新
    ├── 应用商店审核
    ├── 热更新
    └── 版本兼容性
```

### 1.2 移动端测试分类

| 测试类型 | 测试内容 | 工具示例 | 重要性 |
|---------|---------|---------|--------|
| **功能测试** | 核心业务流程 | 手工测试 | ⭐⭐⭐⭐⭐ |
| **兼容性测试** | 不同设备/系统 | 云测平台 | ⭐⭐⭐⭐ |
| **性能测试** | 启动时间、内存 | PerfDog、GT | ⭐⭐⭐⭐ |
| **安全测试** | 数据加密、权限 | MobSF、Frida | ⭐⭐⭐⭐ |
| **自动化测试** | 回归测试 | Appium、Espresso | ⭐⭐⭐ |
| **用户体验测试** | 交互流畅度 | 手工测试 | ⭐⭐⭐⭐ |
| **网络测试** | 弱网、断网 | QNET、Charles | ⭐⭐⭐⭐ |

### 3.4 常用操作

```python
class MobileActions:
    def __init__(self, driver):
        self.driver = driver

    # 基本操作
    def click_element(self, locator):
        """点击元素"""
        element = self.driver.find_element(*locator)
        element.click()

    def input_text(self, locator, text):
        """输入文本"""
        element = self.driver.find_element(*locator)
        element.clear()
        element.send_keys(text)

    def get_text(self, locator):
        """获取文本"""
        element = self.driver.find_element(*locator)
        return element.text

    # 手势操作
    def swipe(self, start_x, start_y, end_x, end_y, duration=1000):
        """滑动"""
        self.driver.swipe(start_x, start_y, end_x, end_y, duration)

    def tap(self, x, y, duration=500):
        """点击坐标"""
        self.driver.tap([(x, y)], duration)

    def pinch(self, element, percent=200, steps=50):
        """捏合（放大/缩小）"""
        self.driver.pinch(element, percent, steps)

    def drag_and_drop(self, source, destination):
        """拖拽"""
        self.driver.drag_and_drop(source, destination)

    # 滚动
    def scroll_to_element(self, locator):
        """滚动到元素"""
        element = self.driver.find_element(*locator)
        self.driver.execute_script(
            "mobile: scroll",
            {"direction": "down", "element": element}
        )

    def scroll_down(self):
        """向下滚动"""
        self.driver.execute_script(
            "mobile: scroll",
            {"direction": "down"}
        )

    def scroll_up(self):
        """向上滚动"""
        self.driver.execute_script(
            "mobile: scroll",
            {"direction": "up"}
        )

    # 设备操作
    def press_back(self):
        """返回键（Android）"""
        self.driver.back()

    def press_home(self):
        """Home键"""
        self.driver.press_keycode(3)

    def press_menu(self):
        """菜单键"""
        self.driver.press_keycode(82)

    def lock_screen(self):
        """锁屏"""
        self.driver.lock(5)  # 锁屏5秒

    def unlock_screen(self):
        """解锁"""
        self.driver.unlock()

    def shake_device(self):
        """摇晃设备"""
        self.driver.shake()

    # 截图
    def take_screenshot(self, name):
        """截图"""
        self.driver.save_screenshot(f"{name}.png")

    # 应用管理
    def install_app(self, app_path):
        """安装应用"""
        self.driver.install_app(app_path)

    def remove_app(self, package_name):
        """卸载应用"""
        self.driver.remove_app(package_name)

    def reset_app(self):
        """重置应用"""
        self.driver.reset()

    def background_app(self, seconds):
        """后台运行应用"""
        self.driver.background_app(seconds)

    # 权限处理
    def grant_permission(self, package_name, permission):
        """授权权限（Android）"""
        self.driver.execute_script(
            "mobile: changePermissions",
            {
                "action": "grant",
                "appPackage": package_name,
                "permissions": permission
            }
        )

    def revoke_permission(self, package_name, permission):
        """撤销权限（Android）"""
        self.driver.execute_script(
            "mobile: changePermissions",
            {
                "action": "revoke",
                "appPackage": package_name,
                "permissions": permission
            }
        )

    # 弹窗处理
    def accept_alert(self):
        """接受弹窗"""
        self.driver.switch_to.alert.accept()

    def dismiss_alert(self):
        """取消弹窗"""
        self.driver.switch_to.alert.dismiss()

    def get_alert_text(self):
        """获取弹窗文本"""
        return self.driver.switch_to.alert.text
```

```python
    def test_login(self):
        """测试登录功能"""
        # 点击"我的"标签
        mine_tab = (AppiumBy.ID, "com.example.shopping:id/tab_mine")

        self.driver.find_element(*password_input).send_keys("123456")

        # 点击登录
        submit_btn = (AppiumBy.ID, "com.example.shopping:id/btn_submit")

        assert "欢迎" in text

        """测试添加商品到购物车"""
        # 搜索商品
        search_box = (AppiumBy.ID, "com.example.shopping:id/et_search")

        # 下拉刷新

        # 滚动浏览
```

```python
# 使用云测平台 API（以 Testin 为例）
import requests

class CloudTestPlatform:
    """云测平台接口"""

    def __init__(self, api_key):
        self.api_key = api_key
        self.base_url = "https://api.testin.cn"

    def upload_app(self, app_path):
        """上传应用"""
        url = f"{self.base_url}/v2/app/upload"
        files = {"file": open(app_path, "rb")}
        response = requests.post(url, files=files,
                               params={"api_key": self.api_key})
        return response.json()

    def start_test(self, app_id, devices):
        """启动测试"""
        url = f"{self.base_url}/v2/task/create"
        data = {
            "api_key": self.api_key,
            "app_id": app_id,
            "devices": devices,  # 设备列表
            "test_types": ["functional", "performance", "security"]
        }
        response = requests.post(url, json=data)
        return response.json()

    def get_report(self, task_id):
        """获取测试报告"""
        url = f"{self.base_url}/v2/task/report"
        params = {"api_key": self.api_key, "task_id": task_id}
        response = requests.get(url, params=params)
        return response.json()

# 使用示例
platform = CloudTestPlatform("your_api_key")
app_info = platform.upload_app("app.apk")
task = platform.start_test(app_info["app_id"],
                          ["Samsung_Galaxy_S21", "iPhone_13"])
report = platform.get_report(task["task_id"])
```

```python
# 使用 PerfDog（性能狗）API
class PerformanceTester:
    """性能测试工具"""

    def __init__(self):
        self.metrics = {}

    def test_app_launch(self, package_name, iterations=5):
        """测试应用启动时间"""
        import subprocess
        import time

        times = []
        for i in range(iterations):
            # 强制停止应用
            subprocess.run([
                "adb", "shell", "am", "force-stop", package_name
            ])
            time.sleep(2)

            # 冷启动并计时
            start = time.time()
            subprocess.run([
                "adb", "shell", "am", "start", "-n",
                f"{package_name}/.MainActivity"
            ])

            # 等待应用完全启动
            while True:
                result = subprocess.run([
                    "adb", "shell", "dumpsys", "window",
                    "|", "grep", "mCurrentFocus"
                ], capture_output=True, text=True)

                if package_name in result.stdout:
                    break

            end = time.time()
            times.append(end - start)

        self.metrics["launch_time"] = {
            "avg": sum(times) / len(times),
            "min": min(times),
            "max": max(times),
            "all": times
        }
        return self.metrics["launch_time"]

    def test_memory_usage(self, package_name):
        """测试内存使用"""
        import subprocess

        # 获取内存信息
        result = subprocess.run([
            "adb", "shell", "dumpsys", "meminfo", package_name
        ], capture_output=True, text=True)

        # 解析输出
        lines = result.stdout.split("\n")
        memory_info = {}

        for line in lines:
            if "TOTAL" in line:
                parts = line.split()
                memory_info["total"] = int(parts[1])
            elif "Native" in line:
                parts = line.split()
                memory_info["native"] = int(parts[1])
            elif "Dalvik" in line:
                parts = line.split()
                memory_info["dalvik"] = int(parts[1])

        self.metrics["memory"] = memory_info
        return memory_info

    def test_cpu_usage(self, package_name, duration=30):
        """测试CPU使用率"""
        import subprocess
        import time

        cpu_samples = []
        end_time = time.time() + duration

        while time.time() < end_time:
            result = subprocess.run([
                "adb", "shell", "top", "-n", "1", "|", "grep", package_name
            ], capture_output=True, text=True)

            if result.stdout:
                # 解析CPU使用率
                parts = result.stdout.split()
                if len(parts) > 8:
                    cpu = float(parts[8].replace("%", ""))
                    cpu_samples.append(cpu)

            time.sleep(2)

        self.metrics["cpu"] = {
            "avg": sum(cpu_samples) / len(cpu_samples),
            "max": max(cpu_samples),
            "samples": cpu_samples
        }
        return self.metrics["cpu"]

    def test_network_traffic(self, package_name, test_duration=60):
        """测试网络流量"""
        import subprocess
        import time

        # 重置流量统计
        subprocess.run([
            "adb", "shell", "cat", "/proc/net/uid_stats/10000", ">",
            "/dev/null", "2>&1"
        ])

        time.sleep(test_duration)

        # 获取流量数据
        result = subprocess.run([
            "adb", "shell", "cat", "/proc/net/uid_stats/10000"
        ], capture_output=True, text=True)

        # 解析流量（简化）
        self.metrics["network"] = {
            "rx_bytes": 0,  # 接收字节
            "tx_bytes": 0   # 发送字节
        }

        return self.metrics["network"]

    def generate_report(self):
        """生成性能测试报告"""
        report = {
            "timestamp": time.time(),
            "metrics": self.metrics,
            "summary": {
                "launch_time_ok": self.metrics.get("launch_time", {}).get("avg", 0) < 2.0,
                "memory_ok": self.metrics.get("memory", {}).get("total", 0) < 300000,  # 300MB
                "cpu_ok": self.metrics.get("cpu", {}).get("avg", 0) < 30.0
            }
        }
        return report

# 使用示例

print("测试内存...")
```

```python
class NetworkSimulator:
    """网络模拟工具"""

    def __init__(self):
        self.original_network = None

        """
        设置网络条件（需要root权限）
        latency: 延迟(ms)
        packet_loss: 丢包率(%)
        bandwidth: 带宽(kbps)
        """
        # 使用 tc 命令设置网络限制
        commands = [
            f"adb shell tc qdisc add dev wlan0 root netem delay {latency}ms",
            f"adb shell tc qdisc add dev wlan0 root netem loss {packet_loss}%",
            f"adb shell tc qdisc add dev wlan0 root netem rate {bandwidth}kbit"
        ]

        for cmd in commands:
            subprocess.run(cmd.split())

    def reset_network(self):
        """重置网络"""
        subprocess.run([
            "adb", "shell", "tc", "qdisc", "del", "dev", "wlan0", "root"
        ])

    def test_weak_network(self, app_package, test_actions):
        """弱网测试"""
        # 设置弱网条件
        self.set_network_condition(
            latency=500,      # 500ms延迟
            packet_loss=10,   # 10%丢包
            bandwidth=100     # 100kbps带宽
        )

        results = []
        for action in test_actions:
            start = time.time()
            try:
                # 执行操作
                result = action()
                elapsed = time.time() - start
                results.append({
                    "action": action.__name__,
                    "time": elapsed,

                })

                results.append({
                    "action": action.__name__,
                    "time": time.time() - start,
                    "success": False,
                    "error": str(e)
```

```python
# 移动端安全测试基础
class MobileSecurityTest:
    """移动端安全测试"""

    def __init__(self, package_name):
        self.package = package_name

    def check_manifest(self):
        """检查AndroidManifest.xml"""
        # 使用 aapt 解析 APK
        import subprocess

        result = subprocess.run([
            "aapt", "dump", "badging", "app.apk"
        ], capture_output=True, text=True)

        # 检查敏感权限
        sensitive_permissions = [
            "READ_SMS", "SEND_SMS", "READ_CONTACTS",
            "ACCESS_FINE_LOCATION", "CAMERA", "RECORD_AUDIO"
        ]

        found_permissions = []
        for perm in sensitive_permissions:
            if perm in result.stdout:
                found_permissions.append(perm)

        return found_permissions

    def check_exported_components(self):
        """检查导出组件"""
        result = subprocess.run([
            "adb", "shell", "dumpsys", "package", self.package
        ], capture_output=True, text=True)

        # 检查导出的Activity、Service等
        exported = []
        lines = result.stdout.split("\n")

        for line in lines:
            if "exported=true" in line.lower():
                exported.append(line.strip())

        return exported

    def check_network_security(self):
        """检查网络安全配置"""
        # 检查是否允许明文HTTP
        result = subprocess.run([
            "adb", "shell", "cat",
            f"/data/data/{self.package}/network_security_config.xml"
        ], capture_output=True, text=True)

        if result.returncode == 0:
            if "allowHttp" in result.stdout or "cleartextTrafficPermitted" in result.stdout:
                return "WARNING: 明文HTTP可能被允许"

        return "OK"

    def check_root_detection(self):
        """检查Root检测"""
        # 尝试访问su命令
        result = subprocess.run([
            "adb", "shell", "which", "su"
        ], capture_output=True, text=True)

        if result.returncode == 0:
            return "设备已Root"
        return "设备未Root"

    def run_security_scan(self):
        """运行安全扫描"""
        report = {
            "package": self.package,
            "timestamp": time.time(),
            "findings": []
        }

        # 检查敏感权限
        perms = self.check_manifest()
        if perms:
            report["findings"].append({
                "level": "HIGH",
                "category": "权限",
                "description": "应用请求敏感权限",
                "details": perms
            })

        # 检查导出组件
        exported = self.check_exported_components()
        if exported:
            report["findings"].append({
                "level": "MEDIUM",
                "category": "组件安全",
                "description": "存在导出组件",
                "details": exported
            })

        # 检查网络安全
        network = self.check_network_security()
        if "WARNING" in network:
            report["findings"].append({
                "level": "HIGH",
                "category": "网络安全",
                "description": network
            })

        return report

# 使用示例
security_test = MobileSecurityTest("com.example.app")
report = security_test.run_security_scan()

print("安全测试报告:")
for finding in report["findings"]:
    print(f"[{finding['level']}] {finding['category']}: {finding['description']}")
```

```python
class CharlesProxy:
    """Charles 代理工具"""

    def __init__(self, host="localhost", port=8888):
        self.host = host
        self.port = port

    def get_requests(self):
        """获取请求列表"""
        import requests

        # Charles Remote API
        url = f"http://{self.host}:{self.port}/charles/api/requests"
        response = requests.get(url)
        return response.json()

    def clear_requests(self):
        """清空请求"""
        import requests
        url = f"http://{self.host}:{self.port}/charles/api/clear"
        requests.post(url)

    def filter_requests(self, url_pattern):
        """过滤请求"""
        all_requests = self.get_requests()
        return [req for req in all_requests if url_pattern in req["url"]]
```
