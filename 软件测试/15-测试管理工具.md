# TestRail API 配置示例
import requests
import json

class TestRailAPI:
    def __init__(self, base_url, username, password):
        self.base_url = base_url.rstrip('/')
        self.auth = (username, password)
        self.headers = {'Content-Type': 'application/json'}

    def _request(self, method, uri, data=None):
        """发送API请求"""
        url = f"{self.base_url}/index.php?/api/v2/{uri}"
        response = requests.request(
            method, url,
            auth=self.auth,
            headers=self.headers,
            json=data
        )
        return response.json()

    # 项目管理
    def get_projects(self):
        """获取所有项目"""
        return self._request('GET', 'get_projects')

    def get_project(self, project_id):
        """获取单个项目"""
        return self._request('GET', f'get_project/{project_id}')

    # 测试套件
    def get_suites(self, project_id):
        """获取测试套件"""
        return self._request('GET', f'get_suites/{project_id}')

    def add_suite(self, project_id, name, description=""):
        """添加测试套件"""
        data = {
            "name": name,
            "description": description
        }
        return self._request('POST', f'add_suite/{project_id}', data)

    # 测试用例
    def get_cases(self, project_id, suite_id=None):
        """获取测试用例"""
        uri = f'get_cases/{project_id}'
        if suite_id:
            uri += f'&suite_id={suite_id}'
        return self._request('GET', uri)

    def add_case(self, suite_id, title, **kwargs):
        """添加测试用例"""
        data = {
            "title": title,
            "template_id": kwargs.get('template_id', 1),
            "type_id": kwargs.get('type_id', 1),
            "priority_id": kwargs.get('priority_id', 2),
            "estimate": kwargs.get('estimate', ''),
            "milestone_id": kwargs.get('milestone_id'),
            "refs": kwargs.get('refs', '')
        }
        # 添加自定义字段
        for key, value in kwargs.items():
            if key not in data:
                data[key] = value

        return self._request('POST', f'add_case/{suite_id}', data)

# 使用示例
testrail = TestRailAPI(
    base_url="https://your-company.testrail.io",
    username="your-email@example.com",
    password="your-api-key"
)

# 获取项目
projects = testrail.get_projects()
print(f"找到 {len(projects)} 个项目")

# 添加测试套件
suite = testrail.add_suite(
    project_id=1,
    name="用户管理模块",
    description="测试用户注册、登录、权限管理"
)

def generate_test_report(testrail, project_id, run_id=None):
    """生成测试报告"""
    report = {
        "project": testrail._request('GET', f'get_project/{project_id}'),
        "summary": {},
        "details": []
    }

    if run_id:
        # 获取运行结果
        tests = testrail._request('GET', f'get_tests/{run_id}')

        # 统计
        total = len(tests)
        passed = len([t for t in tests if t['status_id'] == 1])
        failed = len([t for t in tests if t['status_id'] == 2])
        blocked = len([t for t in tests if t['status_id'] == 3])

        report["summary"] = {
            "total": total,
            "passed": passed,
            "failed": failed,
            "blocked": blocked,
            "pass_rate": f"{(passed/total*100):.1f}%" if total > 0 else "0%"
        }

        # 详细信息
        for test in tests:
            report["details"].append({
                "case_id": test['case_id'],
                "title": test['title'],
                "status": test['status_name'],
                "elapsed": test.get('elapsed', ''),
                "comment": test.get('comment', ''),
                "defects": test.get('defects', '')
            })

    return report

# 使用示例
report = generate_test_report(testrail, project_id=1, run_id=123)
print(f"测试通过率: {report['summary']['pass_rate']}")
print(f"总用例数: {report['summary']['total']}")

# Zephyr Scale API
import requests

class ZephyrScale:
    def __init__(self, base_url, api_token):
        self.base_url = base_url.rstrip('/')
        self.headers = {
            'Authorization': f'Bearer {api_token}',
            'Content-Type': 'application/json'
        }

    def create_test_case(self, project_key, **kwargs):
        """创建测试用例"""
        data = {
            "projectKey": project_key,
            "name": kwargs['name'],
            "folderId": kwargs.get('folder_id'),
            "objective": kwargs.get('objective', ''),
            "steps": kwargs.get('steps', []),
            "labels": kwargs.get('labels', [])
        }
        response = requests.post(
            f"{self.base_url}/testcases",
            headers=self.headers,
            json=data
        )
        return response.json()

    def create_test_cycle(self, project_key, **kwargs):
        """创建测试周期"""
        data = {
            "projectKey": project_key,
            "name": kwargs['name'],
            "description": kwargs.get('description', ''),
            "folderId": kwargs.get('folder_id')
        }
        response = requests.post(
            f"{self.base_url}/testcycles",
            headers=self.headers,
            json=data
        )
        return response.json()

    def add_tests_to_cycle(self, cycle_id, test_case_keys):
        """添加测试用例到周期"""
        data = {
            "testCases": test_case_keys
        }
        response = requests.post(
            f"{self.base_url}/testcycles/{cycle_id}/testcases",
            headers=self.headers,
            json=data
        )
        return response.json()

    def update_test_execution(self, execution_id, status, comment=""):
        """更新测试执行结果"""
        data = {
            "status": status,  # PASS, FAIL, BLOCKED, WIP
            "comment": comment
        }
        response = requests.put(
            f"{self.base_url}/testexecutions/{execution_id}",
            headers=self.headers,
            json=data
        )
        return response.json()

class TestExecutionTracker:
    def __init__(self, tool_name):
        self.tool = tool_name
        self.execution_status = {}

    def start_execution(self, test_plan_id):
        """开始执行测试"""
        self.execution_status = {
            "plan_id": test_plan_id,
            "start_time": __import__('datetime').datetime.now(),
            "total": 0,
            "passed": 0,
            "failed": 0,
            "blocked": 0,
            "in_progress": 0,
            "cases": []
        }

    def record_execution(self, case_id, status, comment="", defects=None):
        """记录执行结果"""
        execution = {
            "case_id": case_id,
            "status": status,
            "comment": comment,
            "defects": defects,
            "timestamp": __import__('datetime').datetime.now()
        }
        self.execution_status["cases"].append(execution)

        # 更新统计
        self.execution_status["total"] += 1
        if status == "passed":
            self.execution_status["passed"] += 1
        elif status == "failed":
            self.execution_status["failed"] += 1
        elif status == "blocked":
            self.execution_status["blocked"] += 1
        elif status == "in_progress":
            self.execution_status["in_progress"] += 1

    def get_progress(self):
        """获取执行进度"""
        total = self.execution_status["total"]
        if total == 0:
            return "0%"

        completed = (
            self.execution_status["passed"] +
            self.execution_status["failed"] +
            self.execution_status["blocked"]
        )
        return f"{(completed/total*100):.1f}%"

    def get_pass_rate(self):
        """获取通过率"""
        total = self.execution_status["passed"] + self.execution_status["failed"]
        if total == 0:
            return "0%"
        return f"{(self.execution_status['passed']/total*100):.1f}%"

    def generate_daily_report(self):
        """生成日报"""
        report = f"""
测试执行日报
================
执行进度: {self.get_progress()}
通过率: {self.get_pass_rate()}

统计:
- 总用例: {self.execution_status['total']}
- 已通过: {self.execution_status['passed']}
- 失败: {self.execution_status['failed']}
- 阻塞: {self.execution_status['blocked']}
- 进行中: {self.execution_status['in_progress']}

今日发现缺陷: {len([c for c in self.execution_status['cases'] if c['defects']])}
"""
        return report

{test_case.get('steps', '')}

## 实际结果
{failure_info['actual']}

## 预期结果
{test_case.get('expected', '')}

## 环境信息
- 浏览器: {failure_info.get('browser', 'N/A')}
- 系统: {failure_info.get('os', 'N/A')}
- 版本: {failure_info.get('version', 'N/A')}

## 附件
{failure_info.get('screenshots', '')}
{failure_info.get('logs', '')}
"""

    def _map_priority(self, test_priority):
        """映射优先级"""
        priority_map = {
            "P0": "Highest",
            "P1": "High",
            "P2": "Medium",
            "P3": "Low"
        }
        return priority_map.get(test_priority, "Medium")

    def sync_defect_status(self):
        """同步缺陷状态"""
        # 获取测试用例关联的缺陷
        linked_defects = self.test_tool.get_linked_defects()

            # 如果缺陷已修复，标记测试为待回归

                self.test_tool.update_test_status(
                    defect["test_case_id"],
                    "needs_retest"
                )
