class ECommerceAPIClient:
    """电商API客户端"""

    def __init__(self, base_url: str):
        self.base_url = base_url
        self.session = requests.Session()

    def register(self, username: str, phone: str, password: str) -> Dict:
        """用户注册"""
        response = self.session.post(
            f"{self.base_url}/api/user/register",
            json={
                "username": username,
                "phone": phone,
                "password": password
            }
        )
        return response.json()

    def login(self, phone: str, password: str) -> Dict:
        """用户登录"""
        response = self.session.post(
            f"{self.base_url}/api/user/login",
            json={"phone": phone, "password": password}
        )
        result = response.json()
        if result.get("success"):
            self.session.headers.update({
                "Authorization": f"Bearer {result['token']}"
            })
        return result

    def search_products(self, keyword: str, page: int = 1) -> Dict:
        """搜索商品"""
        response = self.session.get(
            f"{self.base_url}/api/product/search",
            params={"keyword": keyword, "page": page, "size": 20}
        )
        return response.json()

    def get_product_detail(self, product_id: str) -> Dict:
        """获取商品详情"""
        response = self.session.get(
            f"{self.base_url}/api/product/{product_id}"
        )
        return response.json()

    def add_to_cart(self, product_id: str, quantity: int) -> Dict:
        """添加到购物车"""
        response = self.session.post(
            f"{self.base_url}/api/cart",
            json={"product_id": product_id, "quantity": quantity}
        )
        return response.json()

    def create_order(self, product_ids: List[str], address_id: str,
                     coupon_code: Optional[str] = None) -> Dict:
        """创建订单"""
        response = self.session.post(
            f"{self.base_url}/api/order",
            json={
                "product_ids": product_ids,
                "address_id": address_id,
                "coupon_code": coupon_code
            }
        )
        return response.json()

    def pay_order(self, order_id: str, payment_method: str) -> Dict:
        """支付订单"""
        response = self.session.post(
            f"{self.base_url}/api/order/{order_id}/pay",
            json={"payment_method": payment_method}
        )
        return response.json()

    def cancel_order(self, order_id: str) -> Dict:
        """取消订单"""
        response = self.session.post(
            f"{self.base_url}/api/order/{order_id}/cancel"
        )

class BaseECommerceTest(ABC):
    """电商测试基类"""

    @pytest.fixture(autouse=True)
    def setup(self, api_client, test_context):
        """测试设置"""
        self.client = api_client
        self.context = test_context
        self.data_factory = TestDataFactory()

    @abstractmethod
    def get_base_url(self) -> str:
        """获取基础URL"""
        pass

    def assert_success(self, result: Dict, message: str = ""):
        """断言成功"""
        assert result.get("success") is True, f"{message}: {result}"

    def assert_fail(self, result: Dict, message: str = ""):
        """断言失败"""
        assert result.get("success") is False, f"{message}: {result}"

    def assert_order_status(self, order_id: str, expected_status: str):
        """断言订单状态"""
        # 查询订单详情
        response = self.client.session.get(
            f"{self.client.base_url}/api/order/{order_id}"
        )
        order = response.json()
        assert order["status"] == expected_status, \
            f"订单状态错误: 期望 {expected_status}, 实际 {order['status']}"

        # 并发参数
        concurrent_users = 100
        success_count = 0
        fail_count = 0
        lock = threading.Lock()

        def create_order_thread():
            nonlocal success_count, fail_count
            try:
                # 每个线程创建独立客户端
                client = ECommerceAPIClient(self.client.base_url)
                client.login(user_data["phone"], user_data["password"])

                result = client.create_order([product_id], address_id)

                with lock:
                    if result.get("success"):
                        success_count += 1
                    else:
                        fail_count += 1
            except Exception as e:
                with lock:
                    fail_count += 1

        # 启动并发线程
        threads = []
        start_time = time.time()

        for _ in range(concurrent_users):
            t = threading.Thread(target=create_order_thread)
            threads.append(t)
            t.start()

        # 等待所有线程完成
        for t in threads:
            t.join()

        duration = time.time() - start_time

        # 验证结果
        print(f"并发测试结果: 成功 {success_count}, 失败 {fail_count}, 耗时 {duration:.2f}s")

        # 断言：至少90%的请求成功
        assert success_count >= concurrent_users * 0.9, \
            f"并发成功率过低: {success_count}/{concurrent_users}"
