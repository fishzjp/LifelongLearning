## 架构演进时间线

```
1998: LeNet-5      → CNN开山之作
2012: AlexNet      → 深度学习引爆点
2014: VGG          → 深度=性能
2015: ResNet       → 残差连接（里程碑）
2016: DenseNet     → 密集连接
2017: MobileNet    → 深度可分离
2018: ShuffleNet   → 通道混洗
2019: EfficientNet → 复合缩放
2020: ViT/GhostNet → Transformer/高效设计
2021: Swin         → 分层Transformer
2022: ConvNeXt     → 现代CNN
```

## 选择决策树

```python
def choose_architecture(task_type, hardware, data_size, accuracy_req, latency_req=None):
    """
    架构选择决策函数
    
    Args:
        task_type: 'classification', 'detection', 'segmentation'
        hardware: 'server', 'desktop', 'mobile', 'edge'
        data_size: 'small', 'medium', 'large'
        accuracy_req: 'high', 'medium', 'low'
        latency_req: 延迟要求(ms)，可选
    
    Returns:
        推荐架构和理由
    """
    recommendations = []
    
    # 硬件约束
    if hardware == 'mobile' or hardware == 'edge':
        if latency_req and latency_req < 20:
            recommendations.append(('MobileNetV3-Small', '极致速度'))
        elif accuracy_req == 'high':
            recommendations.append(('EfficientNet-Lite', '精度速度平衡'))
        else:
            recommendations.append(('MobileNetV2', '通用选择'))
    
    # 任务类型
    if task_type == 'classification':
        if data_size == 'large' and accuracy_req == 'high':
            recommendations.extend([
                ('ConvNeXt', '现代CNN最佳'),
                ('ViT-Base', 'Transformer架构'),
                ('EfficientNet-B7', '传统架构巅峰')
            ])
        elif data_size == 'medium':
            recommendations.extend([
                ('ResNet50', '标准基准'),
                ('EfficientNet-B0', '高效选择'),
                ('DenseNet121', '特征复用')
            ])
        else:
            recommendations.extend([
                ('ResNet18', '简单有效'),
                ('MobileNetV2', '轻量快速'),
                ('GhostNet', '参数高效')
            ])
    
    elif task_type == 'detection':
        if hardware == 'server':
            recommendations.extend([
                ('YOLOv8', '单阶段最佳'),
                ('Faster R-CNN', '两阶段高精度')
            ])
        else:
            recommendations.extend([
                ('YOLOv8n', '轻量检测'),
                ('SSD-MobileNet', '移动端检测')
            ])
    
    elif task_type == 'segmentation':
        if accuracy_req == 'high':
            recommendations.extend([
                ('DeepLabV3+', '语义分割最佳'),
                ('U-Net++', '医学图像')
            ])
        else:
            recommendations.extend([
                ('U-Net', '经典选择'),
                ('FastSCNN', '实时分割')
            ])
    
    # 默认推荐
    if not recommendations:
        recommendations.append(('ResNet50', '默认选择'))
    
    return recommendations

# 使用示例

def architecture_selection_demo():
    """架构选择演示"""
    scenarios = [
        {'task': 'classification', 'hardware': 'mobile', 'data': 'medium', 'acc': 'high', 'lat': 30},
        {'task': 'classification', 'hardware': 'server', 'data': 'large', 'acc': 'high', 'lat': None},
        {'task': 'detection', 'hardware': 'edge', 'data': 'small', 'acc': 'medium', 'lat': 50},
    ]
    
    for i, scenario in enumerate(scenarios, 1):
        print(f"\n场景 {i}: {scenario['task']} on {scenario['hardware']}")
        recs = choose_architecture(
            scenario['task'], 
            scenario['hardware'], 
            scenario['data'], 
            scenario['acc'],
            scenario['lat']
        )
        for arch, reason in recs[:3]:  # 显示前3个推荐
            print(f"  → {arch}: {reason}")
```

### 1. 移动端APP开发

```bash
# 场景：实时图像分类
# 约束：延迟 < 50ms，模型 < 10MB
# 推荐：MobileNetV3 > EfficientNet-Lite > GhostNet

def mobile_app_model():
    # 使用预训练的MobileNetV3
    from torchvision.models import mobilenet_v3_small
    model = mobilenet_v3_small(weights='DEFAULT')
    
    # 量化以进一步减小体积
    model.qconfig = torch.quantization.get_default_qconfig('fbgemm')
    model = torch.quantization.prepare(model, inplace=False)
    model = torch.quantization.convert(model, inplace=False)
    
    return model
```

### 2. 边缘计算设备

```bash
# 场景：工业质检摄像头
# 约束：内存 < 100MB，功耗低
# 推荐：SqueezeNet > MobileNetV2 > EfficientNet-B0

def edge_device_model():
    # 使用SqueezeNet的变体
    class EdgeSqueezeNet(nn.Module):
        def __init__(self):
            super().__init__()
            self.features = nn.Sequential(
                nn.Conv2d(3, 64, 3, 2, 1),
                nn.ReLU(),
                nn.MaxPool2d(3, 2, ceil_mode=True),
                # ... 简化结构
            )
            self.classifier = nn.Sequential(
                nn.Dropout(0.5),
                nn.Conv2d(64, 10, 1),
                nn.AdaptiveAvgPool2d(1)
            )
        
        def forward(self, x):
            x = self.features(x)
            x = self.classifier(x)
            return x.view(x.size(0), -1)
    
    return EdgeSqueezeNet()
```

# 场景：视频目标检测
# 约束：30FPS实时处理
# 推荐：YOLOv8n > MobileNetV2-SSD > EfficientNet-B0

def real_time_video_model():
    # 使用轻量级检测器
    class LightweightDetector(nn.Module):
        def __init__(self, num_classes=80):
            super().__init__()
            # 特征提取器
            self.backbone = mobilenet_v2(pretrained=True).features

            self.anchor_generator = AnchorGenerator(

## 架构复杂度分析工具

```python
def analyze_architecture_complexity(model, input_shape=(1, 3, 224, 224)):
    """
    全面分析架构复杂度
    """
    import thop  # 需要安装: pip install thop
    
    # 参数量
    params = sum(p.numel() for p in model.parameters())
    
    # 计算量 (FLOPs)
    flops, params_est = thop.profile(model, inputs=(torch.randn(*input_shape),), verbose=False)
    
    # 内存占用
    model.eval()
    with torch.no_grad():
        dummy_input = torch.randn(*input_shape)
        output = model(dummy_input)
        memory_mb = torch.cuda.memory_allocated() / 1024**2 if torch.cuda.is_available() else 0
    
    # 层数统计
    conv_count = sum(1 for m in model.modules() if isinstance(m, nn.Conv2d))
    bn_count = sum(1 for m in model.modules() if isinstance(m, nn.BatchNorm2d))
    linear_count = sum(1 for m in model.modules() if isinstance(m, nn.Linear))
    
    print(f"\n📊 架构复杂度分析")
    print(f"{'参数量':<15} {params/1e6:.2f}M")
    print(f"{'计算量':<15} {flops/1e9:.2f}G FLOPs")
    print(f"{'内存占用':<15} {memory_mb:.2f}MB")
    print(f"{'卷积层':<15} {conv_count}")
    print(f"{'BN层':<15} {bn_count}")
    print(f"{'全连接层':<15} {linear_count}")
    
    # 效率指标
    accuracy_efficiency = 76.2 / (params/1e6) if params > 0 else 0  # 假设ResNet50准确率为76.2%
    print(f"{'参数效率':<15} {accuracy_efficiency:.2f}% per M params")
    
    return {
        'params': params,
        'flops': flops,
        'memory_mb': memory_mb,
        'layers': {
            'conv': conv_count,
            'bn': bn_count,
            'linear': linear_count
        }
    }
```

## 硬件约束下的选择

```python
def hardware_aware_selection(memory_mb, latency_ms, accuracy_req):
    """硬件感知的架构选择"""
    
    if memory_mb < 100:  # 极端受限
        return "SqueezeNet 1.0"
    
    elif memory_mb < 500:  # 移动端
        if latency_ms < 20:
            return "MobileNetV3-Small"
        else:
            return "MobileNetV2"
    
    elif memory_mb < 2000:  # 边缘计算
        if accuracy_req == "high":
            return "EfficientNet-B0"
        else:
            return "ShuffleNetV2"
    
    else:  # 服务器
        if accuracy_req == "very_high":
            return "EfficientNet-B7"
        else:
            return "ResNet50"
```

## 🎯 快速决策指南

### 问题1: 我应该从哪个架构开始？

**答案**: 
- **学习/研究**: ResNet50
- **移动端**: MobileNetV2
- **高精度**: EfficientNet-B0
- **极致轻量**: SqueezeNet

### 问题2: 如何选择合适的大小？

**答案**: 
- **参数量 < 5M**: 移动端/边缘设备
- **参数量 5-50M**: 通用场景
- **参数量 > 50M**: 服务器/高精度需求

### 问题3: 何时使用Transformer架构？

**答案**: 
- **数据量 > 10万**: 可以尝试ViT
- **需要SOTA精度**: Swin Transformer
- **计算资源充足**: ViT-Large
