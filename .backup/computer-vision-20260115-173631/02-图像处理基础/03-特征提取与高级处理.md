```python
def compute_gradient(img):
    """计算图像梯度"""
    # Sobel算子
    kernel_x = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]])
    kernel_y = np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]])

    grad_y = convolve(img.astype(np.float32), kernel_y)

    magnitude = np.sqrt(grad_x**2 + grad_y**2)
    direction = np.arctan2(grad_y, grad_x)

    return grad_x, grad_y, magnitude, direction
```

### 2.1 Harris角点检测

```python
print("\n=== Harris角点检测 ===")

def harris_corner_detector(img, k=0.04, threshold=0.01):
    """Harris角点检测器"""

    # 1. 计算梯度
    grad_x = convolve(img.astype(np.float32), np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]))
    grad_y = convolve(img.astype(np.float32), np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]]))

    # 2. 计算结构张量分量
    Ixx = gaussian_filter(grad_x**2, sigma=1)
    Iyy = gaussian_filter(grad_y**2, sigma=1)
    Ixy = gaussian_filter(grad_x * grad_y, sigma=1)

    # 3. 计算Harris响应
    det = Ixx * Iyy - Ixy**2
    trace = Ixx + Iyy
    R = det - k * trace**2

    # 4. 非极大值抑制
    corners = np.zeros_like(R)
    for i in range(1, R.shape[0]-1):
        for j in range(1, R.shape[1]-1):
            if R[i, j] > threshold and R[i, j] == np.max(R[i-1:i+2, j-1:j+2]):
                corners[i, j] = R[i, j]

    return corners, R
```

### 2.2 Shi-Tomasi角点检测

```python
print("\n=== Shi-Tomasi角点检测 ===")

def shi_tomasi_detector(img, max_corners=100, quality_level=0.01, min_distance=5):
    """Shi-Tomasi角点检测器（更好的角点）"""

    # 计算梯度
    grad_x = convolve(img.astype(np.float32), np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]))
    grad_y = convolve(img.astype(np.float32), np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]]))

    # 计算结构张量
    Ixx = gaussian_filter(grad_x**2, sigma=1)
    Iyy = gaussian_filter(grad_y**2, sigma=1)
    Ixy = gaussian_filter(grad_x * grad_y, sigma=1)

    # 计算最小特征值
    corners = []
    h, w = img.shape

    for i in range(1, h-1):
        for j in range(1, w-1):
            M = np.array([[Ixx[i, j], Ixy[i, j]],
                         [Ixy[i, j], Iyy[i, j]]])

            # 最小特征值
            eigenvalues = np.linalg.eigvals(M)
            min_eigenvalue = np.min(eigenvalues)

            if min_eigenvalue > quality_level * 1000:
                corners.append((j, i, min_eigenvalue))

    # 非极大值抑制
    corners = sorted(corners, key=lambda x: x[2], reverse=True)
    selected = []

    for corner in corners:
        x, y, val = corner
        # 检查距离
        too_close = False
        for sx, sy, _ in selected:
            if np.sqrt((x-sx)**2 + (y-sy)**2) < min_distance:
                too_close = True
                break

        if not too_close:
            selected.append(corner)

        if len(selected) >= max_corners:
            break

    return selected
```

```python
print("\n=== FAST角点检测 ===")

def fast_corner_detector(img, threshold=20, n=12):
    """FAST角点检测（快速角点）"""

    # 简化版本：检查圆形邻域
    h, w = img.shape
    corners = np.zeros_like(img)

    # 圆形邻域偏移（16个点）
    offsets = [
        (0, 3), (1, 3), (2, 2), (3, 1),
        (3, 0), (3, -1), (2, -2), (1, -3),
        (0, -3), (-1, -3), (-2, -2), (-3, -1),
        (-3, 0), (-3, 1), (-2, 2), (-1, 3)
    ]

    for i in range(3, h-3):
        for j in range(3, w-3):
            center = img[i, j]

            # 检查连续的n个点
            bright_count = 0
            dark_count = 0

            for dx, dy in offsets:
                neighbor = img[i+dy, j+dx]
                if neighbor > center + threshold:
                    bright_count += 1
                elif neighbor < center - threshold:
                    dark_count += 1

            # 如果有连续n个点满足条件
            if bright_count >= n or dark_count >= n:
                corners[i, j] = 1

    return corners
```

```python
print("\n=== 轮廓检测 ===")

def find_contours_simple(img, threshold=128):
    """简化版轮廓检测"""
    # 二值化
    binary = img > threshold

    # 使用连通域分析
    from scipy.ndimage import label
    labeled, num_features = label(binary)

    contours = []
    for i in range(1, num_features + 1):
        mask = labeled == i
        # 获取边界
        y, x = np.where(mask)
        if len(x) > 0:
            # 简单边界框作为轮廓表示
            min_x, max_x = x.min(), x.max()
            min_y, max_y = y.min(), y.max()
            contours.append({
                'label': i,
                'bbox': (min_x, min_y, max_x, max_y),
                'area': len(x),
                'center': ((min_x + max_x) // 2, (min_y + max_y) // 2)
            })

    return contours, labeled
```

### 3.2 轮廓特征

```python
print("\n=== 轮廓特征计算 ===")

def compute_contour_features(contour_img, label_img, contour_id):
    """计算轮廓特征"""

    mask = label_img == contour_id
    y, x = np.where(mask)

    if len(x) == 0:
        return None

    # 1. 几何特征
    area = len(x)
    perimeter = 0  # 简化：边界像素数

    # 计算周长（边界像素）
    from scipy.ndimage import binary_dilation
    dilated = binary_dilation(mask)
    boundary = dilated & ~mask
    perimeter = np.sum(boundary)

    # 2. 中心点
    center_x = np.mean(x)
    center_y = np.mean(y)

    # 3. 方向（主轴）
    # 计算中心矩
    x_centered = x - center_x
    y_centered = y - center_y

    # 协方差矩阵
    cov_xx = np.sum(x_centered**2) / area
    cov_yy = np.sum(y_centered**2) / area
    cov_xy = np.sum(x_centered * y_centered) / area

    # 主轴方向
    angle = 0.5 * np.arctan2(2 * cov_xy, cov_xx - cov_yy)

    # 4. 形状描述子
    # 圆形度
    if perimeter > 0:
        circularity = 4 * np.pi * area / (perimeter ** 2)
    else:
        circularity = 0

    # 宽高比
    bbox = (x.min(), y.min(), x.max(), y.max())
    width = bbox[2] - bbox[0]
    height = bbox[3] - bbox[1]
    aspect_ratio = width / height if height > 0 else 0

    return {
        'area': area,
        'perimeter': perimeter,
        'center': (center_x, center_y),
        'angle': angle,
        'circularity': circularity,
        'aspect_ratio': aspect_ratio,
        'bbox': bbox
    }

# 计算每个轮廓的特征

fig, axes = plt.subplots(2, 2, figsize=(10, 10))
axes = axes.flatten()

for idx, contour in enumerate(contours):
```

## 形态学操作

### 4.1 基础形态学操作

```python
print("\n=== 形态学基础 ===")

def morphology_demo():
    """形态学操作演示"""

    # 创建测试图像（带噪声的文字）
    img = np.zeros((80, 100), dtype=np.uint8)
    # 模拟文字
    img[20:30, 20:30] = 255
    img[20:30, 40:50] = 255
    img[20:30, 60:70] = 255
    img[40:50, 30:60] = 255

    # 添加椒盐噪声
    img = add_salt_pepper_noise(img, 0.05, 0.05)

    # 结构元素
    kernel = np.ones((3, 3), dtype=bool)

    # 1. 腐蚀（Erosion）
    def erosion(img, kernel):
        h, w = img.shape
        kh, kw = kernel.shape
        pad_h, pad_w = kh//2, kw//2

        result = np.zeros_like(img)
        for i in range(pad_h, h-pad_h):
            for j in range(pad_w, w-pad_w):
                region = img[i-pad_h:i+pad_h+1, j-pad_w:j+pad_w+1]
                if np.all(region[kernel]):
                    result[i, j] = 255
        return result

    # 2. 膨胀（Dilation）
    def dilation(img, kernel):
        h, w = img.shape
        kh, kw = kernel.shape
        pad_h, pad_w = kh//2, kw//2

        result = np.zeros_like(img)
        for i in range(pad_h, h-pad_h):
            for j in range(pad_w, w-pad_w):
                region = img[i-pad_h:i+pad_h+1, j-pad_w:j+pad_w+1]
                if np.any(region[kernel]):
                    result[i, j] = 255
        return result

    # 应用操作
    eroded = erosion(img, kernel)
    dilated = dilation(img, kernel)

    # 开运算（先腐蚀后膨胀）
    opened = dilation(erosion(img, kernel), kernel)
```

```python
    # 1. 顶帽变换（Top Hat）
    def top_hat(img, kernel_size=5):
        from scipy.ndimage import minimum_filter
        opened = minimum_filter(img, size=kernel_size)
        return img - opened

    # 2. 黑帽变换（Black Hat）
    def black_hat(img, kernel_size=5):
        from scipy.ndimage import maximum_filter
        closed = maximum_filter(img, size=kernel_size)
        return closed - img

    # 3. 骨架化（简化版）
    def skeletonize(img):
        """简化骨架化"""
        from scipy.ndimage import binary_erosion
        skeleton = np.zeros_like(img)
        img_binary = img > 0

        while np.any(img_binary):
            eroded = binary_erosion(img_binary)
            temp = img_binary - eroded
            skeleton = skeleton | temp
            img_binary = eroded.copy()

        return skeleton.astype(np.uint8) * 255

    # 应用操作
    top = top_hat(img, 5)
    black = black_hat(img, 5)
    skeleton = skeletonize(img)
```

```python
def compute_hog_simple(img, cell_size=8, bins=9):
    """简化版HOG特征计算"""

    # 1. 计算梯度
    grad_x = convolve(img.astype(np.float32), np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]))
    grad_y = convolve(img.astype(np.float32), np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]]))

    magnitude = np.sqrt(grad_x**2 + grad_y**2)
    direction = np.arctan2(grad_y, grad_x) * 180 / np.pi
    direction = direction % 180  # 0-180度

    # 2. 分块计算直方图
    h, w = img.shape
    hog_features = []

    for i in range(0, h - cell_size, cell_size):
        for j in range(0, w - cell_size, cell_size):
            # 提取cell
            cell_mag = magnitude[i:i+cell_size, j:j+cell_size]
            cell_dir = direction[i:i+cell_size, j:j+cell_size]

            # 计算直方图
            hist = np.zeros(bins)
            bin_edges = np.linspace(0, 180, bins + 1)

            for b in range(bins):
                mask = (cell_dir >= bin_edges[b]) & (cell_dir < bin_edges[b+1])
                hist[b] = np.sum(cell_mag[mask])

            hog_features.append(hist)

    return np.array(hog_features)
```

```python
def compute_texture_features(img, window_size=5):
    """计算纹理特征"""

    h, w = img.shape
    features = []

    for i in range(0, h - window_size, window_size):
        for j in range(0, w - window_size, window_size):
            window = img[i:i+window_size, j:j+window_size]

            # 1. 均值
            mean = np.mean(window)

            # 2. 标准差
            std = np.std(window)

            # 3. 对比度
            contrast = np.max(window) - np.min(window)

            # 4. 平滑度（基于二阶导数）
            grad_x = np.gradient(window, axis=1)
            grad_y = np.gradient(window, axis=0)
            smoothness = 1 / (1 + np.mean(grad_x**2 + grad_y**2))

            features.append([mean, std, contrast, smoothness])

    return np.array(features)
```

```python
def practice3_feature_matching():

    print("练习3：特征匹配")
    print("="*60)

    # 创建模板
    template = np.zeros((50, 50), dtype=np.uint8)
    template[10:40, 10:40] = 255

    # 创建目标（带变换）
    target = np.zeros((100, 100), dtype=np.uint8)
    target[30:60, 30:60] = 255  # 相同形状，不同位置

    # 添加噪声
    template = add_gaussian_noise(template, sigma=10)
    target = add_gaussian_noise(target, sigma=10)

    # 提取特征点（使用Shi-Tomasi）
    corners_t = shi_tomasi_detector(template, max_corners=10, quality_level=0.01)
    corners_target = shi_tomasi_detector(target, max_corners=20, quality_level=0.01)

    if len(corners_t) == 0 or len(corners_target) == 0:
        print("未检测到足够特征点")
        return

    # 简单描述子：局部灰度均值
    def get_descriptor(img, point, size=5):
        x, y, _ = point
        pad = size // 2
        region = img[max(0, y-pad):y+pad+1, max(0, x-pad):x+pad+1]
        return np.mean(region)

    # 计算描述子
    desc_t = [get_descriptor(template, c) for c in corners_t]
    desc_target = [get_descriptor(target, c) for c in corners_target]

    # 匹配（最近邻）
    matches = []
    for i, dt in enumerate(desc_t):
        distances = [abs(dt - dt_target) for dt_target in desc_target]
        min_idx = np.argmin(distances)
        if distances[min_idx] < 20:  # 阈值
            matches.append((i, min_idx, distances[min_idx]))

    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(12, 4))

    axes[0].imshow(template, cmap='gray', vmin=0, vmax=255)
    tx = [c[0] for c in corners_t]
    ty = [c[1] for c in corners_t]
    axes[0].scatter(tx, ty, c='red', s=50, marker='o')
    axes[0].set_title(f'模板\n({len(corners_t)}个特征)')
    axes[0].axis('off')

    axes[1].imshow(target, cmap='gray', vmin=0, vmax=255)
    gx = [c[0] for c in corners_target]
    gy = [c[1] for c in corners_target]
    axes[1].scatter(gx, gy, c='blue', s=50, marker='o')
    axes[1].set_title(f'目标\n({len(corners_target)}个特征)')
    axes[1].axis('off')

    # 显示匹配
    axes[2].imshow(target, cmap='gray', vmin=0, vmax=255, alpha=0.5)
    for match in matches:
        t_idx, g_idx, dist = match
        tx, ty = corners_t[t_idx][:2]
        gx, gy = corners_target[g_idx][:2]
        # 调整位置（模板在目标中的相对位置）
        offset_x, offset_y = 30, 30
        axes[2].plot([tx+offset_x, gx], [ty+offset_y, gy], 'g-', linewidth=2)
        axes[2].scatter([tx+offset_x, gx], [ty+offset_y, gy], c='green', s=30)

    axes[2].set_title(f'匹配结果\n({len(matches)}个匹配)')
    axes[2].axis('off')

    plt.tight_layout()
    plt.show()

    print(f"\n特征匹配结果：")
    print(f"模板特征：{len(corners_t)}")
    print(f"目标特征：{len(corners_target)}")
    print(f"成功匹配：{len(matches)}")

    if matches:
        print("\n匹配详情：")
        for i, (t_idx, g_idx, dist) in enumerate(matches[:5]):
            print(f"  匹配{i+1}: 模板点{t_idx} ↔ 目标点{g_idx}, 距离={dist:.2f}")
```
