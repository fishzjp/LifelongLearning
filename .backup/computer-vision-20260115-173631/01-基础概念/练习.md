```python
import numpy as np
import matplotlib.pyplot as plt

def exercise1_image_operations():
    """
    练习1: 图像的矩阵表示与操作
    """
    print("="*60)
    print("练习1: 图像的矩阵表示与操作")
    print("="*60)

    # 1. 创建自定义图案图像
    print("\n1. 创建自定义图案...")
    image = np.zeros((100, 100), dtype=np.uint8)

    # 创建棋盘格图案
    square_size = 20
    for i in range(0, 100, square_size):
        for j in range(0, 100, square_size):
            if (i // square_size + j // square_size) % 2 == 0:
                image[i:i+square_size, j:j+square_size] = 255

    print(f"   图像形状: {image.shape}")
    print(f"   数据类型: {image.dtype}")
    print(f"   像素值范围: [{image.min()}, {image.max()}]")

    # 2. 矩阵运算 - 亮度调整
    print("\n2. 亮度调整...")

    # 增加亮度
    bright_image = np.clip(image.astype(np.int16) + 50, 0, 255).astype(np.uint8)

    # 降低亮度
    dark_image = np.clip(image.astype(np.int16) - 50, 0, 255).astype(np.uint8)

    print(f"   原图均值: {image.mean():.2f}")
    print(f"   增亮后均值: {bright_image.mean():.2f}")
    print(f"   变暗后均值: {dark_image.mean():.2f}")

    # 3. 矩阵运算 - 对比度调整
    print("\n3. 对比度调整...")

    # 增加对比度
    high_contrast = np.clip(image.astype(np.float32) * 1.5, 0, 255).astype(np.uint8)

    # 降低对比度
    low_contrast = np.clip(image.astype(np.float32) * 0.5, 0, 255).astype(np.uint8)

    print(f"   原图标准差: {image.std():.2f}")
    print(f"   高对比度标准差: {high_contrast.std():.2f}")
    print(f"   低对比度标准差: {low_contrast.std():.2f}")

    # 4. 提取和修改特定区域
    print("\n4. 区域操作...")

    # 提取左上角区域
    top_left = image[:50, :50]

    # 提取右下角区域
    bottom_right = image[50:, 50:]

    # 在图像中心创建一个圆形遮罩
    center = (50, 50)
    radius = 30
    y, x = np.ogrid[:100, :100]
    mask = (x - center[0])**2 + (y - center[1])**2 <= radius**2

    # 应用圆形遮罩
    masked_image = image.copy()
    masked_image[~mask] = 0

    print(f"   左上角区域形状: {top_left.shape}")
    print(f"   圆形遮罩像素数: {mask.sum()}")

    # 5. 可视化所有结果
    print("\n5. 生成可视化...")
    fig, axes = plt.subplots(2, 4, figsize=(16, 8))

    axes[0, 0].imshow(image, cmap='gray')
    axes[0, 0].set_title('原始图像')
    axes[0, 0].axis('off')

    axes[0, 1].imshow(bright_image, cmap='gray')
    axes[0, 1].set_title('增加亮度 (+50)')
    axes[0, 1].axis('off')

    axes[0, 2].imshow(dark_image, cmap='gray')
    axes[0, 2].set_title('降低亮度 (-50)')
    axes[0, 2].axis('off')

    axes[0, 3].imshow(high_contrast, cmap='gray')
    axes[0, 3].set_title('高对比度 (×1.5)')
    axes[0, 3].axis('off')

    axes[1, 0].imshow(low_contrast, cmap='gray')
    axes[1, 0].set_title('低对比度 (×0.5)')
    axes[1, 0].axis('off')

    axes[1, 1].imshow(top_left, cmap='gray')
    axes[1, 1].set_title('左上角区域')
    axes[1, 1].axis('off')

    axes[1, 2].imshow(bottom_right, cmap='gray')
    axes[1, 2].set_title('右下角区域')
    axes[1, 2].axis('off')

    axes[1, 3].imshow(masked_image, cmap='gray')
    axes[1, 3].set_title('圆形遮罩')
    axes[1, 3].axis('off')

    plt.tight_layout()
    plt.savefig('/tmp/exercise1_result.png', dpi=100, bbox_inches='tight')
    print("   结果已保存到: /tmp/exercise1_result.png")

    print("\n" + "="*60)
    print("练习1完成!")
    print("="*60)

    return {
        'original': image,
        'bright': bright_image,
        'dark': dark_image,
        'high_contrast': high_contrast,
        'low_contrast': low_contrast,
        'masked': masked_image
    }

# 运行练习
if __name__ == "__main__":
    results = exercise1_image_operations()
```

### 🎯 任务描述

实现多种RGB到灰度图像的转换方法：
1. 平均值法
2. 加权平均法（ITU-R BT.601标准）
3. 仅保留单通道
4. 比较不同方法的效果

### 💡 提示

**提示1 - 图像读取**:
```python
from PIL import Image
import numpy as np

# 读取彩色图像
img = Image.open('path/to/image.jpg')
img_array = np.array(img)  # shape: (H, W, 3)

# 分离通道
r = img_array[:, :, 0]
g = img_array[:, :, 1]
b = img_array[:, :, 2]
```

**提示2 - 转换公式**:
- 平均值法: `gray = (r + g + b) / 3`
- 加权平均法: `gray = 0.299*r + 0.587*g + 0.114*b`
- 最大值法: `gray = max(r, g, b)`

**提示3 - 为什么使用加权？**
人眼对绿色最敏感，红色次之，蓝色最不敏感

```python
import numpy as np
import matplotlib.pyplot as plt
from PIL import Image

def exercise2_rgb_to_grayscale():
    """
    练习2: RGB与灰度图像转换
    """
    print("="*60)
    print("练习2: RGB与灰度图像转换")
    print("="*60)

    # 1. 创建测试图像（如果没有真实图像）
    print("\n1. 创建测试彩色图像...")
    height, width = 100, 100
    color_image = np.zeros((height, width, 3), dtype=np.uint8)

    # 创建渐变图案
    for i in range(height):
        for j in range(width):
            # R通道：从左到右渐变
            color_image[i, j, 0] = int(255 * j / width)
            # G通道：从上到下渐变
            color_image[i, j, 1] = int(255 * i / height)
            # B通道：对角线渐变
            color_image[i, j, 2] = int(255 * (i + j) / (height + width))

    print(f"   彩色图像形状: {color_image.shape}")
    print(f"   R通道范围: [{color_image[:, :, 0].min()}, {color_image[:, :, 0].max()}]")
    print(f"   G通道范围: [{color_image[:, :, 1].min()}, {color_image[:, :, 1].max()}]")
    print(f"   B通道范围: [{color_image[:, :, 2].min()}, {color_image[:, :, 2].max()}]")

    # 2. 方法1: 平均值法
    print("\n2. 方法1: 平均值法...")
    gray_avg = np.mean(color_image, axis=2).astype(np.uint8)
    print(f"   公式: gray = (R + G + B) / 3")
    print(f"   灰度值范围: [{gray_avg.min()}, {gray_avg.max()}]")

    # 3. 方法2: 加权平均法（ITU-R BT.601）
    print("\n3. 方法2: 加权平均法 (ITU-R BT.601)...")
    gray_weighted = (
        0.299 * color_image[:, :, 0] +
        0.587 * color_image[:, :, 1] +
        0.114 * color_image[:, :, 2]
    ).astype(np.uint8)
    print(f"   公式: gray = 0.299*R + 0.587*G + 0.114*B")
    print(f"   灰度值范围: [{gray_weighted.min()}, {gray_weighted.max()}]")

    # 4. 方法3: 仅保留绿色通道
    print("\n4. 方法3: 仅保留绿色通道...")
    gray_green = color_image[:, :, 1].copy()
    print(f"   公式: gray = G")
    print(f"   灰度值范围: [{gray_green.min()}, {gray_green.max()}]")

    # 5. 方法4: 最大值法
    print("\n5. 方法4: 最大值法...")
    gray_max = np.max(color_image, axis=2)
    print(f"   公式: gray = max(R, G, B)")
    print(f"   灰度值范围: [{gray_max.min()}, {gray_max.max()}]")

    # 6. 方法5: 最小值法
    print("\n6. 方法5: 最小值法...")
    gray_min = np.min(color_image, axis=2)
    print(f"   公式: gray = min(R, G, B)")
    print(f"   灰度值范围: [{gray_min.min()}, {gray_min.max()}]")

    # 7. 计算各种方法的统计信息
    print("\n7. 方法对比分析...")
    methods = {
        '平均值法': gray_avg,
        '加权平均法': gray_weighted,
        '绿色通道': gray_green,
        '最大值法': gray_max,
        '最小值法': gray_min
    }

    print(f"{'方法':<12} {'均值':<10} {'标准差':<10} {'最小值':<8} {'最大值':<8}")
    print("-" * 50)
    for name, gray in methods.items():
        print(f"{name:<12} {gray.mean():<10.2f} {gray.std():<10.2f} "
              f"{gray.min():<8} {gray.max():<8}")

    # 8. 可视化对比
    print("\n8. 生成可视化对比...")
    fig, axes = plt.subplots(2, 4, figsize=(16, 8))

    # 第一行：原图和RGB通道
    axes[0, 0].imshow(color_image)
    axes[0, 0].set_title('原始彩色图像')
    axes[0, 0].axis('off')

    axes[0, 1].imshow(color_image[:, :, 0], cmap='Reds')
    axes[0, 1].set_title('R通道')
    axes[0, 1].axis('off')

    axes[0, 2].imshow(color_image[:, :, 1], cmap='Greens')
    axes[0, 2].set_title('G通道')
    axes[0, 2].axis('off')

    axes[0, 3].imshow(color_image[:, :, 2], cmap='Blues')
    axes[0, 3].set_title('B通道')
    axes[0, 3].axis('off')

    # 第二行：不同转换方法
    axes[1, 0].imshow(gray_avg, cmap='gray')
    axes[1, 0].set_title('平均值法')
    axes[1, 0].axis('off')

    axes[1, 1].imshow(gray_weighted, cmap='gray')
    axes[1, 1].set_title('加权平均法\n(推荐)')
    axes[1, 1].axis('off')

    axes[1, 2].imshow(gray_max, cmap='gray')
    axes[1, 2].set_title('最大值法')
    axes[1, 2].axis('off')

    axes[1, 3].imshow(gray_min, cmap='gray')
    axes[1, 3].set_title('最小值法')
    axes[1, 3].axis('off')

    plt.tight_layout()
    plt.savefig('/tmp/exercise2_result.png', dpi=100, bbox_inches='tight')
    print("   结果已保存到: /tmp/exercise2_result.png")

    # 9. 方法优缺点分析
    print("\n9. 各方法优缺点分析:")
    print("-" * 60)
    print("【平均值法】")
    print("  优点: 计算简单，各通道权重相同")
    print("  缺点: 未考虑人眼对不同颜色的感知差异")
    print("  适用: 快速预览，对准确性要求不高的场景")
    print()
    print("【加权平均法 (ITU-R BT.601)】")
    print("  优点: 符合人眼感知，结果自然（推荐使用）")
    print("  缺点: 计算稍复杂")
    print("  适用: 大多数应用场景，标准转换方法")
    print()
    print("【仅保留绿色通道】")
    print("  优点: 计算最快，保留人眼最敏感的通道")
    print("  缺点: 丢失红蓝信息，可能失真")
    print("  适用: 快速处理，绿色占主导的图像")
    print()
    print("【最大值法】")
    print("  优点: 保留最亮的像素，图像较亮")
    print("  缺点: 可能丢失暗部细节")
    print("  适用: 需要突出亮部的场景")
    print()
    print("【最小值法】")
    print("  优点: 保留最暗的像素，突出暗部")
    print("  缺点: 图像整体偏暗")
    print("  适用: 特殊艺术效果")

    print("\n" + "="*60)
    print("练习2完成!")
    print("="*60)

    return methods

# 运行练习
if __name__ == "__main__":
    results = exercise2_rgb_to_grayscale()
```

```python
import numpy as np
import matplotlib.pyplot as plt
import cv2

def exercise3_geometric_transformations():
    """
    练习3: 图像几何变换
    """
    print("="*60)
    print("练习3: 图像几何变换")
    print("="*60)

    # 1. 创建测试图像
    print("\n1. 创建测试图像...")
    img = np.zeros((200, 200, 3), dtype=np.uint8)

    # 绘制彩色方块
    img[50:150, 50:150] = [255, 0, 0]  # 红色方块
    img[75:125, 75:125] = [0, 255, 0]  # 绿色方块
    img[90:110, 90:110] = [0, 0, 255]  # 蓝色方块

    # 添加文字标记
    cv2.putText(img, 'TOP', (80, 40), cv2.FONT_HERSHEY_SIMPLEX,
                0.8, (255, 255, 255), 2)
    cv2.putText(img, 'BOTTOM', (60, 180), cv2.FONT_HERSHEY_SIMPLEX,
                0.8, (255, 255, 255), 2)

    height, width = img.shape[:2]
    print(f"   图像尺寸: {width}×{height}")

    # 2. 平移变换
    print("\n2. 平移变换...")
    tx, ty = 30, 20  # 向右30像素，向下20像素
    M_translate = np.float32([[1, 0, tx], [0, 1, ty]])
    img_translated = cv2.warpAffine(img, M_translate, (width, height),
                                    borderMode=cv2.BORDER_REPLICATE)
    print(f"   平移量: ({tx}, {ty})")

    # 3. 旋转变换
    print("\n3. 旋转变换...")
    angle = 45  # 逆时针45度
    center = (width // 2, height // 2)
    scale = 1.0
    M_rotate = cv2.getRotationMatrix2D(center, angle, scale)
    img_rotated = cv2.warpAffine(img, M_rotate, (width, height),
                                  borderMode=cv2.BORDER_REPLICATE)
    print(f"   旋转角度: {angle}°")
    print(f"   旋转中心: {center}")

    # 4. 缩放变换
    print("\n4. 缩放变换...")
    scale_factor = 0.5
    new_width = int(width * scale_factor)
    new_height = int(height * scale_factor)
    img_scaled = cv2.resize(img, (new_width, new_height),
                            interpolation=cv2.INTER_AREA)
    print(f"   缩放因子: {scale_factor}")
    print(f"   新尺寸: {new_width}×{new_height}")

    # 5. 翻转变换
    print("\n5. 翻转变换...")
    img_flip_h = cv2.flip(img, 1)  # 水平翻转
    img_flip_v = cv2.flip(img, 0)  # 垂直翻转
    img_flip_hv = cv2.flip(img, -1)  # 同时翻转
    print("   水平翻转: 完成")
    print("   垂直翻转: 完成")
    print("   同时翻转: 完成")

    # 6. 组合变换：旋转+缩放
    print("\n6. 组合变换（旋转+缩放）...")
    angle = 30
    scale = 0.8
    M_combined = cv2.getRotationMatrix2D(center, angle, scale)
    img_combined = cv2.warpAffine(img, M_combined, (width, height),
                                   borderMode=cv2.BORDER_REPLICATE)
    print(f"   旋转角度: {angle}°, 缩放因子: {scale}")

    # 7. 可视化所有变换
```
